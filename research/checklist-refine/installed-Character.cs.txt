	public void addCampaign(game.net.data.StoC.Campaign c)
	{
		if (!campaigns_.ContainsKey((ushort)c.campaign.idcampaign))
		{
			return;
		}
		Campaign campaign = campaigns_[(ushort)c.campaign.idcampaign];
		foreach (db_character_campaign_chest chest in c.chests)
		{
			if (campaign.chests_.ContainsKey((byte)chest.idchest))
			{
				campaign.chests_[(byte)chest.idchest].make(chest, campaign);
			}
		}
		foreach (db_character_campaign_troop troop in c.troops)
		{
			if (campaign.troops_.ContainsKey((byte)troop.idtroop))
			{
				campaign.troops_[(byte)troop.idtroop].make(troop, campaign, campaign.anecdote);
			}
		}
		campaign.make(c.campaign);
	}

	public Artifact addArtifacts(game.net.data.StoC.Artifact a)
	{
		Artifact artifact = new Artifact(a);
		artifacts[artifact.idartifact_template] = artifact;
		if (artifact.idgeneral_template != ushort.MaxValue)
		{
			if (Global.character_.form_generals_.ContainsKey(artifact.idgeneral_template))
			{
				FormGeneral formGeneral = Global.character_.form_generals_[artifact.idgeneral_template];
				if (formGeneral != null)
				{
					formGeneral.artifact = artifact;
					formGeneral.CalcAllParam();
				}
			}
			else
			{
				ushort idgeneral_template = artifact.idgeneral_template;
				for (int i = 0; i < Global.character_.GeneralsOutSquad.Count; i++)
				{
					if (Global.character_.GeneralsOutSquad[i].id == idgeneral_template)
					{
						Global.character_.GeneralsOutSquad[i].artifact = artifact;
						break;
					}
				}
			}
		}
		if (artifact.template.isspecial > 0 && !arti_gens.ContainsKey(artifact.idartifact_template))
		{
			int num = -1;
			Hashtable templates = Global.getTemplates("general_origin");
			foreach (DictionaryEntry item in templates)
			{
				_general_origin general_origin = item.Value as _general_origin;
				if (general_origin.origin == artifact.template.origin && (num == -1 || num > general_origin.form))
				{
					arti_gens[artifact.idartifact_template] = general_origin.idgeneral_template;
					num = general_origin.form;
				}
			}
		}
		return artifact;
	}

	public void handle_genswitch(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCSwitchGen_0x182 sCSwitchGen_0x = JSProtoBuf.Serializer.Deserialize<SCSwitchGen_0x182>(stream);
		addGeneral((ushort)sCSwitchGen_0x.newid, sCSwitchGen_0x.insquad, sCSwitchGen_0x.weaponid, sCSwitchGen_0x.armorid, sCSwitchGen_0x.artifactid, sCSwitchGen_0x.amuletid);
		foreach (Formation value in formations_.Values)
		{
			for (int i = 0; i < value.spotgen.Count; i++)
			{
				if (value.spotgen[i] == (int)sCSwitchGen_0x.oldid)
				{
					value.spotgen[i] = (int)sCSwitchGen_0x.newid;
				}
			}
		}
	}

	public void handelFormUnlock(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCFormationUnlock_0x135 sCFormationUnlock_0x = JSProtoBuf.Serializer.Deserialize<SCFormationUnlock_0x135>(stream);
		if (!formations_[(int)sCFormationUnlock_0x.id].availability)
		{
			formations_[(int)sCFormationUnlock_0x.id].availability = true;
			if (formations_[(int)sCFormationUnlock_0x.id].spotgen.Count == 0)
			{
				formations_[(int)sCFormationUnlock_0x.id].spotgen.Add(0);
			}
			else
			{

	private uint CheckRepLevel()
	{
		List<_artifact_template> list = new List<_artifact_template>();
		list.Clear();
		Hashtable templates = Global.getTemplates("artifact_template");
		foreach (DictionaryEntry item in templates)
		{
			_artifact_template artifact_template = (_artifact_template)item.Value;
			if (artifact_template.canbuy == 1)
			{
				list.Add(item.Value as _artifact_template);
			}
		}
		list.Sort((_artifact_template _a, _artifact_template _b) => _a.reputation.CompareTo(_b.reputation));
		for (int num = 0; num < list.Count; num++)
		{
			if (Global.character_.reputation < list[num].reputation)
			{
				uint result = 0u;
				if (num >= 1)
				{
					result = list[num - 1].idartifact_template;
				}
				return result;
			}
		}
		return 0u;
	}

	public bool isGeneralVisible(ushort genid)
	{
		if (Global.getTemplate("general_template", genid) is _general_template general_template)
		{
			switch (general_template.special_event)
			{
			case 0:
				return false;
			case 1:
				switch (genid)
				{
				case 211:
				{
					bool flag = false;
					foreach (GeneralsSimple item in GeneralsOutSquad)
					{
						if (item.id == 22)
						{
							flag = true;
						}
					}
					if (form_generals_.ContainsKey(22) || flag)
					{
						return true;
					}
					return false;
				}
				case 234:
					return true;
				case 239:
					if (Global.character_.campaigns_.ContainsKey(72))
					{
						if (Global.character_.campaigns_[72].finished)
						{
							return true;
						}
						return false;
					}
					return false;
				case 37:

	public void updateQuest(SCQuestOne onequest)
	{
		ushort key = (ushort)onequest.info.idquest;
		if (onequest.state == 0)
		{
			if (!quests.ContainsKey(key))
			{
				addQuests(onequest.info);
			}
		}
		else if (onequest.state == 1)
		{
			if (!quests.ContainsKey(key))
			{
				return;
			}
			quests[key].condition_param = onequest.info.condition_param;
			quests[key].condition_flag = onequest.info.condition_flag;
			if (onequest.info.condition_flag)
			{
				if (!trace_finished.ContainsKey(key))
				{
					trace_finished[key] = quests[key];
				}
				if (trace_main.ContainsKey(key))
				{
					trace_main.Remove(key);
				}
				else if (trace_sub.ContainsKey(key))
				{
					trace_sub.Remove(key);
				}
				else if (trace_daily.ContainsKey(key))
				{
					trace_daily.Remove(key);
				}
			}
		}
		else if (onequest.state == 2 && quests.ContainsKey(key))
		{
			quests[key].state = true;
			if (trace_finished.ContainsKey(key))
			{
				trace_finished.Remove(key);
			}
		}
	}

	public void tosort_member(string sort_by)
	{
		member_sorted.Sort(delegate(GuildMember a, GuildMember b)
		{
			switch (sort_by)
			{
			case "today_contribution":
				if (!member_sort_asc)
				{
					if (a.today_contribution != b.today_contribution)
					{
						return b.today_contribution.CompareTo(a.today_contribution);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.today_contribution != b.today_contribution)
				{
					return a.today_contribution.CompareTo(b.today_contribution);
				}
				return a.name.CompareTo(b.name);
			case "total_contribution":
				if (!member_sort_asc)
				{
					if (a.total_contribution != b.total_contribution)
					{
						return b.total_contribution.CompareTo(a.total_contribution);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.total_contribution != b.total_contribution)
				{
					return a.total_contribution.CompareTo(b.total_contribution);
				}
				return a.name.CompareTo(b.name);
			case "level":
				if (!member_sort_asc)
				{
					if (a.level != b.level)
					{
						return b.level.CompareTo(a.level);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.level != b.level)
				{
					return a.level.CompareTo(b.level);
				}
				return a.name.CompareTo(b.name);
			case "class":
				if (!member_sort_asc)
				{
					if (a.class_type != b.class_type)
					{
						return b.class_type.CompareTo(a.class_type);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.class_type != b.class_type)
				{
					return a.class_type.CompareTo(b.class_type);
				}
				return a.name.CompareTo(b.name);
			case "logout":
				if (!member_sort_asc)
				{
					if (a.isonline != b.isonline)
					{
						return b.isonline.CompareTo(a.isonline);
					}
					if (a.last_logout != b.last_logout)
					{
						return b.last_logout.CompareTo(a.last_logout);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.isonline != b.isonline)
				{
					return a.isonline.CompareTo(b.isonline);
				}
				if (a.last_logout != b.last_logout)
				{
					return a.last_logout.CompareTo(b.last_logout);
				}
				return a.name.CompareTo(b.name);
			default:
				if (!member_sort_asc)
				{
					return b.name.CompareTo(a.name);
				}
				return a.name.CompareTo(b.name);
			}
		});
	}

	private void tosort_guilds(string sort_by)
	{

		{
			switch (sort_by)
			{
			case "member":
				if (!guild_sort_asc)
				{
					if (a.size != b.size)
					{
						return b.size.CompareTo(a.size);

				}
				return a.name.CompareTo(b.name);
			case "level":
				if (!guild_sort_asc)
				{
					if (a.level != b.level)
					{
						return b.level.CompareTo(a.level);
					}
					return b.name.CompareTo(a.name);
				}
				if (a.level != b.level)
				{
					return a.level.CompareTo(b.level);
				}
				return a.name.CompareTo(b.name);
			default:
				return b.name.CompareTo(a.name);
			}
		});
	}

	private int Getdays(double seconds)
	{
		int num = 0;
		DateTime dateTime = new DateTime(1970, 1, 1).AddSeconds(seconds);
		num = ((dateTime.Month + 1 <= 12) ? (num + DateTime.DaysInMonth(dateTime.Year, dateTime.Month + 1)) : (num + DateTime.DaysInMonth(dateTime.Year + 1, 1)));
		if (dateTime.Month + 2 > 12)
		{
			return num + DateTime.DaysInMonth(dateTime.Year + 1, 1);
		}
		return num + DateTime.DaysInMonth(dateTime.Year, dateTime.Month + 2);
	}

	{

		{
		case 54u:
			key = (uint)(type * 10000000 + param0 * 1000 + param1);

		case 53u:

						list.Add(general_origin2.idgeneral_template);

			}
			for (ushort num = 0; num < list.Count; num++)
			{
				uint key2 = list[num] * 10000;
				if (treasuryItems.ContainsKey(key2))
				{
					treasuryItems.Remove(key2);
				}
			}

		}
	}

	public void addGodListSlot(uint id)
	{
		if (!(Global.getTemplate("god_list_slot_template", id) is _god_list_slot_template god_list_slot_template))
		{
			return;
		}
		GodListSlot godListSlot = null;
		uint num = 0u;
		foreach (GodListSlot value in Global.character_.godlist.m_slots.Values)
		{
			_god_list_slot_template god_list_slot_template2 = Global.getTemplate("god_list_slot_template", value.idslot) as _god_list_slot_template;
			if (god_list_slot_template.slot_type == god_list_slot_template2.slot_type && god_list_slot_template.slot_rank > god_list_slot_template2.slot_rank)
			{
				num = value.idslot;
				break;
			}
		}
		if (num != 0)
		{
			uint general_id = godlist.m_slots[num].general_id;
			uint l = godlist.m_slots[num].level;
			godListSlot = new GodListSlot((ushort)id, l, general_id);
			godlist.m_slots.Remove(num);
		}
		else
		{
			godListSlot = new GodListSlot((ushort)id);
		}
		godlist.m_slots.Add(godListSlot.idslot, godListSlot);
	}
