	protected override void AssetLoaded(string AssetName)
	{
		foreach (_Treasure value in treasuresDic.Values)
		{
			if (value != null && AssetName == value.getIconName())
			{
				value.setTexIcon(BaseAsset.LoadTexture(value.getIconName()));
			}
		}
		foreach (List<_Treasure> value2 in Global.character_.m_newYearCheckinAwards.Values)
		{
			foreach (_Treasure item in value2)
			{
				if (item != null && AssetName == item.getIconName())

			}
		}
	}

	private uint getSignInLevel()
	{
		uint num = 0u;
		uint num2 = 5u;
		uint num3 = Global.character_.checkin_times;
		if (num3 % 30 == 0 && Global.character_.checkin_today == 0)
		{
			num3 = Global.character_.checkin_times + 1;
		}
		num = ((num3 % 30 != 0) ? (Global.character_.checkin_times / 30) : (Global.character_.checkin_times / 30 - 1));
		if (num3 == 0)
		{
			num = 0u;
		}
		return num;
	}

	private void drawSignInBonus()
	{
		int num = 1;
		foreach (int treasuresDicKey in treasuresDicKeys)
		{
			if (treasuresDic.ContainsKey(treasuresDicKey))
			{
				if (treasuresDic[treasuresDicKey].texIcon != null)
				{
					__BaseUI.Label(__BaseUI._rect(103 + 62 * (num % 8), 38 + 54 * (num / 8), 30f, 30f), new GUIContent(treasuresDic[treasuresDicKey].texIcon, "bonus" + treasuresDicKey), __BaseUI.normalStyle, GUI.color);
				}
				else
				{
					AddAssetSub(treasuresDic[treasuresDicKey].getIconName(), isimage: false);
					StartLoadingSub();
				}
				if (!showNextMonthBonus() && treasuresDicKey <= getSignInTimes())
				{

					__BaseUI.Label(__BaseUI._rect(103 + 62 * (num % 8) + 14, 38 + 54 * (num / 8) + 12, 23f, 22f), string.Empty, ui_yiqian);
				}
				num++;
			}
		}
	}

	public override void drawTooltips(string tip)
	{
		base.drawTooltips(tip);
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (tip.Contains("newyearbonus:"))
		{
			string text = tip.Replace("newyearbonus:", string.Empty);
			string[] array = text.Split(',');
			uint key = uint.Parse(array[0]);
			int num = int.Parse(array[1]);
			if (Global.character_.m_newYearCheckinAwards.ContainsKey(key) && Global.character_.m_newYearCheckinAwards[key].Count > num)
			{
				Global.character_.m_newYearCheckinAwards[key][num].PrepTooltip();
			}
		}
		else if (tip.Contains("bonus"))
		{
			int key2 = int.Parse(tip.Replace("bonus", string.Empty));
			if (treasuresDic.ContainsKey(key2))
			{
				treasuresDic[key2].PrepTooltip();
			}
		}
		else if (tip != null)
		{
			tooltipUI.ShowTooltipContent(string.Empty, tip, 200, __BaseUI._fontSize);
		}
	}

	private void setTreasures()
	{
		int num = 0;
		Hashtable templates = Global.getTemplates("checkin_award_once");
		foreach (object value in templates.Values)
		{
			if (value is _checkin_award_once checkin_award_once && num < checkin_award_once.day)
			{
				num = checkin_award_once.day;
			}
		}
		uint num2 = Global.character_.checkin_times;
		if (showNextMonthBonus())
		{
			num2 = Global.character_.checkin_times + 1;
		}
		if (num2 <= num)
		{
			int num3 = ((num2 % 30 != 0) ? ((int)Global.character_.checkin_times / 30) : ((int)Global.character_.checkin_times / 30 - 1));
			if (num2 == 0)
			{
				num3 = 0;
			}
			Hashtable templates2 = Global.getTemplates("checkin_award_once");
			foreach (object value2 in templates2.Values)
			{
				if (value2 is _checkin_award_once checkin_award_once2)
				{
					int num4 = ((checkin_award_once2.day % 30 != 0) ? (checkin_award_once2.day / 30) : (checkin_award_once2.day / 30 - 1));
					int num5 = ((checkin_award_once2.day % 30 != 0) ? (checkin_award_once2.day % 30) : 30);
					if (num3 == num4)
					{
						treasuresDicKeys.Add(num5);
						_Treasure treasure = new _Treasure();
						treasure.make((byte)checkin_award_once2.treasure_type, checkin_award_once2.param0, checkin_award_once2.param1);
						AddAssetSub(treasure.getIconName(), isimage: false);
						treasuresDic[num5] = treasure;
					}
				}
			}
		}
		else
		{
			Hashtable templates3 = Global.getTemplates("checkin_award");
			foreach (object value3 in templates3.Values)
			{
				if (value3 is _checkin_award checkin_award)
				{
					treasuresDicKeys.Add(checkin_award.day);
					_Treasure treasure2 = new _Treasure();
					treasure2.make((byte)checkin_award.treasure_type, checkin_award.param0, checkin_award.param1);
					AddAssetSub(treasure2.getIconName(), isimage: false);
					treasuresDic[checkin_award.day] = treasure2;
				}
			}
		}
		StartLoadingSub();
		treasuresDicKeys.Sort();
	}

	private bool showNextMonthBonus()
	{
		if (Global.character_.checkin_times % 30 == 0 && Global.character_.checkin_today == 0)
		{
			return true;
		}
		return false;
	}

	private uint getSignInTimes()
	{
		uint num = Global.character_.checkin_times;
		if (num != 0)
		{
			num = ((Global.character_.checkin_times % 30 != 0) ? (Global.character_.checkin_times % 30) : 30u);
		}
		return num;
	}

	public void cs_all_bonus_get_new()
	{
		if (!sended)
		{
			sended = true;
			CSAllBonusGetNew_0x394 pkt = new CSAllBonusGetNew_0x394();
			RootUI.m_client.Send(916, pkt, sc_all_bonus_get_new);
		}
	}

	public void sc_all_bonus_get_new(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		sended = false;
		SCAllBonusGetNew_0x394 sCAllBonusGetNew_0x = JSProtoBuf.Serializer.Deserialize<SCAllBonusGetNew_0x394>(stream);
		if (sCAllBonusGetNew_0x.arena)
		{
			foreach (_character_arena_bonus bonuse in Global.character_.arena.bonuses)
			{
				if (bonuse.time == sCAllBonusGetNew_0x.arenatime)
				{
					Global.character_.arena.bonuses.Remove(bonuse);
					break;
				}
			}
		}
		if (sCAllBonusGetNew_0x.salary)
		{
			Global.character_.patrol_last_time = (int)sCAllBonusGetNew_0x.salarytime;
		}
		if (sCAllBonusGetNew_0x.pointrace)
		{
			foreach (_character_pointrace_bonus bonuse2 in Global.character_.pointrace.bonuses)
			{
				if (bonuse2.time == sCAllBonusGetNew_0x.pointracetime)
				{
					Global.character_.pointrace.bonuses.Remove(bonuse2);
					break;
				}
			}
		}
		if (sCAllBonusGetNew_0x.checkin)
		{
			Global.character_.checkin_times++;
			Global.character_.checkin_today = 1;
		}
		List<_Treasure> list = new List<_Treasure>();
		foreach (Treasure bonuse3 in sCAllBonusGetNew_0x.bonuses)
		{
			_Treasure treasure = new _Treasure();
