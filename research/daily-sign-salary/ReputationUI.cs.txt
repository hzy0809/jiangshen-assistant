	protected override void AssetLoaded(string AssetName)
	{
		if (AssetName.Contains("item"))
		{
			Hashtable templates = Global.getTemplates("artifact_template");
			foreach (object value in templates.Values)
			{
				_artifact_template artifact_template = (_artifact_template)value;
				_artifact_forge_info artifact_forge_info = Database.Artifact.get_forge_info(artifact_template.idartifact_template, 0);
				if (artifact_forge_info != null && AssetName == "item" + artifact_forge_info.icon)
				{
					dictTex[artifact_template.idartifact_template] = BaseAsset.LoadTexture(AssetName);
					return;
				}
			}
		}
		foreach (_Treasure value2 in spBonuses.Values)
		{
			if (value2 != null && AssetName == value2.getIconName())
			{
				value2.setTexIcon(BaseAsset.LoadTexture(AssetName));
			}
		}
	}

	public void cs_reputation_up()
	{
		if (!sended)
		{

	}

	public void sc_reputation_up(MemoryStream stream, object arg)
	{
		sended = false;
		if (stream == null)
		{
			return;
		}
		SCReputationUp_0x332 sCReputationUp_0x = JSProtoBuf.Serializer.Deserialize<SCReputationUp_0x332>(stream);
		switch (sCReputationUp_0x.operate)
		{
		case SCReputationUp_0x332.Operate.OK:
			setCurRep();
			Global.character_.reputation_level++;
			if (!(Global.getTemplate("reputation", Global.character_.reputation_level) is _reputation reputation))
			{
				break;
			}
			switch (reputation.trigger)
			{
			case 1:
			{
				int result4 = 0;
				if (int.TryParse(reputation.param, out result4) && Global.getTemplate("artifact_template", (uint)result4) is _artifact_template)
				{
					Database.Artifact artifact = new Database.Artifact((ushort)result4);
					if (!Global.character_.artifacts.ContainsKey(artifact.idartifact_template))
					{
						Global.character_.artifacts.Add(artifact.idartifact_template, artifact);
					}
					SpecialGetUI component3 = GetComponent<SpecialGetUI>();
					component3.ShowSPGet_Artifact(artifact.idartifact_template);
				}
				break;
			}
			case 6:
			{
				string param = reputation.param;
				string s = param.Substring(0, 1);
				string s2 = param.Substring(2, 1);
				int result = 0;
				int result2 = 0;
				if (int.TryParse(s, out result) && int.TryParse(s2, out result2))
				{
					SpecialGetUI component = GetComponent<SpecialGetUI>();
					component.ShowSPGet_Form((uint)result);
				}
				break;
			}
			case 7:
			{
				string param2 = reputation.param;
				int result3 = 0;
				if (int.TryParse(param2, out result3))
				{
					SpecialGetUI component2 = GetComponent<SpecialGetUI>();
					component2.ShowSPGet_Spot((uint)result3);
				}
				break;
			}
			case 8:
				if (sCReputationUp_0x.bonus != null)
				{
					List<_Treasure> list = new List<_Treasure>();
					_Treasure treasure = new _Treasure();
					treasure.make(sCReputationUp_0x.bonus);
					list.Add(treasure);
					BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
					battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, EnableAll);
					battleResultUI.MyLoading();
				}
				break;
			}
			ScrollControl.doscroll(scroll_index, 92);
			break;
		}
	}

	public void sc_reputation_salary(MemoryStream stream, object arg)
	{
		sended = false;
		if (stream != null)
		{
			SCReputationSalary_0x333 sCReputationSalary_0x = JSProtoBuf.Serializer.Deserialize<SCReputationSalary_0x333>(stream);
			switch (sCReputationSalary_0x.operate)
			{
			case SCReputationSalary_0x333.Operate.OK:
			{
