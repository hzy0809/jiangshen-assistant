	public void cs_online_get()
	{
		CSOnlineGet_0x317 pkt = new CSOnlineGet_0x317();
		RootUI.m_client.Send(791, pkt, sc_online_get);
	}

	public void sc_online_get(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCOnlineGet_0x317 sCOnlineGet_0x = JSProtoBuf.Serializer.Deserialize<SCOnlineGet_0x317>(stream);
		switch (sCOnlineGet_0x.operate)
		{
		case SCOnlineGet_0x317.Operate.OK:
		{
			List<_Treasure> list = new List<_Treasure>();
			for (int i = 0; i < sCOnlineGet_0x.type_id.Count; i++)
			{
				if (sCOnlineGet_0x.type_id[i] != -1)
				{
					_Treasure treasure = new _Treasure();
					treasure.make((byte)sCOnlineGet_0x.type_id[i], 0, (int)sCOnlineGet_0x.amount[i]);
					list.Add(treasure);
					__BaseUI.drawStateBubble(treasure.name + sCOnlineGet_0x.amount[i]);
				}
			}
			BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
			battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, EnableAll);
			battleResultUI.MyLoading();
			MainUI.set_online_showTip(tip: true);
			Global.character_.OnlineGift.got++;
			Global.character_.OnlineGift.setNextTime();
			break;
		}
		case SCOnlineGet_0x317.Operate.Error:
		{
			_online_awards_template online_awards_template = Global.getTemplate("online_awards_template", (uint)(Global.character_.OnlineGift.got + 1)) as _online_awards_template;
			MainUI.set_online_showTip(tip: false);
			if (online_awards_template != null)
			{
				MyLoading();
				return;
			}
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20430u), rmg);
			break;
		}
		}
		MainUI.SetSubUI(null);
	}
