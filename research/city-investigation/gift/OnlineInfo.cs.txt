using Database;

public class OnlineInfo
{
	public int refresh_time;

	public int got;

	public int next_get_time;

	public void refresh()
	{
		got = 0;
		if (Global.getTemplate("online_awards_template", (uint)(got + 1)) is _online_awards_template online_awards_template)
		{
			next_get_time = (int)(TimeUtil.Now + online_awards_template.time * 60);
		}
	}

	public void setNextTime()
	{
		if (Global.getTemplate("online_awards_template", (uint)(got + 1)) is _online_awards_template online_awards_template)
		{
			next_get_time = (int)(TimeUtil.Now + online_awards_template.time * 60);
		}
	}

	public bool isNextAvailable()
	{
		if (Global.getTemplate("online_awards_template", (uint)(got + 1)) is _online_awards_template)
		{
			return true;
		}
		return false;
	}
}
