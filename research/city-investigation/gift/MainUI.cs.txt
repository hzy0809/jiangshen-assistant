	private class TrialMazeStamina
	{
		public int stamina;

		public int stamina_limit;

		public int refresh_timestamp;

		public int refresh_inverval;

		public TrialMazeStamina(SCTrialMazeStaminaRefresh_0x2B3 pkt)
		{
			stamina = (int)pkt.stamina;
			stamina_limit = (int)pkt.stamina_limit;
			refresh_timestamp = (int)pkt.refresh_timestamp;
			refresh_inverval = (int)pkt.refresh_interval;
		}

		public void Update()
		{
			if (TimeUtil.Now > refresh_inverval + refresh_timestamp)
			{
				int num = (int)(TimeUtil.Now - refresh_timestamp) / refresh_inverval;
				stamina += num;
				if (stamina > stamina_limit)
				{
					stamina = stamina_limit;
				}
				refresh_timestamp += refresh_inverval * num;
			}
		}
	}

	public static bool IsSubUI(BaseAsset ui)
	{
		return (m_subui == ui) ? true : false;
	}

	public static void SetSubUI(__BaseUI ui)
	{
		if (m_subui == ui)
		{
			return;
		}
		LoadScreen component = GameObject.Find("/LoadUI").GetComponent<LoadScreen>();
		if (m_subui != null)
		{
			component.GetCurUI().BroadcastMessage("CloseBuildingInsideUI", SendMessageOptions.DontRequireReceiver);
			if (ui == null)
			{
				component.GetCurUI().BroadcastMessage("ResetCapitalCamera", SendMessageOptions.DontRequireReceiver);
				component.GetCurUI().BroadcastMessage("EnableMouseDown", SendMessageOptions.DontRequireReceiver);
			}
			m_subui.Hide();
		}
		m_subui = ui;
	}

	public static bool handle_sub_ui(__BaseUI ui, string id)
	{
		if (IsSubUI(ui))
		{
			SetSubUI(null);
			return false;
		}
		__BaseUI.subui = id;
		SetSubUI(ui);
		return true;
	}

	public static BaseUI GetSubUI()
	{

	public void expandUI(string name)
	{
		foreach (_ButtonMain item in btnBottom)
		{
			if (item.btnSubs == null)
			{
				continue;
			}
			foreach (_ButtonMain btnSub in item.btnSubs)
			{
				if (!(btnSub.name == name))
				{
					continue;
				}
				if (item.expand)
				{
					item.expand = false;
					item.current = false;
					item.state = _ButtonMain.STATE.NONE;
					item.subWidth = 0f;
				}
				item.expand = true;
				item.current = true;
				item.state = _ButtonMain.STATE.EXPANDING1;
				{
					foreach (_ButtonMain item2 in btnBottom)
					{
						if (item != item2 && item2.expand)
						{
							item2.expand = false;
							item2.current = false;
							item2.state = _ButtonMain.STATE.NONE;
							item2.subWidth = 0f;
						}
					}
					return;
				}
			}
		}
	}

	public void handle_button(uint idbutton)
	{
		_ui_button ui_button = Global.getTemplate("ui_button", idbutton) as _ui_button;
		foreach (_ButtonMain item in btnBottom)
		{
			if (item.btnSubs == null)
			{
				continue;
			}
			foreach (_ButtonMain btnSub in item.btnSubs)
			{
				if (btnSub.name == ui_button.name)
				{
					if (btnSub.handler != null)
					{
						btnSub.handler();
					}
					return;
				}
			}
		}
	}

	public Rect getButtonRect(string name)
	{
		foreach (_ButtonMain item in btnBottom)
		{
			if (item.name == name)
			{
				return item.getRect();
			}
			if (item.btnSubs == null)
			{
				continue;
			}
			foreach (_ButtonMain btnSub in item.btnSubs)
			{
				if (btnSub.name == name)
				{
					return btnSub.getRect();
				}
			}
		}
		foreach (_ButtonMain item2 in btnTop)
		{
			if (BaseAsset._PHONEUI && !BaseAsset.isipad)
			{
				continue;
			}
			if (item2.name == name)
			{
				if (item2.btnMain_type != _ButtonMain.btnType.NOEXPANDING && topBarMood == TOPBARMOOD.SHRINK)
				{
					return __BaseUI._rect(Screen.width - 260 - 87 - 62, 22f, 59f, 59f);
				}
				return item2.getRect();
			}
			if (item2.btnSubs == null)
			{
				continue;
			}
			foreach (_ButtonMain btnSub2 in item2.btnSubs)
			{
				if (btnSub2.name == name)
				{
					return btnSub2.getRect();
				}
			}
		}
		foreach (_ButtonMain item3 in btnBottomRight)
		{
			if (item3.name == name)
			{
				return item3.getRect();
			}
			if (item3.btnSubs == null)
			{
				continue;
			}
			foreach (_ButtonMain btnSub3 in item3.btnSubs)
			{
				if (btnSub3.name == name)
				{
					return btnSub3.getRect();
				}
			}
		}
		return new Rect(0f, 0f, 0f, 0f);
	}

	public bool isButtonShrinked(string name)
	{
		if (topBarMood == TOPBARMOOD.NORMAL)
		{
			return false;
		}
		if (topBarMood == TOPBARMOOD.SHRINK)
		{
			foreach (_ButtonMain item in btnTop)
			{
				if (item.name == name)
				{
					return item.btnMain_type != _ButtonMain.btnType.NOEXPANDING;
				}
			}
		}
		return false;
	}

	public void _convoy()
	{
		uint num = TimeUtil.Now - TimeUtil.ThisDay;
		if (!(Global.getTemplate("convoy_param", 1u) is _convoy_param convoy_param))
		{
			return;
		}
		if ((num >= convoy_param.begin_time1 && num <= convoy_param.end_time1 + 1800) || (num >= convoy_param.begin_time2 && num <= convoy_param.end_time2 + 1800))
		{
			DisableAll();
			GameObject.Find("Convoy/ConvoyUI").GetComponent<ConvoyUI>().cs_convoy_refresh(isOpenUI: true);
		}
		else if (convoy_param.auto_vip <= Global.character_.VIPLEVEL)
		{
			if (num > convoy_param.end_time2 + 1800)
			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20839u) + __BaseUI.load_str(20869u), null, showAutoConvoy, _bCancel: true);
			}
			else
			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20837u) + __BaseUI.load_str(20869u), null, showAutoConvoy, _bCancel: true);
			}
		}
		else if (num > convoy_param.end_time2 + 1800)
		{
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20839u), null);
		}
		else
		{
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20837u), null);
		}
	}

	private int _resource_soul_count()
	{
		if (!checkUI("resource_soul"))
		{
			return 0;
		}
		_general_soul_start_mode[] array = new _general_soul_start_mode[3]
		{
			Global.getTemplate("general_soul_start_mode", 1u) as _general_soul_start_mode,
			Global.getTemplate("general_soul_start_mode", 2u) as _general_soul_start_mode,
			Global.getTemplate("general_soul_start_mode", 3u) as _general_soul_start_mode
		};
		byte[] array2 = new byte[3]
		{
			Global.character_.generalsoul_blue_times,
			Global.character_.generalsoul_purple_times,
			Global.character_.generalsoul_orange_times
		};
		int num = 0;
		if (Global.character_.generalsoul_blue_times < array[0].free_times)
		{
			num += array[0].free_times - Global.character_.generalsoul_blue_times;
		}
		if ((Global.character_.level >= array[1].require_level || Global.character_.VIPLEVEL >= array[1].nolimit_viplevel) && Global.character_.generalsoul_purple_times < array[1].free_times)
		{
			num += array[1].free_times - Global.character_.generalsoul_purple_times;
		}
		if (Global.character_.level >= array[2].require_level && Global.character_.VIPLEVEL >= array[2].nolimit_viplevel && Global.character_.generalsoul_orange_times < array[2].free_times)
		{
			num += array[2].free_times - Global.character_.generalsoul_orange_times;
		}
		return num;
	}

	private bool check_new_system_tip()
	{
		Hashtable templates = Global.getTemplates("new_system_tip");
		if (Global.getTemplate("new_system_tip", (uint)templates.Count) is _new_system_tip new_system_tip)
		{
			if (new_system_tip.idskill == 0)
			{
				Database.Campaign campaign = Global.character_.getCampaign(new_system_tip.idcampaign);
				if (!campaign.finished && campaign.troops_.ContainsKey(new_system_tip.idtroop))
				{
					if (campaign.troops_[new_system_tip.idtroop].killed)
					{
						return false;
					}
				}
				else if (campaign.finished)
				{
					return false;
				}
			}
			else
			{
				if (Global.character_.skills.ContainsKey(new_system_tip.param))
				{
					return false;
				}
				foreach (Skill value in Global.character_.skills.Values)
				{
					if (Global.getTemplate("skill", value.idskill) is _skill skill && skill.former_skill == new_system_tip.param)
					{
						return false;
					}
				}
			}
			return true;
		}
		return false;
	}

	private void draw_new_system_tip()
	{
		_new_system_tip new_system_tip = null;
		Hashtable templates = Global.getTemplates("new_system_tip");
		for (int i = 1; i <= templates.Count; i++)
		{
			new_system_tip = Global.getTemplate("new_system_tip", (uint)i) as _new_system_tip;
			if (new_system_tip.idskill == 0)
			{
				Database.Campaign campaign = Global.character_.getCampaign(new_system_tip.idcampaign);
				if (campaign != null && !campaign.finished && campaign.troops_.ContainsKey(new_system_tip.idtroop) && !campaign.troops_[new_system_tip.idtroop].killed)
				{
					break;
				}
			}
			else
			{
				if (Global.character_.skills.ContainsKey(new_system_tip.param))
				{
					continue;
				}
				bool flag = false;
				foreach (Skill value in Global.character_.skills.Values)
				{
					if (Global.getTemplate("skill", value.idskill) is _skill skill && skill.former_skill == new_system_tip.param)
					{
						flag = true;
						break;
					}
				}
				if (!flag)
				{
					break;
				}
			}
		}
		if (new_system_tip == null)
		{
			return;
		}
		if (btnNewSystemTipUI.Button(IsEnable(), __BaseUI._rect(0f, 0f, 261f, 72f), new GUIContent(string.Empty, string.Empty)))
		{
			NewSystemTipUI component = BaseUI.GetRootUI().GetComponent<NewSystemTipUI>();
			if (handle_sub_ui(component, string.Empty))
			{
				component.MyLoading();
			}
		}
		if (new_system_tip.level - Global.character_.level > 0)
		{
			__BaseUI.Label(__BaseUI._rect(97f, 15f, 126f, 22f), string.Empty, ui_zi_zs);
			__BaseUI.OutlinedLabel(__BaseUI._rect(132f, 15f, 41f, 22f), __BaseUI.load_str(20088u, new_system_tip.level - Global.character_.level), __BaseUI._textCenter, __BaseUI._clrWhite, 16);
		}
		else
		{
			__BaseUI.Label(__BaseUI._rect(97f, 15f, 126f, 22f), string.Empty, ui_zi_jjkf);
		}
		__BaseUI.OutlinedLabel(__BaseUI._rect(100f, 37f, 200f, 22f), new_system_tip.desc, __BaseUI._textNLeft, __BaseUI._clrYellowLight, 14);
		drawTipIcon(__BaseUI._rect(30f, 6f, 59f, 59f), new_system_tip);
		__BaseUI.Label(__BaseUI._rect(18f, -6f, 84f, 84f), string.Empty, ui_di);
		addFrameAnimation("NewSystemTipMain", tipLightstyList, anime_timer, GUITOScreen(__BaseUI._rect(18f, -6f, 84f, 84f)), anime_idx, isCirculate: true);
	}

	private void sc_finish_quest(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			Enable();
			return;
		}
		SCQuestFinish_0x108 sCQuestFinish_0x = JSProtoBuf.Serializer.Deserialize<SCQuestFinish_0x108>(stream);
		if (!sCQuestFinish_0x.state)
		{
			return;
		}
		uint id = (uint)arg;
		_quest quest = Global.getTemplate("quest", id) as _quest;
		BonusGetUI bonusGetUI = BaseUI.GetBonusGetUI();
		List<_Treasure> list = new List<_Treasure>();
		foreach (_quest_reward reward in quest.m_rewards)
		{
			_Treasure treasure = new _Treasure();
			treasure.make(reward.reward_type, reward.param0, reward.param1);
			list.Add(treasure);
		}
		bonusGetUI.SetBonus(rmg, this, list, Enable);
		bonusGetUI.MyLoading();
	}
