	private void makeBars()
	{
		if (m_reputationList == null)
		{
			m_reputationList = new List<int>();
			Hashtable templates = Global.getTemplates("artifact_template");
			foreach (DictionaryEntry item in templates)
			{
				_artifact_template artifact_template = (_artifact_template)item.Value;
				if (artifact_template.canbuy == 1)
				{
					m_reputationList.Add((item.Value as _artifact_template).reputation);
				}
			}
			m_reputationList.Sort((int _a, int _b) => _a.CompareTo(_b));
		}
		m_expStart = Global.character_.experience;
		m_soulStart = Global.character_.practice;
		m_expGet = 0;
		m_soulGet = 0;
		m_expLevelUp = false;
		m_expRequire = Global.character_.getExpRequire();
		m_expLevel = Global.character_.level;
		m_soulRequire = Global.character_.NextStarTechPractice;
		tianfu_soul = (m_barCount = 0);
		m_bonusCount = 0;
		if (mine_msg != string.Empty)
		{
			height = 160f;
		}
		else
		{
			height = 110f;
		}
		if (m_vctBonuses != null)
		{
			foreach (_Treasure vctBonuse in m_vctBonuses)
			{
				if (vctBonuse.name == "experience")
				{
					if (m_expGet == 0)
					{
						m_barCount++;
					}
					m_expGet += vctBonuse.param[1];
				}
				else if (vctBonuse.name == "practice")
				{
					if (m_soulGet == 0)
					{
						m_barCount++;
					}
					m_soulGet += vctBonuse.param[1];
				}
				else
				{
					m_bonusCount++;
				}
			}
		}
		if ((float)m_expGet > m_expRequire - (float)m_expStart)
		{
			m_expGetShow = (int)(m_expRequire - (float)m_expStart);
		}
		else
		{
			m_expGetShow = m_expGet;
		}
		if ((float)m_soulGet > m_soulRequire - (float)m_soulStart)
		{
			m_soulGetShow = (int)(m_soulRequire - (float)m_soulStart);
		}
		else
		{
			m_soulGetShow = m_soulGet;
		}
		if (m_soulGetShow < 0)
		{
			m_soulGetShow = 0;
		}
		if (m_barCount > 0)
		{
			m_barHeight = 30f;
			m_barHeight += m_barCount * 30 + (m_barCount - 1) * 10;
		}
		else
		{
			m_barHeight = 0f;
		}
		if (m_bonusCount > 0)
		{
			m_bonusHeight = 30f;
			m_bonusHeight += (float)(Math.Ceiling((float)m_bonusCount / 2f) * 50.0 + (Math.Ceiling((float)m_bonusCount / 2f) - 1.0) * 16.0);
		}
		else
		{
			m_bonusHeight = 0f;
		}
		if (m_barHeight > 0f && m_bonusHeight > 0f)
		{
			height += 20f;
		}
		if (m_bonusCount <= 10)
		{
			height += m_barHeight + m_bonusHeight;
		}
		else
		{
			height += m_barHeight + 330f;
		}
		if (texTitle != null)
		{
			height += texSize.y;
		}
		if (desc != null)
		{
			height += desc.height;
		}
		if (m_clit != null)
		{
			height += 60f;
		}
		if (autoClose)
		{
			height -= 40f;
		}
	}

	public void myVisible()
	{
		m_startTime = DateTime.Now;
		Visible();
		Enable();
		if (m_vctBonuses != null)
		{
			if (Global.character_.gold < 0)
			{
				List<_Treasure> list = new List<_Treasure>();
				foreach (_Treasure vctBonuse in m_vctBonuses)
				{
					if (vctBonuse.name == "gold")
					{
						if (!m_bNotIgnoreGold)
						{
							list.Add(vctBonuse);
						}
						else if (Global.character_.gold < 0)
						{
							Global.character_.gold += vctBonuse.param[1];
							vctBonuse.bAdd = false;
						}
					}
				}
				foreach (_Treasure item in list)
				{
					m_vctBonuses.Remove(item);
				}
			}
			m_bNotIgnoreGold = false;
			foreach (_Treasure vctBonuse2 in m_vctBonuses)
			{
				if (vctBonuse2.param[1] > 0)
				{
					__BaseUI.drawStateBubble(vctBonuse2.name + vctBonuse2.param[1]);
					if (vctBonuse2.bAdd)
					{
						_treasure_type treasureTypeTemplate = Global.getTreasureTypeTemplate(vctBonuse2.type);
						if (treasureTypeTemplate.type == 2)
						{
							int param = (vctBonuse2.type << 8) | vctBonuse2.param[0];
							if (vctBonuse2.add_to_treasury)
							{
								Global.character_.addParamToTreasury(vctBonuse2.type, vctBonuse2.param[0], vctBonuse2.param[1]);
							}
							else
							{
								Global.character_.addParam("material", param, vctBonuse2.param[1]);
							}
						}
						else if (vctBonuse2.add_to_treasury)
						{
							Global.character_.addParamToTreasury(vctBonuse2.type, vctBonuse2.param[0], vctBonuse2.param[1]);
						}
						else
						{
							Global.character_.addParam(vctBonuse2.name, vctBonuse2.param[0], vctBonuse2.param[1]);
						}
					}
				}
				if (vctBonuse2.name == "general")
				{
					_general_get_id = vctBonuse2.param[0];
					RootBase.add_UIFunc(GeneralGet);
				}
			}
		}
		if (m_iTreasureCount > 0)
		{
			Global.character_.treasure_count = (byte)m_iTreasureCount;
			Global.character_.treasure_filter_count = 0;
			Global.character_.treasure_filter = false;
		}
	}
