	private void cs_building_produce_0x486(byte id)
	{
		CSBuildingProduce_0x486 cSBuildingProduce_0x = new CSBuildingProduce_0x486();
		cSBuildingProduce_0x.id = building.id;
		cSBuildingProduce_0x.idproduct = id;
		cSBuildingProduce_0x.is_immediate = Global.character_.building_immediate;
		RootUI.m_client.Send(1158, cSBuildingProduce_0x, sc_building_produce_0x486);
	}

	private void sc_building_produce_0x486(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCBuildingProduce_0x486 sCBuildingProduce_0x = JSProtoBuf.Serializer.Deserialize<SCBuildingProduce_0x486>(stream);
		if (sCBuildingProduce_0x.result != 0)
		{
			return;
		}
		Global.character_.progressBuildingQuest((byte)sCBuildingProduce_0x.idproduct);
		Building building = Global.character_.getBuilding((byte)sCBuildingProduce_0x.id);
		building.setProduct((byte)sCBuildingProduce_0x.idproduct);
		building.lucky = sCBuildingProduce_0x.lucky;
		building.start_time = sCBuildingProduce_0x.start_time;
		if (sCBuildingProduce_0x.gold_consume != 0)
		{
			Global.character_.gold -= (int)sCBuildingProduce_0x.gold_consume;
			__BaseUI.drawStateBubble("gold-" + sCBuildingProduce_0x.gold_consume);
		}
		if (sCBuildingProduce_0x.mana_consume != 0)
		{
			Global.character_.mana -= (int)sCBuildingProduce_0x.mana_consume;
			__BaseUI.drawStateBubble("mana-" + sCBuildingProduce_0x.mana_consume);
		}
		if (sCBuildingProduce_0x.is_immediate)
		{
			BuildingHarvestUI.harvest((byte)sCBuildingProduce_0x.id, sCBuildingProduce_0x.start_time, new delegate_OnClose(ProduceImmediate));
			return;
		}
		GameObject.Find("Capital/CapitalUI").GetComponent<CapitalUI>().RefreshBuildingProducts();
		MyClose();
		if (m_bSub)
		{
			EnableAll();
		}
		else
		{
			Enable();
		}
	}

	private void ProduceImmediate()
	{
		EnableAll();
		Building building = this.building;
		building.idproduct = 0;
		if (building.product != null && building.product.mesh != null)
		{
			Object.Destroy(building.product.mesh);
		}
		building.product = null;
	}

	public void cs_building_clear_upgrade_cd(byte id)
	{
		if (!send)
		{
			send = true;
			CSBuildingClearUpgradeCD_0x3A0 cSBuildingClearUpgradeCD_0x3A = new CSBuildingClearUpgradeCD_0x3A0();
			cSBuildingClearUpgradeCD_0x3A.idbuilding = id;
			RootUI.m_client.Send(928, cSBuildingClearUpgradeCD_0x3A, sc_building_clear_upgrade_cd, id);
		}
	}

	public void sc_building_clear_upgrade_cd(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		byte key = (byte)arg;
		send = false;
		SCBuildingClearUpgradeCD_0x3A0 sCBuildingClearUpgradeCD_0x3A = JSProtoBuf.Serializer.Deserialize<SCBuildingClearUpgradeCD_0x3A0>(stream);
		uint result = sCBuildingClearUpgradeCD_0x3A.result;
		if (result != 0 && result == 1)
		{
			if (Global.character_.buildings.ContainsKey(key))
			{
				Building building = Global.character_.buildings[key];
				building.upgrade_finish = TimeUtil.Now - 1;
			}
			Global.character_.gold -= (int)sCBuildingClearUpgradeCD_0x3A.goldcost;
			__BaseUI.drawStateBubble("gold-" + sCBuildingClearUpgradeCD_0x3A.goldcost);
			MainUI.SetSubUI(null);
		}
	}
