	public static void wantHarvest(Building b, delegate_OnClose onClose, bool bBubble)
	{
		if (_bHarvesting)
		{
			return;
		}
		int capacityMax = Global.character_.capacityMax;
		int num = 0;
		uint time = b.product.time;
		if (b.start_time + time > TimeUtil.Now)
		{
			num = (int)(time + b.start_time - TimeUtil.Now);
		}
		if (num > time)
		{
			num = (int)time;
		}
		_building_upgrade building_upgrade = new _building_upgrade();
		Hashtable templates = Global.getTemplates("building_upgrade");
		foreach (DictionaryEntry item in templates)
		{
			_building_upgrade building_upgrade2 = item.Value as _building_upgrade;
			if (building_upgrade2.idbuilding == b.idbuilding && building_upgrade2.level == b.level)
			{
				building_upgrade = building_upgrade2;
				break;
			}
		}
		int num2 = (int)((float)b.product.Output * (float)building_upgrade.product_addition / 100f + (float)b.product.Output);
		int num3 = (int)((time - num) * num2 / time);
		_bBubble = bBubble;
		if (b.product.mana_output != 0 && num3 + Global.character_.mana > capacityMax)
		{
			_idbuilding = b.id;
			_onClose = onClose;
			BaseAsset.lds.GetCurUI().ToDisable();
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20176u), null, WantHarvestOK, _bCancel: true);
		}
		else
		{
			cs_building_harvest_0x482(b.id, onClose);
		}
	}

		num *= ((!building.lucky) ? 1f : 1.5f);
		Hashtable templates = Global.getTemplates("building_upgrade");
		foreach (DictionaryEntry item in templates)
		{
			_building_upgrade building_upgrade = item.Value as _building_upgrade;
			if (building_upgrade.idbuilding == building.idbuilding && building_upgrade.level == building.level)
			{
				num *= (float)building_upgrade.product_addition / 100f + 1f;
				break;
			}
		}
		List<_Treasure> list = new List<_Treasure>();
		makeOutput(4, (uint)Mathf.RoundToInt((float)product.money_output * num), list);
		makeOutput(30, (uint)Mathf.RoundToInt((float)product.stone_output * num), list);
		int capacityMax = Global.character_.capacityMax;
		makeOutput(9, (uint)Mathf.RoundToInt((float)product.spirit_output * num), list);
		makeOutput(10, (uint)Mathf.RoundToInt((float)product.enchant_output * num), list);
		uint num2 = (uint)Mathf.RoundToInt((float)product.mana_output * num);
		if (num2 != 0 && Global.character_.mana < capacityMax)
		{
			if (Global.character_.mana + num2 > capacityMax)
			{
				num2 = (uint)Mathf.RoundToInt((float)capacityMax - (float)Global.character_.mana);
			}
			makeOutput(5, num2, list);
		}
		building.start_time = start_time;
		if (list.Count > 0)
		{
			if (!_bBubble)
			{
				BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
				battleResultUI.Set(null, string.Empty, string.Empty, 0, 0, list, arg as delegate_OnClose);
				battleResultUI.MyLoading();
				return;
			}
			BonusGetUI component = BaseUI.GetRootUI().GetComponent<BonusGetUI>();
			int num3 = 0;
			Vector3 position = building.mesh.transform.position;
			position.y = building.mesh.collider.bounds.center.y + building.mesh.collider.bounds.extents.y;
			position = Camera.main.WorldToScreenPoint(position);
			position.y = (float)Screen.height - position.y;
			position.x -= 30f;
			position.y -= 50f;
			foreach (_Treasure item2 in list)
			{
				BonusGetUI.getBonus(item2);
				Texture2D texIcon = null;
				if (item2.type == 4)
				{
					texIcon = __BaseUI._iconMoney;
				}
				else if (item2.type == 8)
				{
					texIcon = __BaseUI._iconExp;
				}
				else if (item2.type == 30)
				{
					texIcon = __BaseUI._iconStone;
				}
				else if (item2.type == 9)
				{
					texIcon = __BaseUI._iconFruit;
				}
				else if (item2.type == 10)
				{
					texIcon = __BaseUI._iconEnchant;
				}
				else if (item2.type == 5)
				{
					texIcon = __BaseUI._iconMana;
				}
				RootBase._drawBubble(item2.param[1], __BaseUI._rect(position.x, position.y + (float)(num3 * 20), 200f, 30f), texIcon, 1f, -5f, -20f);
				num3++;
			}
			if (building.data.automatic == 1)
			{
				building.idproduct = 0;
				if (building.product.mesh != null)
				{
					UnityEngine.Object.Destroy(building.product.mesh);
				}
				building.product = null;
			}
			if (arg != null)
			{
				(arg as delegate_OnClose)();
			}
		}
		else if (arg != null)
		{
			(arg as delegate_OnClose)();
		}
	}

	public void sc_building_gathering_0x485(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCBuildingGathering_0x485 sCBuildingGathering_0x = JSProtoBuf.Serializer.Deserialize<SCBuildingGathering_0x485>(stream);
		if (sCBuildingGathering_0x.result != 0)
		{
			return;
		}
		Building building = Global.character_.getBuilding((byte)sCBuildingGathering_0x.id);
		building.gathering = sCBuildingGathering_0x.gathering;
		building.start_time = sCBuildingGathering_0x.start_time;
		building.playAnim();
		if (building.product == null)
		{
			return;
		}
		if (building.gathering)
		{
			if (building.product.spirit_output != 0)
			{
				Global.character_.woodYield += building.product.spirit_output * 3600 / building.product.time;
			}
			if (building.product.enchant_output != 0)
			{
				Global.character_.mineYield += building.product.enchant_output * 3600 / building.product.time;
			}
			if (building.product.mana_output != 0)
			{
				Global.character_.manaYield += building.product.mana_output * 3600 / building.product.time;
			}
		}
		else
		{
			if (building.product.spirit_output != 0)
			{
				Global.character_.woodYield -= building.product.spirit_output * 3600 / building.product.time;
			}
			if (building.product.enchant_output != 0)
			{
				Global.character_.mineYield -= building.product.enchant_output * 3600 / building.product.time;
			}
			if (building.product.mana_output != 0)
			{
				Global.character_.manaYield -= building.product.mana_output * 3600 / building.product.time;
			}
		}
	}
