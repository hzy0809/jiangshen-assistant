	public void MyLoading()
	{
		m_buildings.Clear();
		foreach (Building value in Global.character_.buildings.Values)
		{
			if (value.m_products.Count > 0)
			{
				m_buildings.Add(value);
			}
		}
		Hashtable templates = Global.getTemplates("building");
		foreach (DictionaryEntry item in templates)
		{
			_building building = item.Value as _building;
			if (building.open_level <= Global.character_.level && Global.character_.limitBuildingCount(building) && building.type != 1)
			{
				Building building2 = new Building(building);
				building2.notbuild = true;
				m_buildings.Add(building2);
			}
		}
		foreach (Building building3 in m_buildings)
		{
			if (building3.product != null)
			{
				AddAssetSub("product" + building3.product.icon);
			}
		}
		SortBuilding();
		AddAsset("ui_buildingall");
		CollectingStart(bCollecting: false);
		StartLoading();
	}

	protected override void AssetLoaded(string AssetName)

		}
	}

	public void SortBuilding()
	{
		m_timeRefresh = 0f;
		List<Building> list = new List<Building>();
		foreach (Building building in m_buildings)
		{
			int num = 0;
			bool flag = false;
			foreach (Building item in list)
			{
				int num2 = CompareBuilding(building, item);
				if (num2 == -1)
				{
					list.Insert(num, building);
					flag = true;
					break;
				}
				num++;
			}
			if (!flag)
			{
				list.Add(building);
			}
		}
		m_buildings = list;
	}

	private int CompareBuilding(Building a, Building b)
	{
		if (a.canHarvest() && b.canHarvest())
		{
			return (b.data.open_level <= a.data.open_level) ? 1 : (-1);
		}
		if (a.canHarvest())
		{
			return -1;
		}
		if (b.canHarvest())
		{
			return 1;
		}
		if (a.canProduce() && b.canProduce())
		{
			return (b.data.open_level <= a.data.open_level) ? 1 : (-1);
		}
		if (a.canProduce())
		{
			return -1;
		}
		if (b.canProduce())
		{
			return 1;
		}
		if (a.isProducing() && b.isProducing())
		{
			return (a.start_time + a.product.time <= b.start_time + b.product.time) ? 1 : (-1);
		}
		if (a.isProducing())
		{
			return -1;
		}
		if (b.isProducing())
		{
			return 1;
		}
		if (a.build_time > TimeUtil.Now && b.build_time > TimeUtil.Now)
		{
			return (a.build_time <= b.build_time) ? 1 : (-1);
		}
		if (a.build_time > TimeUtil.Now)
		{
			return -1;
		}
		if (b.build_time > TimeUtil.Now)
		{
			return 1;
		}
		if (a.notbuild && b.notbuild)
		{
			return (b.data.open_level <= a.data.open_level) ? 1 : (-1);
		}
		if (a.notbuild)
		{
			return -1;
		}
		if (b.notbuild)
		{
			return 1;
		}
		return (a.cd_time <= b.cd_time) ? 1 : (-1);
	}

			if (idx % 2 == 0)
			{
				rc.y += 23f;
			}
			else
			{
				rc.y -= 23f;
				rc.x += 95f;
			}
			idx++;
			__BaseUI.EndGroup();
