	public static Rect DrawArrowMessage(Vector2 point, ArrowFace direction, Vector2 rc_size)
	{
		if (rc_size.x < 32f)
		{
			rc_size.x = 32f;
		}
		Rect rect = default(Rect);
		Rect result = new Rect
		{
			width = rc_size.x,
			height = rc_size.y
		};
		if (direction == ArrowFace.Down)
		{
			rect.x = point.x - rc_size.x / 2f - (float)tga_kuang_lpart.border.left;
			rect.y = point.y + 5f - (float)tga_kuang_down.border.bottom - rc_size.y - (float)tga_kuang_down.border.top;
			rect.width = (float)tga_kuang_lpart.border.left + rc_size.x + (float)tga_kuang_rpart.border.right;
			rect.height = (float)tga_kuang_down.border.top + rc_size.y + (float)tga_kuang_down.border.bottom;
			result.x = point.x - rc_size.x / 2f;
			result.y = rect.y + (float)tga_kuang_down.border.top;
			Label(new Rect(point.x - 16f, rect.y, 32f, rect.height), string.Empty, tga_kuang_down);
			Label(new Rect(rect.x, rect.y, point.x - 16f - rect.x, rect.height), string.Empty, tga_kuang_lpart);
			Label(new Rect(point.x + 16f, rect.y, rect.x + rect.width - point.x - 16f, rect.height), string.Empty, tga_kuang_rpart);
		}
		return result;
	}

	public static Rect DrawArrowMessage(Rect rc, ArrowFace face, int width)
	{
		if (rc == new Rect(0f, 0f, 0f, 0f))
		{
			return new Rect(0f, 0f, 0f, 0f);
		}
		int num = 32;
		int num2 = 139;
		width += 110;
		Rect arrowrc = default(Rect);
		switch (face)
		{
		case ArrowFace.Left:
		{
			int num3 = (int)rc.x + (int)rc.width - 10;
			int num4 = (int)rc.center.y - num2 / 2;
			arrowrc.x = num3;
			arrowrc.y = num4;
			arrowrc.width = width;
			arrowrc.height = num2;
			Label(arrowrc, string.Empty, tga_kuang_left);
			break;
		}
		case ArrowFace.Right:
			arrowrc.x = rc.x - (float)width + 10f;
			arrowrc.y = rc.center.y - (float)(num2 / 2);
			arrowrc.width = width;
			arrowrc.height = num2;
			Label(arrowrc, string.Empty, tga_kuang_right);
			break;
		case ArrowFace.Up:
			arrowrc.x = rc.center.x - (float)(num / 2);
			arrowrc.y = rc.y + rc.height - 10f;
			arrowrc.width = num;
			arrowrc.height = num2;
			Label(arrowrc, string.Empty, tga_kuang_up);
			DrawArrowParts(ref arrowrc, width);
			break;
		case ArrowFace.Down:
			arrowrc.x = rc.center.x - (float)(num / 2);
			arrowrc.y = rc.y - (float)num2 + 10f;
			arrowrc.width = num;
			arrowrc.height = num2;
			if (arrowrc.x < (float)width * 0.5f)
			{
				arrowrc.x += 30f;
			}
			Label(arrowrc, string.Empty, tga_kuang_down);
			DrawArrowParts(ref arrowrc, width);
			break;
		}
		return arrowrc;
	}

			FrameAnimation frameAnimation = new FrameAnimation(name, styList, intervalTime, rc, animeStartIdx, isCirculate, isreversal, pausetime);
			frameAnimationDic.Add(frameAnimation.m_name, frameAnimation);
			return frameAnimation;
		}
		if (frameAnimationDic[name].getRect() != rc)
		{
			setRectFrameAnimation(name, rc);
		}
		return null;
	}

	public void removeFrameAnimation(string name)
	{
		if (isContainsFrameAnimationKey(name))
		{
			frameAnimationDic.Remove(name);
		}
	}

	public void updateFrameAnimation()
	{
		Dictionary<string, FrameAnimation> dictionary = new Dictionary<string, FrameAnimation>();
		foreach (FrameAnimation value in frameAnimationDic.Values)
		{
			value.update();
			if (value.isFinished())
			{
				dictionary.Add(value.m_name, value);
			}
		}
		foreach (FrameAnimation value2 in dictionary.Values)
		{
			frameAnimationDic.Remove(value2.m_name);
		}
		dictionary.Clear();
		if (BaseAsset.lds != null && BaseAsset.lds.IsBattling())
		{
			clearFrameAnimation();
		}
	}

	public bool isContainsFrameAnimationKey(string name)
	{
		return frameAnimationDic.ContainsKey(name);
	}

	public void drawFrameAnimation()
	{
		updateFrameAnimation();
		foreach (FrameAnimation value in frameAnimationDic.Values)
		{
			value.draw();
		}
	}

	public void setRectFrameAnimation(string name, Rect rc)
	{
		foreach (KeyValuePair<string, FrameAnimation> item in frameAnimationDic)
		{
			if (item.Key == name)
			{
				item.Value.setRect(rc);
			}
		}
	}

	public void clearFrameAnimation()
	{
		frameAnimationDic.Clear();
	}

	public float drawTopResourceBar(Texture2D[] texZis, Texture2D[] texIcons, long[] resources, float offset)
	{
		int num = Math.Max(texZis.Length, texIcons.Length);
		float[] array = new float[num];
		for (int i = 0; i < num; i++)
		{
			array[i] = CalcSize(resources[i].ToString(), _numLeft, _fontSize).x;
		}
		setResourceBar(texZis, texIcons, resources, offset);
		BeginGroup(_rect((_rcCur.width - resource_bar_width) / 2f + offset, 15f, resource_bar_width, 36f));
		Label(_rect(0f, 0f, resource_bar_width, 36f), string.Empty, _box15Style);
		BeginGroup(_rect(spacingLeft, 0f, resource_bar_width - (float)spacingLeft - (float)spacingRight, 36f));
		float num2 = 0f;
		for (int j = 0; j < num; j++)
		{
			if (j < texZis.Length && texZis[j] != null)
			{
				GUI.DrawTexture(_rect(num2, (36 - texZis[j].height) / 2, texZis[j].width, texZis[j].height), texZis[j]);
				num2 += (float)texZis[j].width;
			}
			if (j < texIcons.Length && texIcons[j] != null)
			{
				GUI.DrawTexture(_rect(num2, (36 - texIcons[j].height) / 2, texIcons[j].width, texIcons[j].height), texIcons[j]);
				num2 += (float)texIcons[j].width;
			}
			Label(_rect(num2, 8f, array[j], 20f), resources[j].ToString(), _numLeft, _clrWhite, _fontSize);
			num2 += (float)spacingGroup + array[j];
		}
		EndGroup();
		EndGroup();
		return resource_bar_width;
	}

	public void setResourceBar(Texture2D[] texZis, Texture2D[] texIcons, long[] resources, float offset)
	{
		for (int i = 0; i < 10; i++)
		{
			resource_bubble_posX[i] = 0f;
		}
		int num = Math.Max(texZis.Length, texIcons.Length);
		float[] array = new float[num];
		resource_bar_offset = offset;
		resource_bar_width = spacingLeft + spacingGroup * (num - 1) + spacingRight;
		for (int j = 0; j < num; j++)
		{
			if (j > 0)
			{
				resource_bubble_posX[j] += resource_bubble_posX[j - 1] + (float)spacingGroup + array[j - 1];
			}
			if (j < texZis.Length && texZis[j] != null)
			{
				resource_bar_width += texZis[j].width;
				resource_bubble_posX[j] += texZis[j].width;
			}
			if (j < texIcons.Length && texIcons[j] != null)
			{
				resource_bar_width += texIcons[j].width;
				resource_bubble_posX[j] += texIcons[j].width;
			}
			array[j] = CalcSize(resources[j].ToString(), _numLeft, _fontSize).x;
			resource_bar_width += array[j];
		}
		for (int k = 0; k < 10; k++)
		{
			resource_bubble_posX[k] += (float)spacingLeft + ((float)Screen.width - w) / 2f + (w - resource_bar_width) / 2f + resource_bar_offset;
		}
	}

	public bool canBuySpItem(int treasure_type, int param0, int param1)
	{
		switch (treasure_type)
		{
		case 0:
			if (Global.character_.haveOneGeneral((uint)param0))
			{
				GetMessageBox().setMessage(string.Empty, load_str(9214u), rmg);
				return false;
			}
			return true;
		case 36:
			if (Global.character_.artifacts.ContainsKey((ushort)param0) || Global.character_.treasuryItems.ContainsKey((uint)(3600000 + param0)))
			{
				GetMessageBox().setMessage(string.Empty, load_str(9213u), rmg);
				return false;
			}
			return true;
		case 43:
			if (Global.character_.isAmuletFull())
			{
				GetMessageBox().setMessage(string.Empty, load_str(9212u), rmg);
				return false;
			}
			return true;
		case 54:
			if (Global.character_.isGemFull())
			{
				GetMessageBox().setMessage(string.Empty, load_str(21068u), rmg);
				return false;
			}
			return true;
		case 53:
			if (Global.getTemplate("spirit_skill", (uint)param1) is _spirit_skill spirit_skill)
			{
				for (int i = 0; i < Global.character_.spirit.skills.Count; i++)
				{
					if (Global.character_.spirit.skills[i].skill_type == spirit_skill.skill_type)
					{
						GetMessageBox().setMessage(string.Empty, load_str(34015u), rmg);
						return false;
					}
				}
			}
			return true;
		case 90:
			return canBuyGodslot(param0, needmsg: true);
		default:
			return true;
		}
	}

	public bool canBuyGodslot(int param0, bool needmsg)
	{
		_god_list_slot_template god_list_slot_template = Global.getTemplate("god_list_slot_template", (uint)param0) as _god_list_slot_template;
		uint num = 0u;
		uint num2 = 0u;
		if (god_list_slot_template != null)
		{
			num2 = god_list_slot_template.slot_type;
			num = god_list_slot_template.slot_rank;
		}
		foreach (object value in Global.getTemplates("god_list_slot_template").Values)
		{
			_god_list_slot_template god_list_slot_template2 = (_god_list_slot_template)value;
			if (god_list_slot_template2 != null && god_list_slot_template2.slot_type == num2 && (Global.character_.godlist.m_slots.ContainsKey(god_list_slot_template2.id) || Global.character_.treasuryItems.ContainsKey((uint)(9000000 + god_list_slot_template2.id))) && god_list_slot_template2.slot_rank >= num)
			{
				if (needmsg)
				{
					GetMessageBox().setMessage(string.Empty, load_str(35531u), rmg);
				}
				return false;
			}
		}
		return true;
	}
