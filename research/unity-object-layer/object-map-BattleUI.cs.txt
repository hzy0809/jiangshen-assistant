	protected override void Web_OnGUI()
	{
		GUI.depth = __BaseUI.UIDepth_Announce;
		__BaseUI.my_enabled = IsEnable();

		GUI.Label(new Rect(0f, 0f, Screen.width, Screen.height), string.Empty, bgStyle);

		bool flag = GUI.Toggle(__BaseUI._rectRB(120f, 60f, 80f, 22f), !interpreter.ShowEffect, string.Empty, styleHideEffect);
		if (flag != !interpreter.ShowEffect)
		{
			interpreter.ShowEffect = !interpreter.ShowEffect;
			PlayerPrefs.SetInt(showEffectKey, interpreter.ShowEffect ? 1 : 0);
		}
		bool flag2 = GUI.Toggle(__BaseUI._rectRB(210f, 60f, 80f, 22f), !bShowHpBar, string.Empty, styleHideHpBar);
		if (flag2 != !bShowHpBar)
		{
			bShowHpBar = !bShowHpBar;
			PlayerPrefs.SetInt(showHpKey, bShowHpBar ? 1 : 0);
			BroadcastMessage("SwitchDisplayMode", bShowHpBar);
		}
		int num = 280;
		if (interpreter.packet.left_army.beast != null || interpreter.packet.right_army.beast != null)
		{
			bool flag3 = GUI.Toggle(__BaseUI._rectRB(num + 20, 60f, 80f, 22f), !interpreter.ShowSpiritAnim, string.Empty, styleHideSpirit);
			if (flag3 != !interpreter.ShowSpiritAnim)
			{
				interpreter.ShowSpiritAnim = !interpreter.ShowSpiritAnim;
				PlayerPrefs.SetInt(showSpiritKey, interpreter.ShowSpiritAnim ? 1 : 0);
			}
			num += 90;
		}
		if (bNeedWaitForBattleEnd && bUseBattleSurrender && !showResultBtn() && (Global.character_.gm || Time.realtimeSinceStartup - startupTime > 10f))
		{
			if (surrenderBtn.Button(bBtn: true, __BaseUI._rectRB(num, 85f, 72f, 72f), string.Empty))
			{
				__BaseUI.GetMessageBox().setMessage(__BaseUI.load_str(20086u), __BaseUI.load_str(20087u), null, ConfirmSurrender, _bCancel: true);
			}
			num += 90;
		}
		if (showResultBtn() && resultBtn.Button(bBtn: true, __BaseUI._rectRB(num + 20, 85f, 72f, 72f), string.Empty))
		{
			if (!canSkipBattle())
			{

			}
			else
			{
				bBattle_fighting = false;
				interpreter.SetArmyToFinalState();
				if (bUseBattleEnd)
				{
					if (bSendBattleEnd)
					{
						if (idcampaign == 70)
						{
							Global.ExternalCall("CampaignComplete", idtroop.ToString());
						}
						cs_battle_end_0x408();
						bSendBattleEnd = false;
					}
				}
				else if (m_pktBattleEnd != null)
				{
					BattleEnd(m_pktBattleEnd.pkt);
				}
			}
		}
		if (!bBattle_fighting && bBattle_started && (!bNeedWaitForBattleEnd || !bUseBattleEnd) && (!interpreter.isFinished || interpreter.MainQueueFinished))
		{
			OpenResultWindow();
		}

		{
			if (bToggleDebugParam = GUI.Toggle(new Rect(0f, 450f, 50f, 20f), bToggleDebugParam, "Param"))
			{
				drawDebugParam();
			}

		}

		BattleReport.ArmyInfo armyInfo2 = ((!m_show_defender_result) ? interpreter.packet.right_army : interpreter.packet.left_army);

	}

	public override void drawTooltips(string tip)
	{
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (tip.Contains("property"))
		{
			string text = tip.Replace("property", "tip_");
			if (text != string.Empty)
			{
				text = __BaseUI.get_string(text);
				tooltipUI.ShowTooltipCommon2(text, string.Empty, 250);
			}
		}
		else if (tip != null)
		{
			tooltipUI.ShowTooltipContent(string.Empty, tip, 500, 14);
		}
	}

	private bool showResultBtn()
	{
		if (delegateShowResultBtn != null && !delegateShowResultBtn())
		{
			return false;
		}
		return Global.character_.gm || (m_viewType == enumBattleViewType.viewDefault && Time.realtimeSinceStartup - startupTime > 1f) || m_viewType == enumBattleViewType.viewReport || (m_viewType == enumBattleViewType.viewFull && m_bViewAll) || (m_viewType == enumBattleViewType.viewVipSkip && (Global.character_.my_vip_template.skip_normal_battle || Global.character_.isFromOldSrv || m_bViewAll)) || interpreter.PlayedTurnCount > 10;
	}

	private bool canSkipBattle()
	{
		return Global.character_.gm || (m_viewType != enumBattleViewType.viewVipSkip && m_viewType != enumBattleViewType.viewFull) || m_bViewAll || interpreter.PlayedTurnCount > 10 || Global.character_.my_vip_template.skip_normal_battle;
	}

	public void cs_save_report(ulong id, bool save)
	{
		Disable();
		CSBattleReportSave_0x261 cSBattleReportSave_0x = new CSBattleReportSave_0x261();
		cSBattleReportSave_0x.idbattle_report = id;
		cSBattleReportSave_0x.saved = save;
		RootUI.m_client.Send(609, cSBattleReportSave_0x, sc_save_report);
	}

	private void sc_save_report(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCBattleReportSave_0x261 sCBattleReportSave_0x = JSProtoBuf.Serializer.Deserialize<SCBattleReportSave_0x261>(stream);
			if (sCBattleReportSave_0x.result == 0)
			{
				if (sCBattleReportSave_0x.saved)
				{
					for (int i = 0; i < Global.character_.recent_battle_report.Count; i++)
					{
						if (Global.character_.recent_battle_report[i].idbattle_report == sCBattleReportSave_0x.idbattle_report)
						{
							Global.character_.recent_battle_report[i].saved = true;
							Global.character_.saved_battle_report.Insert(0, Global.character_.recent_battle_report[i]);
						}
					}
				}
				else
				{
					for (int j = 0; j < Global.character_.saved_battle_report.Count; j++)
					{
						if (Global.character_.saved_battle_report[j].idbattle_report == sCBattleReportSave_0x.idbattle_report)
						{
							Global.character_.saved_battle_report[j].saved = false;
							Global.character_.saved_battle_report.RemoveAt(j);
							break;
						}
					}
				}
			}
		}
		Enable();
	}

	protected override void Show()
	{

		m_playerdamage = 0;
		if (idcampaign == 70)

			Global.ExternalCall("CampaignStart", idtroop.ToString());
		}
		interpreter.ShowEffect = PlayerPrefs.GetInt(showEffectKey, 1) > 0;
		bShowHpBar = PlayerPrefs.GetInt(showHpKey, 1) > 0;
		interpreter.ShowSpiritAnim = PlayerPrefs.GetInt(showSpiritKey, 1) > 0;
		InitTroop();

		{
			m_prevUI.BroadcastMessage("MyMouseExit", SendMessageOptions.DontRequireReceiver);
			m_prevUI.gameObject.SetActiveRecursively(state: false);
		}
		SendMessage("ArmyUIShow", SendMessageOptions.DontRequireReceiver);
		InitUIResource();
		battle_param_name = new string[battle_param_tip.Length];
		for (int i = 0; i < battle_param_tip.Length; i++)
		{
			battle_param_name[i] = __BaseUI.get_string(battle_param_tip[i]);
		}
		startupTime = Time.realtimeSinceStartup;
		if (top_ui.first_visible)
		{
			top_ui.MyShow();
		}
		if (spirit_top_ui.first_visible)
		{
			spirit_top_ui.MyShow();
		}
		if (wood_top_ui != null && wood_top_ui.first_visible)
		{
			wood_top_ui.MyShow();
		}
		BaseAsset.lds.Hide();
		Visible();
		rmg.ToEnable();
	}

		base.Destroy();
		Time.timeScale = 1f;
		if (m_oldUI != null)

		if (m_battle_end != null)

	}

	public int getBattleLevel()
	{
		int num = 0;
		switch (interpreter.packet.result_status)
		{
		case BattleReport.ResultStatus.LEFT_WIN:
			num = (int)(5f * (float)m_attacker_lose_num / (float)m_attacker_num);
			if (num > 4)
			{
				num = 4;
			}
			if (num < 0)
			{
				num = 0;
			}
			if (!m_show_defender_result)
			{
				num = 4 - num + 6;
			}
			break;
		case BattleReport.ResultStatus.DRAW:
			num = 5;
			break;
		case BattleReport.ResultStatus.LEFT_LOSE:
			num = (int)(5f * (float)m_defender_lose_num / (float)m_defender_num);
			if (num > 4)
			{
				num = 4;
			}
			if (num < 0)
			{
				num = 0;
			}
			if (m_show_defender_result)
			{
				num = 4 - num + 6;
			}
			break;
		}
		return num;
	}

	private void fight()
	{
		if (bBattle_fighting || !interpreter.allLegionFinished() || !interpreter.MainQueueFinished)
		{
			m_BattleLength -= Time.deltaTime;
			bBattle_fighting = interpreter.fight();
			if (!bBattle_fighting)
			{
				m_bViewAll = true;
				if (bUseBattleEnd)
				{
					if (bSendBattleEnd)
					{
						if (idcampaign == 70)
						{
							Global.ExternalCall("CampaignComplete", idtroop.ToString());
						}
						cs_battle_end_0x408();
						bSendBattleEnd = false;
					}
				}
				else if (m_pktBattleEnd != null)
				{
					BattleEnd(m_pktBattleEnd.pkt);
				}
			}
		}
		interpreter.Update();
	}

	private void cs_battle_surrender_0x407()
	{
		CSBattleSurrender_0x407 pkt = new CSBattleSurrender_0x407();
		RootUI.m_client.Send(1031, pkt, sc_battle_surrender_0x407);
	}

	private void sc_battle_surrender_0x407(Stream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCBattleSurrender_0x407 sCBattleSurrender_0x = JSProtoBuf.Serializer.Deserialize<SCBattleSurrender_0x407>(stream);
		__BaseUI.GetBattleResultUI().m_battleLevel = 0;
		m_battleLevel = 0;
		m_attacker_lose_num = m_attacker_num;
		m_defender_lose_num = 0;
		bNeedWaitForBattleEnd = false;
		bUseBattleEnd = false;
		bBattle_fighting = false;
		bBattle_surrended = true;
		if (sCBattleSurrender_0x.type == 0)
		{
			CampaignUI campaignUI = (CampaignUI)m_parentUI;
			if (Global.character_.campaigns_[idcampaign].troops_[idtroop].isNewbie == 1)
			{
				campaignUI.setBattleGuide(idcampaign, idtroop);
			}
			else
			{
				campaignUI.setBattleFailed();
			}
		}
		else if (sCBattleSurrender_0x.type != 1 && sCBattleSurrender_0x.type == 6)
		{
			battle_exit();
		}
		interpreter.SetArmyToFinalState();
	}

	private void cs_battle_end_0x408()
	{
		CSBattleEnd_0x408 pkt = new CSBattleEnd_0x408();
		RootUI.m_client.Send(1032, pkt, sc_battle_end_0x408);
	}

	private void BattleEnd(SCBattleEnd_0x408 pkt)
	{
		switch (pkt.result)
		{
		case 0u:
			if (pkt.type == 0)
			{
				(m_parentUI as CampaignUI).BattleEnd(pkt);
			}
			else if (pkt.type == 1)
			{
				(m_parentUI as BuildingMonsterScript).BattleEnd(pkt);
			}
			else if (pkt.type == 2)
			{
				(m_parentUI as TrialLevelUI).BattleEnd(pkt.iswin);
			}
			else if (pkt.type == 3)
			{
				(m_parentUI as TrialUI).BattleEnd(pkt.iswin);
			}
			else if (pkt.type == 4)
			{
				(m_parentUI as GraveLevelUI).BattleEnd(pkt);
			}
			else if (pkt.type == 5)
			{
				(m_parentUI as DivinityUI).BattleEnd(pkt);
			}
			else if (pkt.type == 6)
			{
				BattleWoodwarResultUI component = GetComponent<BattleWoodwarResultUI>();
				if ((bool)component)
				{
					component.resultStatus = pkt.dummy_challenge_status;
					component.resultChallengeLevelFinished = pkt.dummy_challenge_level_finished;
				}
			}
			else if (pkt.type == 7)
			{
				(m_parentUI as SealMonsterUI).BattleEnd(pkt);
			}
			else if (pkt.type == 8)
			{
				(m_parentUI as GeneralBookFormationUI).BattleEnd(pkt.iswin);
			}
			break;
		case 1u:
			__BaseUI.GetMessageBox().setMessage(__BaseUI.load_str(20090u), __BaseUI.get_string("accel_warning"), null);
			break;
		}
		m_reportid = pkt.idbattle_report;
		bUseBattleEnd = false;
		bSendBattleEnd = true;
		bNeedWaitForBattleEnd = false;
		m_pktBattleEnd = null;
	}

	private void sc_battle_end_0x408(Stream stream, object arg)
	{
		if (stream != null)
		{
			SCBattleEnd_0x408 pkt = JSProtoBuf.Serializer.Deserialize<SCBattleEnd_0x408>(stream);
			BattleEnd(pkt);
		}
	}

	private bool isContainCard(int id_card)

		foreach (BattleReport.ArmyInfo.CardInfo card in interpreter.packet.left_army.cards)

	}

	public void battle_exit()
	{
		rmg.ToHide();
		rmg.ToDisable();
		RootMessenger prevUI = m_prevUI;
		if (prevUI != null)
		{
			prevUI.gameObject.SetActiveRecursively(state: true);
			prevUI.ToEnable();
			prevUI.BroadcastMessage("PrepareLoading", prevUI.LevelID, SendMessageOptions.DontRequireReceiver);
			prevUI.BroadcastMessage("StartLoading", SendMessageOptions.DontRequireReceiver);
			prevUI.BroadcastMessage("ResetRendering", SendMessageOptions.DontRequireReceiver);
			prevUI.BroadcastMessage("MyMouseExit", SendMessageOptions.DontRequireReceiver);
			Destroy();
			prevUI.BroadcastMessage("PostInit", SendMessageOptions.DontRequireReceiver);
		}
		else
		{
			BaseAsset.lds.Visible();
			Destroy();
			RootMessenger oldUI = BaseAsset.lds.GetOldUI();
			BaseAsset.lds.SetOldUI(rmg);
			GameObject.Find("Root/RootUI").GetComponent<RootUI>().LoadLevel(oldUI.LevelID, rmg);
		}
		BaseAsset.lds.SetBattling(bBattling: false);
		if (OnClose != null)
		{
			OnClose();
			OnClose = null;
		}
	}
