	public static void Update()
	{
		m_bubble.Update();
		if (!Global.m_bInit)
		{
			return;
		}
		dblclk_time += Time.deltaTime;
		if (bDblClick)
		{
			bDblClick = false;
		}
		if (bWantDblClick2)
		{
			bWantDblClick2 = false;
			bDblClick = true;
		}
		if (bWantDblClick && Input.GetMouseButtonUp(0))
		{
			bWantDblClick2 = true;
			bWantDblClick = false;
		}
		if (Input.GetMouseButtonDown(0))
		{
			if (!bFirstClick)
			{
				bFirstClick = true;
				dblclk_time = 0f;
			}
			else if (bFirstClick)
			{
				if (dblclk_time < 0.3f)
				{
					bWantDblClick = true;
					dblclk_time = 0f;
					bFirstClick = false;
				}
				else
				{
					dblclk_time = 0f;
				}
			}
		}
		if (bMouseUp)
		{
			bMouseUp = false;
		}
		if (bWantMouseUp)
		{
			bWantMouseUp = false;
			bMouseUp = true;
		}
		if (Input.GetMouseButtonUp(0))
		{
			bWantMouseUp = true;
		}
	}

	public static void add_UIFunc(delegate_OnClose uifunc, bool bFront)
	{
		if (uifunc != null && !vctUIFuncs.Contains(uifunc))
		{
			if (bFront)
			{
				vctUIFuncs.Insert(0, uifunc);
			}
			else
			{
				vctUIFuncs.Add(uifunc);
			}
		}
	}

	public static void _drawBubble(long value, Rect rc, Texture2D texIcon, float duration, float ox, float my)
	{
		if (value != 0L)
		{
			GUIContent gUIContent = new GUIContent();
			GUIStyle gUIStyle = new GUIStyle();
			GUIStyle gUIStyle2 = new GUIStyle();
			Color target_clr = new Color(0f, 0f, 0f, 0f);
			if (value > 0)
			{
				gUIStyle.normal.textColor = new Color(0.2f, 1f, 0f, 1f);
				gUIContent.text = "+" + value;
				target_clr = new Color(0.2f, 1f, 0f, 0f);
			}
			else if (value < 0)
			{
				gUIStyle.normal.textColor = new Color(1f, 0.2f, 0f, 1f);
				gUIContent.text = value.ToString();
				target_clr = new Color(1f, 0.2f, 0f, 0f);
			}
			gUIStyle2.normal.textColor = new Color(0f, 0f, 0f, 1f);
			gUIStyle.font = __BaseUI.DmgFont;
			gUIStyle.fontSize = 30;
			gUIStyle2.font = __BaseUI.DmgFont;
			gUIStyle2.fontSize = 30;
			m_bubble.AddBubble(gUIContent, gUIStyle, rc, new Vector2(rc.x, rc.y + my), target_clr, duration);
		}
		if (texIcon != null)
		{
			GUIStyle gUIStyle3 = new GUIStyle();
			gUIStyle3.normal.textColor = new Color(1f, 1f, 1f, 1f);
			Rect current_rc = new Rect(rc.x - (float)texIcon.width + ox, rc.y + (float)((30 - texIcon.height) / 2), texIcon.width, texIcon.height);
			gUIStyle3.normal.background = texIcon;
			BubbleManager.Bubble bubble = m_bubble.AddBubble(new GUIContent(string.Empty), gUIStyle3, current_rc, new Vector2(current_rc.x, current_rc.y + my), new Color(1f, 1f, 1f, 0f), duration);
			bubble.setScale(bScale: true, null);
		}
	}

		if (stream == null)
		{
			return;
		}
		SCCooldownClear_0x20 sCCooldownClear_0x = JSProtoBuf.Serializer.Deserialize<SCCooldownClear_0x20>(stream);
		if (sCCooldownClear_0x.result == 0)
		{
			Global.character_.cooldowns.Remove((byte)sCCooldownClear_0x.idcooldown);
			Global.character_.gold -= (int)sCCooldownClear_0x.gold_consume;
			__BaseUI.drawStateBubble("gold-" + sCCooldownClear_0x.gold_consume);
			if (arg is delegate_OnClose delegate_OnClose)
			{
				delegate_OnClose();
			}
		}
	}
