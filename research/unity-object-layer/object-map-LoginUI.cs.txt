	private IEnumerator LoadMD5()
	{
		Logger.Log("LoadMD5");
		string url = string.Empty;
		if (Application.platform == RuntimePlatform.WindowsEditor || Application.platform == RuntimePlatform.OSXEditor)
		{
			url = "file://" + Application.dataPath + "/../Assets/asset_md5.txt";
		}
		else if (Application.platform == RuntimePlatform.OSXWebPlayer || Application.platform == RuntimePlatform.WindowsWebPlayer || Application.platform == RuntimePlatform.WindowsPlayer)
		{
			url = LoadIndexScript.getDataUrl("WebAssets/asset_md5.txt");
		}
		else if (Application.platform == RuntimePlatform.IPhonePlayer)
		{
			url = ((!(Global.i18n == "zh-tw")) ? ("file://" + Application.dataPath + "/../IOSAssets/asset_md5.txt") : ("file://" + Application.dataPath + "/../IOSAssets_tw/asset_md5.txt"));
		}
		WWW asset_www = new WWW(url);
		yield return asset_www;
		AssetMD5.Make(asset_www.bytes);
		WebAssetsManager.SetMd5(asset_www.bytes);
		Show();
	}

	{

			SetUrlAsset(array[0]);

			AddUrlAsset(array[i]);

			foreach (string item in array3)
			{
				list.Add(item);

	private IEnumerator start_login()
	{
		WWWForm form = new WWWForm();
		if (Application.platform == RuntimePlatform.WindowsEditor)
		{
			form.AddField("XDEBUG_SESSION_START", "netbeans-xdebug");
		}
		form.AddField("username", username);
		form.AddField("password", password);
		form.AddField("sid", sid);
		Logger.Log("base " + ServerSetting.url_base);
		WWW www = new WWW(ServerSetting.url_base + "login_editor.php", form);
		yield return www;
		Logger.Log(www.text);
		string str = www.text;
		int idx = str.IndexOf("{");
		str = str.Substring(idx);
		IDictionary jdata = (IDictionary)Json.Deserialize(str);
		string result = jdata["result"].ToString();
		if (result != "ok")
		{
			login_state = LoginState.LoginState_Failed;
			yield break;
		}
		string url = jdata["url"].ToString();
		idx = url.IndexOf("/", url.IndexOf("//") + 2);
		ServerSetting.url_qiantai = url.Substring(0, idx + 1);
		StartCoroutine(start_game(url));
	}

	private IEnumerator start_game(string url)
	{
		if (first_login != url)
		{
			first_login = url;

	private bool isHoliday()
	{
		int[,] array = new int[10, 3]
		{
			{ 2021, 9, 19 },
			{ 2021, 9, 20 },
			{ 2021, 9, 21 },
			{ 2021, 10, 1 },
			{ 2021, 10, 2 },
			{ 2021, 10, 3 },
			{ 2021, 10, 4 },
			{ 2021, 10, 5 },
			{ 2021, 10, 6 },
			{ 2021, 10, 7 }
		};
		int[,] array2 = new int[3, 3]
		{
			{ 2021, 9, 18 },
			{ 2021, 9, 26 },
			{ 2021, 10, 9 }
		};
		DateTime now = DateTime.Now;
		for (int i = 0; i < array.Length / 3; i++)
		{
			if (array[i, 0] == now.Year && array[i, 1] == now.Month && array[i, 2] == now.Day)
			{
				return true;
			}
		}
		if (now.DayOfWeek == DayOfWeek.Sunday || now.DayOfWeek == DayOfWeek.Saturday)
		{
			for (int j = 0; j < array2.Length / 3; j++)
			{
				if (array2[j, 0] == now.Year && array2[j, 1] == now.Month && array2[j, 2] == now.Day)
				{
					return false;
				}
			}
			return true;
		}
		return false;
	}

	private static void sc_oldplayer_bonus_0x36(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCOldPlayerBonus_0x36 sCOldPlayerBonus_0x = JSProtoBuf.Serializer.Deserialize<SCOldPlayerBonus_0x36>(stream);
			if (sCOldPlayerBonus_0x.result == 1)
			{
				Global.character_.return_gift = true;
			}
			else if (sCOldPlayerBonus_0x.result == 2)
			{
				Logger.Log("old player return hash failed");
			}
		}
	}

	private IEnumerator store_user_id(ulong id)
	{
		WWWForm form = new WWWForm();
		if (Application.platform == RuntimePlatform.WindowsEditor || Application.platform == RuntimePlatform.Android || Application.platform == RuntimePlatform.IPhonePlayer)
		{
			form.AddField("XDEBUG_SESSION_START", "netbeans-xdebug");
		}
		form.AddField("uid", id.ToString());
		string url = ServerSetting.url_base + "account/store_uid.php";
		if (Application.platform != RuntimePlatform.WindowsWebPlayer && Application.platform != RuntimePlatform.OSXWebPlayer)
		{
			url = url + "?PHPSESSID=" + ServerSetting.session_id;
		}
		WWW www = new WWW(url, form);
		yield return www;
		Logger.Log(www.text);
	}

	public bool Refresh()
	{
		if (BaseAsset.lds.IsBattling())
		{
			return false;
		}
		CleanLevel();
		CSPlayerInfo_0x10 cSPlayerInfo_0x = new CSPlayerInfo_0x10();

		RootUI.m_client.Send(16, cSPlayerInfo_0x, sc_playerinfo);

	}

	private void CleanLevel()
	{
		BaseAsset.was.Clear();
		MainUI.SetSubUI(null);
		if (BaseAsset.lds.IsBattling())
		{
			GameObject gameObject = GameObject.Find("/Battle");
			if (gameObject != null)
			{
				GameObject gameObject2 = GameObject.Find("/Battle/BattleUI");
				if (gameObject2 != null)
				{
					gameObject2.SendMessage("ResetPrevUI", SendMessageOptions.DontRequireReceiver);
				}
				gameObject.SendMessage("ToDestroy", SendMessageOptions.DontRequireReceiver);
			}
			BaseAsset.lds.SetBattling(bBattling: false);
		}
		if (BaseAsset.lds.GetCurUI() != null)
		{
			BaseAsset.lds.GetCurUI().ToDestroy();
		}
		RootBase.clear_UIFunc();
		MailUI component = BaseUI.GetRootUI().GetComponent<MailUI>();
		if (component != null)
		{
			component.ResetMailList();
		}
		BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
		if (battleResultUI != null && battleResultUI.IsVisible())
		{
			battleResultUI.Hide();
		}
		MessageBox messageBox = __BaseUI.GetMessageBox();
		if (messageBox != null && messageBox.IsVisible())
		{
			messageBox.Hide();
		}
		ConsumeUI consumeUI = __BaseUI.GetConsumeUI();
		if (consumeUI != null && consumeUI.IsVisible())
		{
			consumeUI.Hide();
		}
		MeditationUI component2 = BaseUI.GetRootUI().GetComponent<MeditationUI>();
		component2.Hide();
		MeditationUI.waitflag = true;
	}

	public static void cs_user_reconnect()
	{
		tcp_base.connection_state = tcp_base.CONNECTION_STATE.RECONNECTING;
		CSPlayerAutoReconnect_0xF006 cSPlayerAutoReconnect_0xF = new CSPlayerAutoReconnect_0xF006();
		cSPlayerAutoReconnect_0xF.userid = Global.userid;
		cSPlayerAutoReconnect_0xF.zoneid = Global.zoneid;
		cSPlayerAutoReconnect_0xF.tempid = Global.tempid;
		cSPlayerAutoReconnect_0xF.channel_id = uint.Parse(Global.game_data["channel_id"].ToString());
		if (Global.game_data.Contains("rserver_id") && Global.game_data["rserver_id"] != null)
		{
			cSPlayerAutoReconnect_0xF.serverid = uint.Parse(Global.game_data["rserver_id"].ToString());
		}
		else
		{
			cSPlayerAutoReconnect_0xF.serverid = Global.zoneid;
		}
		cSPlayerAutoReconnect_0xF.timestamp = timestamp;
		cSPlayerAutoReconnect_0xF.randnum = randnum;
		cSPlayerAutoReconnect_0xF.hash = hash;
		cSPlayerAutoReconnect_0xF.client_type = Global.game_data["login_type"].ToString();
		cSPlayerAutoReconnect_0xF.client_version = 2u;
		RootUI.m_client.Send(61446, cSPlayerAutoReconnect_0xF, sc_user_reconnect);
		Logger.Log("cs_user_reconnect user:" + Global.userid + " channel:" + cSPlayerAutoReconnect_0xF.channel_id + " zone:" + cSPlayerAutoReconnect_0xF.zoneid, bTime: true);
	}

	private static void sc_user_reconnect(MemoryStream stream, object arg)
	{
		if (BaseAsset.lds.GetCurUI() != null)
		{
			BaseAsset.lds.GetCurUI().ToEnable();
		}
		if (stream != null)
		{
			GCPlayerAutoReconnect_0xF006 gCPlayerAutoReconnect_0xF = JSProtoBuf.Serializer.Deserialize<GCPlayerAutoReconnect_0xF006>(stream);
			Logger.Log("sc_user_reconnect:" + gCPlayerAutoReconnect_0xF.result, bTime: true);
			if (gCPlayerAutoReconnect_0xF.result == 0)
			{
				BaseUI.GetRootUI().GetComponent<RetryConnectUI>().Hide();
				RetryConnectUI.phase = 0;
				tcp_base.connection_state = tcp_base.CONNECTION_STATE.OK;
			}
		}
	}

	private void cs_ping()
	{
		CSPing_0x00 cSPing_0x = new CSPing_0x00();
		cSPing_0x.timestamp = TimeUtil.Now;
		RootUI.m_client.Send(0, cSPing_0x, sc_ping);
	}

	private void sc_ping(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCPing_0x00 sCPing_0x = JSProtoBuf.Serializer.Deserialize<SCPing_0x00>(stream);
			if (sCPing_0x != null)
			{
				_ping = TimeUtil.Now - sCPing_0x.timestamp;
			}
		}
		StartCoroutine(wait_ping(ping_delay));
	}

	private IEnumerator wait_ping(float d)
	{
		yield return new WaitForSeconds(d);
		cs_ping();
	}
