	private void Building_playAnim(GameObject mesh, Building d)
	{
		CutSceneManager component = mesh.GetComponent<CutSceneManager>();
		if (component == null)
		{
			return;
		}
		if (d.product != null && d.build_time == 0 && d.gathering)
		{
			if (d.product.gold_consume != 0)
			{
				component.Play(2);
			}
			else
			{
				component.Play(1);
			}
		}
		else
		{
			component.Play(0);
		}
	}

	public void Building_makeMesh(GameObject mesh, Building d)
	{
		BuildingBehaviourScript buildingBehaviourScript = mesh.AddComponent<BuildingBehaviourScript>();
		buildingBehaviourScript.setBuilding(d, bEdit: false, bFriend: false);
		if (BuildingBehaviourScript.m_bEditVisible || BuildingBuildUI.building != null)
		{
			buildingBehaviourScript.detectMouseClick = false;
			buildingBehaviourScript.HideMesh();
		}
	}

	private void Demon_make(GameObject mesh, __BaseUI parentUI, Demon demon, bool bFriend)
	{
		mesh.AddComponent<DemonScript>().setDemon(parentUI, demon, bFriend);
		if (demon.killed)
		{
			mesh.animation.Play("dead");
		}
	}

	public static void FormationUI_wantBattle(string subui, ushort id_spirit_ascend_campaign, byte idtroop, DelegateFunc.delegate_OnAttack OnAttack, bool viewonly)
	{
		Hashtable hashtable = new Hashtable();
		subui = "NPCBattle";
		hashtable.Add("subui", subui);
		hashtable.Add("campaignid", id_spirit_ascend_campaign);
		hashtable.Add("troopid", idtroop);
		hashtable.Add("substate", "spiritascend");
		hashtable.Add("viewonly", viewonly);
		hashtable.Add("OnAttack", new FormationUI.delegate_OnAttack(OnAttack.Invoke));
		FormationUI component = BaseUI.GetRootUI().GetComponent<FormationUI>();
		component.SubLoading(hashtable);
		MainUI.SetSubUI(component);
	}

	public static void BattleUI_SetReport(RootMessenger rmg, byte[] report, bool needyijian)
	{
		BattleUI component = GameObject.Find("Battle/BattleUI").GetComponent<BattleUI>();
		component.setPrevUI(rmg);
		component.setReport(report, wait_for_end: true, use_battle_end: false, use_battle_surrender: false, null, isopponent: false);
		if (needyijian)
		{
			component.ViewType = enumBattleViewType.viewReport;
		}
		else
		{
			component.ViewType = enumBattleViewType.viewFull;
		}
		BaseUI.GetRootUI().battle_enter(fslds: false);
	}
