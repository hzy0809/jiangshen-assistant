	public static Color ParseColor(string clr)
	{
		string text = "FF000000";
		if (clr.Length < 8)
		{
			clr = text.Substring(0, 8 - clr.Length) + clr;
		}
		float[] array = new float[4];
		for (int i = 0; i < 4; i++)
		{
			array[i] = (float)int.Parse(clr.Substring(i * 2, 2), NumberStyles.AllowHexSpecifier) / 255f;
		}
		return new Color(array[1], array[2], array[3], array[0]);
	}

	public static string getStringFromInt(int num)
	{
		if (num < 0 || num > 999)
		{
			return string.Empty;
		}
		string text = string.Empty;
		if (num >= 100)
		{
			text = text + strNumbers[num / 100] + "百";
			num %= 100;
		}
		if (num >= 10)
		{
			text += strNumbers2[num / 10];
			num %= 10;
		}
		else if (text != string.Empty && num != 0)
		{
			text += "零";
		}
		if (num > 0 || text == string.Empty)
		{
			text += strNumbers[num];
		}
		return text;
	}

		if (d.type == "spirit")
		{
			tlm_spirit.Set(d);
			tlm_spirit.isactive = true;
			string[] array = d.param.Split(',');
			for (int i = 0; i < array.Length; i++)
			{
				string[] array2 = array[i].Split(':');
				if (array2.Length > 1)
				{
					if (array2[0] == "exp")
					{
						tlm_spirit.exp_extern = float.Parse(array2[1]);
					}
					else if (array2[0] == "fruit")
					{
						tlm_spirit.fruit_extern = float.Parse(array2[1]);
					}
					else if (array2[0] == "unlimit")
					{
						tlm_spirit.unlimit_time = array2[1] == "true";
					}
					else if (array2[0] == "diamond_gold_ratio")
					{
						tlm_spirit.diamond_gold_ratio = int.Parse(array2[1]);
					}
				}
			}
		}
		else
		{
			if (!(d.type == "exchange"))
			{
				return;
			}
			tlm_exchange.Set(d);
			tlm_exchange.isactive = true;
			string[] array3 = d.param.Split(',');
			for (int j = 0; j < array3.Length; j++)
			{
				string[] array4 = array3[j].Split(':');
				if (array4.Length > 1)
				{
					if (array4[0] == "money")
					{
						tlm_exchange.money_extern = float.Parse(array4[1]);
					}
					else if (array4[0] == "diamond_gold_ratio")
					{
						tlm_spirit.diamond_gold_ratio = int.Parse(array4[1]);
					}
				}
			}
		}
	}

	public static string DecodeUnicode(string str)
	{
		string text = string.Empty;
		MatchCollection matchCollection = Regex.Matches(str, "\\\\u([\\w]{2})([\\w]{2})", RegexOptions.IgnoreCase);
		byte[] array = new byte[2];
		foreach (Match item in matchCollection)
		{
			array[0] = (byte)int.Parse(item.Groups[2].Value, NumberStyles.HexNumber);
			array[1] = (byte)int.Parse(item.Groups[1].Value, NumberStyles.HexNumber);
			text += Encoding.Unicode.GetString(array);
		}
		return text;
	}

	public static int bagua_activity(int _type, uint treasuretype)
	{
		bool flag = false;
		Hashtable templates = getTemplates("bagua_activity");
		foreach (DictionaryEntry item in templates)
		{
			_bagua_activity bagua_activity = item.Value as _bagua_activity;
			if (bagua_activity.type == _type)
			{
				int num = 0;
				switch (TimeUtil.WeekDay)
				{
				case DayOfWeek.Monday:
					num = 1;
					break;
				case DayOfWeek.Tuesday:
					num = 2;
					break;
				case DayOfWeek.Wednesday:
					num = 3;
					break;
				case DayOfWeek.Thursday:
					num = 4;
					break;
				case DayOfWeek.Friday:
					num = 5;
					break;
				case DayOfWeek.Saturday:
					num = 6;
					break;
				case DayOfWeek.Sunday:
					num = 7;
					break;
				}
				if (num == bagua_activity.day)
				{
					flag = true;
					break;
				}
			}
		}
		return (!flag || treasuretype == 36 || treasuretype == 0) ? 1 : 2;
	}
