using UnityEngine;

public class SecretShopNPC : MeshScript
{
	public Texture2D tex_btn_normal;

	public Texture2D tex_btn_down;

	public Texture2D tex_btn_hover;

	public Texture2D tex_special_btn_normal;

	public Texture2D tex_special_btn_down;

	public Texture2D tex_special_btn_hover;

	public float height = 2f;

	private GUIStyle btnStyle = new GUIStyle();

	private Vector2 btnSize = Vector2.zero;

	private Vector3 btnPos = Vector3.zero;

	public bool SpecialDay
	{
		set
		{
			if (Global.shoplimit != null && Global.shoplimit.shoplimititems.Count > 0)
			{
				btnStyle.normal.background = tex_special_btn_normal;
				btnStyle.hover.background = tex_special_btn_hover;
				btnStyle.active.background = tex_special_btn_down;
			}
			else
			{
				btnStyle.normal.background = tex_btn_normal;
				btnStyle.hover.background = tex_btn_hover;
				btnStyle.active.background = tex_btn_down;
			}
		}
	}

	public override void Start()
	{
		base.Start();
		if (btnStyle.normal.background != null)
		{
			btnSize.x = btnStyle.normal.background.width;
			btnSize.y = btnStyle.normal.background.height;
		}
	}

	private void Update()
	{
		if (Camera.main != null)
		{
			Vector3 position = base.transform.position;
			position.y += height;
			btnPos = Camera.main.WorldToScreenPoint(position);
			btnPos.y = (float)Screen.height - btnPos.y;
		}
	}

	protected override void MyMouseDown()
	{
		SendMessage("SwitchToIdle", SendMessageOptions.DontRequireReceiver);
		MainUI._secret_shop();
	}

	private void OnGUI()
	{
		GUI.depth = __BaseUI.UIDepth_Scene;
		if (Global.shoplimit != null && Global.shoplimit.shoplimititems.Count > 0)
		{
			btnStyle.normal.background = tex_special_btn_normal;
			btnStyle.hover.background = tex_special_btn_hover;
			btnStyle.active.background = tex_special_btn_down;
		}
		else
		{
			btnStyle.normal.background = tex_btn_normal;
			btnStyle.hover.background = tex_btn_hover;
			btnStyle.active.background = tex_btn_down;
		}
		if (GUI.Button(new Rect(btnPos.x - btnSize.x * __BaseUI._screenScale / 2f, btnPos.y - btnSize.y * __BaseUI._screenScale, btnSize.x * __BaseUI._screenScale, btnSize.y * __BaseUI._screenScale), new GUIContent(string.Empty, "ButtonSecretShop"), btnStyle))
		{
			SendMessage("SwitchToIdle", SendMessageOptions.DontRequireReceiver);
			MainUI._secret_shop();
		}
		drawTooltips();
	}

	public override void drawTooltips(string tip)
	{
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (tip == "ButtonSecretShop")
		{
			tooltipUI.ShowTooltipCommon2(__BaseUI.get_string("ms_btn_tooltip"), "ButtonSecretShop", 200);
		}
	}
}
