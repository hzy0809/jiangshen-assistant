	public void RefreshBuildingQuests()
	{

			}
		}

	private void LoadArea(BuildingArea area)
	{
		foreach (Building value in Global.character_.buildings.Values)
		{
			if (!area.m_islands.ContainsKey(value.idisland))
			{
				continue;
			}
			AddAssetSub(value.data.model_name);
			if (value.product != null)
			{
				if (value.product.texIcon == null)
				{
					AddAssetSub("product" + value.product.icon);
				}
				if (value.product.mesh == null)
				{
					AddAssetSub(value.product.model_name);
				}
			}
		}
		foreach (object value2 in Global.getTemplates("building_default").Values)
		{
			_building_default building_default = (_building_default)value2;
			bool flag = false;
			foreach (Building value3 in Global.character_.buildings.Values)
			{
				if (value3.idbuilding == building_default.idbuilding)
				{
					flag = true;
					break;
				}
			}
			if (!flag && Global.getTemplate("building_floor", building_default.idisland) is _building_floor building_floor)
			{
				AddAssetSub(building_floor.floor_name);
			}
		}
	}

	private void CreateArea(BuildingArea area)
	{
		foreach (Building value in Global.character_.buildings.Values)
		{
			if (!area.m_islands.ContainsKey(value.idisland))
			{
				continue;
			}
			if (value.mesh == null)
			{
				value.makeObject();
			}
			if (value.product != null && value.product.mesh == null)
			{
				AssetBundle asset = BaseAsset.was.GetAsset(value.product.model_name);
				if (asset != null)
				{
					value.product.mesh = Object.Instantiate(asset.mainAsset) as GameObject;
					value.product.mesh.transform.parent = value.rootMesh.transform;
					value.product.mesh.transform.localPosition = Vector3.zero;
					value.product.mesh.AddComponent<BuildingProductScript>().setBuilding(value);
				}
			}
		}
	}

	public void loadFloor(byte idisland)
	{
		if (Global.getTemplate("building_floor", idisland) is _building_floor building_floor)
		{
			AddAssetSub(building_floor.floor_name);
			StartLoadingSub();
		}
	}

	public void loadBuilding(byte id)
	{
		Building building = Global.character_.buildings[id];
		AddAssetSub(building.data.model_name);
		StartLoadingSub();
	}

	public Texture2D GetBPicon(int iconid)
	{
		Texture2D result = null;
		if (BuildProductTex.ContainsKey(iconid))
		{
			result = BuildProductTex[iconid];
		}
		else
		{
			AddAssetSub("product" + iconid);
			StartLoadingSub();
		}
		return result;
	}

	protected override void AssetLoaded(string AssetName)
	{
		if (AssetName.Contains("product"))
		{
			string s = AssetName.Replace("product", string.Empty);
			int key = int.Parse(s);
			if (!BuildProductTex.ContainsKey(key))
			{
				Texture2D value = BaseAsset.LoadTexture(AssetName);
				BuildProductTex.Add(key, value);
			}
		}
		foreach (Building value2 in Global.character_.buildings.Values)
		{
			if (value2.product != null && value2.product.texIcon == null)
			{
				value2.product.texIcon = BaseAsset.LoadTexture("product" + value2.product.icon);
			}
		}
		foreach (BuildingQuest value3 in Global.character_.building_quests.Values)
		{
			for (int i = 0; i < 4; i++)
			{
				if (value3.texIcons[i] == null)
				{
					string iconName = value3.getIconName((byte)i);
					if (iconName != string.Empty)
					{
						value3.texIcons[i] = BaseAsset.LoadTexture(iconName);
					}
				}
			}
		}
		foreach (BuildingArea value4 in Global.character_.building_areas.Values)
		{
			CreateArea(value4);
		}
		foreach (Database.BuildingMonster value5 in Global.character_.building_monsters.Values)
		{
			if (!(value5.mesh == null) || !(root_monsters != null))
			{
				continue;
			}
			AssetBundle asset = BaseAsset.was.GetAsset(value5.model_name);
			if (!(asset != null))
			{
				continue;
			}
			value5.mesh = Object.Instantiate(asset.mainAsset) as GameObject;
			BaseAsset.SetLayerRecursively(value5.mesh, 10);
			value5.mesh.transform.position = value5.pos;
			value5.mesh.transform.eulerAngles = value5.rot;
			value5.mesh.transform.localScale = new Vector3(3f, 3f, 3f);
			value5.mesh.transform.parent = root_monsters.transform;
			BuildingMonsterScript buildingMonsterScript = value5.mesh.AddComponent<BuildingMonsterScript>();
			buildingMonsterScript.setMonster(this, value5, bFriend: false);
			if (value5.idkiller != 0L)
			{
				value5.mesh.animation.Play("dead");
			}
			GameObject gameObject = BaseAsset.Instantiate_mainAsset("blob_shadow") as GameObject;
			if (gameObject != null)
			{
				BaseAsset.SetLayerRecursively(gameObject, 10);
				gameObject.tag = "blob_shadow";
				gameObject.transform.parent = value5.mesh.transform;
				gameObject.transform.localPosition = Vector3.zero;
				gameObject.transform.localScale = gameObject.transform.localScale * value5.mesh.transform.localScale.x;
				if (QualitySettings.names[QualitySettings.GetQualityLevel()] != "Fastest")
				{
					OptionUI.disable_blob_shadow(gameObject);
				}
			}
		}
		if (Global.character_.m_demon != null && Global.character_.m_demon.mesh == null)
		{
			Global.character_.m_demon.make(BaseAsset.was.GetAsset(Global.character_.m_demon.model_name), root.transform, this, bFriend: false);
		}
		if (!AssetName.Contains("capital0/zc_xindiban"))
		{
			return;
		}
		foreach (object value6 in Global.getTemplates("building_floor").Values)
		{
			_building_floor building_floor = (_building_floor)value6;
			if (building_floor.floor_name == AssetName)
			{
				AssetBundle asset2 = BaseAsset.was.GetAsset(AssetName);
				GameObject gameObject2 = Object.Instantiate(BaseAsset.was.GetAsset(AssetName).mainAsset) as GameObject;
				gameObject2.transform.parent = root.transform;
				if (Global.getTemplate("building_island", building_floor.idisland) is _building_island d)
				{
					BuildingIsland buildingIsland = new BuildingIsland(d);
					buildingIsland.pos.x += 5f;
					buildingIsland.pos.z += 5f;
					gameObject2.transform.position = buildingIsland.pos;
					floors[building_floor.idisland] = gameObject2;
				}
			}
		}
	}

	public override void Destroy()
	{
		if (bgm_ctrl != null)
		{
			bgm_ctrl.FadeOut(1f);
			bgm_ctrl = null;
		}
		base.Destroy();
		if (root != null)
		{
			Object.Destroy(root);
			root = null;
			mesh = null;
			Building.root_buildings = null;
			root_lines = null;
			root_monsters = null;
			secret_shop_npc = null;
		}
		foreach (BuildingArea value in Global.character_.building_areas.Values)
		{
			foreach (Building value2 in Global.character_.buildings.Values)
			{
				value2.root = null;
			}
		}
		foreach (Database.BuildingMonster value3 in Global.character_.building_monsters.Values)
		{
			value3.mesh = null;
		}
	}

	private void Update()
	{
		if (Global.character_.checkUI(53u) && Global.character_.next_demon_time <= TimeUtil.Now && !m_bRefreshing)
		{
			m_bRefreshing = true;
			cs_player_refresh_demon_0x26();
		}
		foreach (BuildingQuest value in Global.character_.building_quests.Values)
		{
			if (value.m_fNewTime > 0f)
			{
				value.m_fNewTime -= Time.deltaTime;
				if (value.m_fNewTime < 0f)
				{
					value.m_fNewTime = 0f;
				}
			}
			if (!value.m_bRemoving && value.checkQuest())
			{
				value.m_bRemoving = true;
				CSBuildingQuestFinished_0x48B cSBuildingQuestFinished_0x48B = new CSBuildingQuestFinished_0x48B();
				cSBuildingQuestFinished_0x48B.id = value.id;
				RootUI.m_client.Send(1163, cSBuildingQuestFinished_0x48B, null);
			}
		}
		if (Global.character_.building_monsters.Count < 3 && Global.character_.next_monster_time <= TimeUtil.Now && m_bRefreshing)
		{
			m_bRefreshing = true;
			CSBuildingMonsterRefresh_0x48C pkt = new CSBuildingMonsterRefresh_0x48C();
			RootUI.m_client.Send(1164, pkt, null);
		}
	}

	private void _harvest_all()
	{
		int num = 0;
		int num2 = 0;
		int num3 = 0;
		foreach (Building value in Global.character_.buildings.Values)
		{
			if (value.product != null && value.start_time + value.product.time <= TimeUtil.Now)
			{
				num += (int)value.product.spirit_output;
				num2 += (int)value.product.mana_output;
				num3 += (int)value.product.enchant_output;
			}
		}
		int capacityMax = Global.character_.capacityMax;
		if ((num > 0 && num + Global.character_.spirit_food > capacityMax) || (num2 > 0 && num2 + Global.character_.mana > capacityMax) || (num3 > 0 && num3 + Global.character_.enchant_stone > capacityMax))
		{
			DisableAll();
			__BaseUI.GetMessageBox().setMessage(__BaseUI.load_str(20205u), __BaseUI.load_str(20176u), null, WantHarvestAllOK, _bCancel: true);
		}
		else
		{
			cs_building_harvestall_0x488();
		}
	}

	private void sc_player_refresh_demon_0x26(MemoryStream stream, object arg)
	{
		m_bRefreshing = false;
		if (stream == null)
		{
			return;
		}
		SCPlayerRefreshDemon_0x26 sCPlayerRefreshDemon_0x = JSProtoBuf.Serializer.Deserialize<SCPlayerRefreshDemon_0x26>(stream);
		Global.character_.next_demon_time = sCPlayerRefreshDemon_0x.next_demon_time;
		if (Global.character_.m_demon != null)
		{
			if (Global.character_.m_demon.mesh != null)
			{
				Object.Destroy(Global.character_.m_demon.mesh);
			}
			Global.character_.m_demon = null;
		}
		if (sCPlayerRefreshDemon_0x.demon != null)
		{
			Global.character_.m_demon = new Database.Demon(sCPlayerRefreshDemon_0x.demon);
			AddAssetSub(Global.character_.m_demon.model_name);
			StartLoadingSub();
		}
	}

		{
			return;
		}
		SCMysteryShopCheck_0x250 sCMysteryShopCheck_0x = JSProtoBuf.Serializer.Deserialize<SCMysteryShopCheck_0x250>(stream);
		if (sCMysteryShopCheck_0x.result == 0)
		{
			if (Global.character_.mystery_shop == null)
			{
				Global.character_.mystery_shop = new Database.MysteryShop();
			}
			Global.character_.mystery_shop.make(sCMysteryShopCheck_0x.mystery_shop);
			if (secret_shop_npc == null)

				secret_shop_npc = BaseAsset.Instantiate_mainAsset("capital0/shenmishangren") as GameObject;

				secret_shop_npc.transform.parent = root.transform;
				SecretShopNPC componentInChildren = secret_shop_npc.GetComponentInChildren<SecretShopNPC>();
				componentInChildren.SpecialDay = Global.character_.mystery_shop.special_day != 0;
			}
		}
	}
