	private class ChatSingleMsg
	{
		public uint type;

		public string msg;

		public float addTime;

		public string sender;

		public ScenarioSection section;

		public ChatSingleMsg(uint channel, string _msg, string _sender, uint achievement, uint sid)
		{
			type = channel;
			_chat_channel chat_channel = Global.getTemplate("chat_channel", channel) as _chat_channel;
			if (chat_channel != null)
			{
				if (__BaseUI._isunity3_5)
				{
					msg = chat_channel.style_u35 + "［" + chat_channel.desc + "］";
				}
				else
				{
					msg = chat_channel.style_u4 + "【" + chat_channel.desc + "】";
				}
			}
			sender = _sender;
			if (sender != null)
			{
				if (achievement != 0 && Global.getTemplate("achievement", (ushort)achievement) is _achievement achievement2)
				{
					string text = msg;
					msg = text + __BaseUI.clrQualityString(achievement2.color) + achievement2.name + "·" + ((!__BaseUI._isunity3_5) ? chat_channel.style_u4 : chat_channel.style_u35);
				}
				if (channel != 6 && channel != 7)
				{
					string text = msg;
					msg = text + "#usrnm=" + sid + ":\"" + sender + "\"";
				}
				else
				{
					msg += sender;
				}
			}
			msg += "：#i=0";
			msg += _msg;
			addTime = Time.realtimeSinceStartup;
		}

		public void make(int w, GUIStyle style)
		{
			section = new ScenarioSection();
			section.setParams(w, 500, __BaseUI._fontSize, 8, !__BaseUI._isunity3_5);
			section.makes_string(msg, style, bSystem: true);
		}

		public static int Compare(ChatSingleMsg x, ChatSingleMsg y)
		{
			int result = 1;
			if (x != null && x is ChatSingleMsg && y != null && y is ChatSingleMsg)
			{
				result = x.addTime.CompareTo(y.addTime);
			}
			return result;
		}
	}

	public static bool isMouseInChatBox()
	{
		if (!instance || !instance.m_showChat)
		{
			return false;
		}
		Vector3 mousePosition = Input.mousePosition;
		mousePosition.y = (float)Screen.height - mousePosition.y;
		return new Rect(bottom_left.x * __BaseUI._screenScale, (float)Screen.height - (chat_whole_size.y + bottom_left.y) * __BaseUI._screenScale, chat_whole_size.x * __BaseUI._screenScale, chat_whole_size.y * __BaseUI._screenScale).Contains(mousePosition);
	}

	protected override void Web_OnGUI()
	{
		if (BuildingBuildUI.m_bBuildVisible || !m_bInit)
		{
			return;
		}
		GUI.depth = __BaseUI.UIDepth_Main2;
		if ((craft_ui != null && craft_ui.enabled && craft_ui.gameObject.active) || (ladder_ui != null && ladder_ui.enabled && ladder_ui.gameObject.active))
		{
			bottom_left.x = ((float)Screen.width - chat_whole_size.x) / 2f * __BaseUI._screenScale;
			bottom_left.y = 70f * __BaseUI._screenScale;
		}
		else
		{
			bottom_left.x = 60f;
			bottom_left.y = 0f;
		}
		__BaseUI.BeginGroup(__BaseUI._rectB(bottom_left.x, chat_whole_size.y + bottom_left.y, chat_whole_size.x, chat_whole_size.y));
		if (m_showChat)
		{
			GUI.Box(__BaseUI._rect(0f, 0f, chat_whole_size.x, chat_whole_size.y), string.Empty, styleChatBG);
		}
		GUI.Box(__BaseUI._rectB(10f, 38f, chat_whole_size.x - 20f - 60f - 5f, 27f), string.Empty, styleInputBG);
		bool flag = Event.current.type == EventType.KeyDown && (Event.current.keyCode == KeyCode.Return || Event.current.keyCode == KeyCode.KeypadEnter);
		GUI.SetNextControlName("chat input");
		m_user_input = GUI.TextField(__BaseUI._rectB(60f, 32f, chat_whole_size.x - 10f - 60f - 60f - 30f - 5f, 16f), m_user_input, 50, input_style);
		m_user_input = m_user_input.Replace("\r", string.Empty);
		m_user_input = m_user_input.Replace("\n", string.Empty);
		m_user_input = m_user_input.Replace("\t", string.Empty);
		if (GUI.GetNameOfFocusedControl() == "chat input" && flag)
		{
			m_user_input = m_user_input.TrimStart(" \r\n\t".ToCharArray());
			m_user_input = m_user_input.TrimEnd(" \r\n\t".ToCharArray());
			if (m_user_input.Length > 0)
			{
				SendChatMessage(send_channel_id[current_send_channel], m_user_input, null);
			}
		}
		if (m_showChat)
		{
			drawChatWithBox(m_text_area_rect);
		}
		else
		{
			drawChatWithoutBox(m_text_area_rect);
		}
		int num = ((channel_index.Length != 4) ? 85 : 62);
		int num2 = ((channel_index.Length != 4) ? 80 : 62);
		for (int i = 0; i < channel_index.Length; i++)
		{
			bool flag2 = GUI.Toggle(__BaseUI._rectB(12 + i * num, 63f, num2, 22f), selectedFilter[i], string.Empty, styleToggleFilter);
			GUI.Label(__BaseUI._rectB(12 + i * num, 63f, num2, 22f), channel_name_tex[channel_index[i]][flag2 ? 1 : 0], channel_name_style);
			if (flag2 != selectedFilter[i])
			{
				selectedFilter[i] = flag2;
				combine_msg_buffer();
				m_reset_chat_scroll = true;
			}
		}
		if (GUI.Toggle(__BaseUI._rectB(14f, 35f, 46f, 21f), show_channel_list, new GUIContent(send_channel_tex[current_send_channel]), styleBtnChannel))
		{
			if (!show_channel_list)
			{
				PopupUI.Register(DrawChannelSelect, CloseChannelSelect, "chat_channel", this);
				show_channel_list = true;
			}
		}
		else
		{
			show_channel_list = false;
		}
		string text = string.Empty;
		if (Global.game_data.Contains("bbs_url"))
		{
			text = Global.game_data["bbs_url"].ToString();
		}
		if (text != string.Empty && btnluntan.Button(bBtn: true, __BaseUI._rectB(266f, 66f, 56f, 26f), string.Empty))
		{
			Global.IEExplore(text, "bbs_url");
		}
		if (btnSend.Button(bBtn: true, __BaseUI._rectRB(70f, 38f, 60f, 27f), string.Empty))
		{
			m_user_input = m_user_input.TrimStart(" \r\n\t".ToCharArray());
			m_user_input = m_user_input.TrimEnd(" \r\n\t".ToCharArray());
			if (m_user_input.Length > 0)
			{
				SendChatMessage(send_channel_id[current_send_channel], m_user_input, null);
			}
		}
		if (emote_container != null)
		{
			if (GUI.Toggle(__BaseUI._rectRB(103f, 36f, 24f, 24f), show_emote_box, string.Empty, styleBtnEmote))
			{
				if (!show_emote_box)
				{
					show_emote_box = true;
					PopupUI.Register(DrawEmoteBox, CloseEmoteBox, "chat_emote", this);
				}
			}
			else
			{
				show_emote_box = false;
			}
		}
		if (m_showChat)
		{
			if (btnHideChatBox.Button(bBtn: true, __BaseUI._rectRB(36f, 61f, 28f, 19f), string.Empty))
			{
				m_showChat = false;
				rect_dirty = true;
			}
		}
		else if (btnShowChatBox.Button(bBtn: true, __BaseUI._rectRB(36f, 61f, 28f, 19f), string.Empty))
		{
			m_showChat = true;
			rect_dirty = true;
			m_reset_chat_scroll = true;
		}
		if (m_showChat)
		{
			GUI.Button(__BaseUI._rect(0f, 0f, chat_whole_size.x, chat_whole_size.y), string.Empty, __BaseUI._textNLeft);
		}
		else
		{
			GUI.Button(__BaseUI._rect(0f, chat_whole_size.y - 40f, chat_whole_size.x, 40f), string.Empty, __BaseUI._textNLeft);
		}
		__BaseUI.EndGroup();
	}

	private bool DrawEmoteBox()
	{
		Rect position = __BaseUI._rect(bottom_left.x + chat_whole_size.x - 70f - 28f + 24f - emote_box_size.x, (float)Screen.height - bottom_left.y - 38f - emote_box_size.y, emote_box_size.x, emote_box_size.y);
		GUI.Box(new Rect(position.x - 5f, position.y - 5f, position.width + 10f, position.height + 10f), string.Empty, __BaseUI._boxnew10Style);
		GUI.BeginGroup(position);
		for (int i = 0; i < 4; i++)
		{
			for (int j = 0; j < 6; j++)
			{
				int num = i * 6 + j;
				float x = (float)j * emote_size.x + (emote_size.x - emote_container.emotes[num].size.x) / 2f;
				float y = (float)i * emote_size.y + (emote_size.y - emote_container.emotes[num].size.y);
				float x2 = emote_container.emotes[num].size.x;
				float y2 = emote_container.emotes[num].size.y;
				if (num >= emote_container.emotes.Length)
				{
					continue;
				}
				emote_container.DrawIndex(num, __BaseUI._rect(x, y, x2, y2));
				if (GUI.Button(__BaseUI._rect((float)j * emote_size.x, (float)i * emote_size.y, emote_size.x, emote_size.y), string.Empty, styleEmoteSelect))
				{
					if (Global.getTemplate("emote_code", (uint)emote_container.emotes[num].id) is _emote_code emote_code)
					{
						m_user_input = m_user_input + "#" + emote_code.codeChs;
						GUI.FocusControl("chat input");
					}
					PopupUI.EndPopup("chat_emote");
				}

	public void SendChatMessage(uint channel, string content, string receiver)
	{
		if (Global.fcm != 0 && Global.game_data["channel"].ToString() == "37wan")
		{
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(9297u), rmg, shimingrenzhen);
			return;
		}
		if (Global.fcm != 0 && Global.character_.VIPLEVEL <= 0)
		{
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(36512u), rmg);
			return;
		}
		string text = string.Empty;
		int length = content.Length;
		int num = 0;
		while (num < length)
		{
			if (content[num] == '#')
			{
				num++;
				int i = num;
				string text2 = string.Empty;
				bool flag = false;
				for (; i < length; i++)
				{
					text2 += content[i];
					switch (EmoteContainer.Instance.hasEmote(text2))
					{
					case 0:
						flag = true;
						num += text2.Length;
						text = text + "#" + text2;
						continue;
					default:
						continue;
					case -2:
						break;
					}
					break;
				}
				if (!flag)
				{
					text += "##";
				}
			}
			else
			{
				text += content[num++];
			}
		}
		m_user_input = string.Empty;
		if (channel != 4 && (send_cooldown > 0f || last_chat_message == text))
		{
			SCChatMessage_0x220 sCChatMessage_0x = new SCChatMessage_0x220();
			sCChatMessage_0x.channel = 5u;
			sCChatMessage_0x.content = __BaseUI.load_str(20593u);
			sCChatMessage_0x.sender = string.Empty;
			sCChatMessage_0x.receiver = receiver;
			OnReceiveMessage(sCChatMessage_0x);
			return;
		}
		CSChatMessage_0x220 cSChatMessage_0x = new CSChatMessage_0x220();
		cSChatMessage_0x.channel = channel;
		cSChatMessage_0x.content = text;
		if (receiver != null)
		{
			cSChatMessage_0x.receiver = receiver;
		}
		switch (channel)
		{
		case 7u:
			__BaseUI.GetConsumeUI().setConsume(nationwar_broadcast_cost, 66, nation_broadcast, new object[1] { cSChatMessage_0x });
			break;
		case 9u:
			__BaseUI.GetConsumeUI().setConsume(teamfight_broadcast_cost, 66, teamfight_broadcast, new object[1] { cSChatMessage_0x });
			break;
		default:
			send_msg(cSChatMessage_0x);
			break;
		}
	}

	public void OnReceiveMessage(SCChatMessage_0x220 pkt)
	{
		if (pkt.channel == 4)
		{
			if (messenger != null)
			{
				messenger.SendMessage("ReceivePrivateMessage", pkt);
			}
			return;
		}
		string content = pkt.content;
		if (pkt.channel == 1)
		{
			SendMessage("AddAnnounceMessage", content, SendMessageOptions.DontRequireReceiver);
		}
		if (pkt.channel == 5)
		{
			pkt.channel = 1u;
		}
		ChatSingleMsg item = new ChatSingleMsg(pkt.channel, pkt.content, pkt.sender, pkt.achievement, pkt.server_sid);
		uint num = pkt.channel;
		if (num == 7)
		{
			num = 6u;
		}
		if (num == 9)
		{
			num = 8u;
		}
		if (!channel_buffer_msg.ContainsKey(num))
		{
			channel_buffer_msg[num] = new List<ChatSingleMsg>();
		}
		channel_buffer_msg[num].Add(item);
		if (channel_buffer_msg[num].Count > 60)
		{
			channel_buffer_msg[num].RemoveRange(0, channel_buffer_msg[num].Count - 60);
		}
		for (int i = 0; i < channel_index.Length; i++)
		{
			if (selectedFilter[i] && channel_index[i] == num)
			{
				m_chat_msg.Add(item);
				if (m_chat_msg.Count > 60)
				{
					m_chat_msg.RemoveRange(0, m_chat_msg.Count - 60);
				}
				rect_dirty = true;
				break;
			}
		}
	}

	private void recalcChatSize()
	{
		if (!rect_dirty)
		{
			return;
		}
		m_listRect.height = 0f;
		for (int i = 0; i < m_chat_msg.Count; i++)
		{
			if (m_chat_msg[i].section == null)
			{
				m_chat_msg[i].make((int)m_listRect.width, output_style);
			}
			m_listRect.height += m_chat_msg[i].section.height;
		}
		if (!m_showChat)
		{
			m_scroll_pos.y = m_listRect.height - m_text_area_rect.height;
		}
		else
		{
			if (m_reset_chat_scroll)
			{
				m_scroll_pos.y = m_listRect.height - m_text_area_rect.height;
				m_reset_chat_scroll = false;
			}
			else if (Mathf.Abs(m_bottom_scroll_pos.y - m_scroll_pos.y) < 5f)
			{
				m_scroll_pos.y = m_listRect.height - m_text_area_rect.height;
			}
			m_bottom_scroll_pos.y = m_listRect.height - m_text_area_rect.height;
		}
		rect_dirty = false;
	}

	private void checkChannelStatus()
	{
		CHANNEL_STATUS cHANNEL_STATUS = CHANNEL_STATUS.NORMAL;
		GameObject gameObject = GameObject.Find("/TeamFight/TeamFightUI");
		TeamFightUI teamFightUI = null;
		if (gameObject != null)
		{
			teamFightUI = gameObject.GetComponent<TeamFightUI>();
		}
		if (teamFightUI != null && teamFightUI.enabled)
		{
			cHANNEL_STATUS = CHANNEL_STATUS.TEAMFIGHT;
		}
		else if (MainUI._nation_check(string.Empty))
		{
			cHANNEL_STATUS = CHANNEL_STATUS.NATION;
		}
		AssetBundle asset = BaseAsset.was.GetAsset("ui_chat");
		if (cHANNEL_STATUS != channel_status)
		{
			current_send_channel = 0;
			channel_status = cHANNEL_STATUS;
			switch (channel_status)
			{
			case CHANNEL_STATUS.NORMAL:
				SetChannelIndex(new uint[3] { 3u, 2u, 1u });
				enableChannelNum = 2;

	private void SetChannelIndex(uint[] channel_list)
	{
		bool[] array = new bool[channel_list.Length];
		for (int i = 0; i < channel_list.Length; i++)
		{
			array[i] = true;
			for (int j = 0; j < channel_index.Length; j++)
			{
				if (channel_index[j] == channel_list[i])
				{
					array[i] = selectedFilter[j];
					break;
				}
			}
		}
		channel_index = channel_list;
		selectedFilter = array;
	}

	private void combine_msg_buffer()
	{
		m_chat_msg.Clear();
		for (int i = 0; i < selectedFilter.Length; i++)
		{
			if (selectedFilter[i])
			{
				uint key = channel_index[i];
				if (channel_buffer_msg.ContainsKey(key))
				{
					m_chat_msg.AddRange(channel_buffer_msg[key]);
				}
			}
		}
		InsertionSort(m_chat_msg, ChatSingleMsg.Compare);
		if (m_chat_msg.Count > 60)
		{
			m_chat_msg.RemoveRange(0, m_chat_msg.Count - 60);
		}
		rect_dirty = true;
	}

	private static void InsertionSort<T>(List<T> list, Comparison<T> comparison)
	{
		if (list == null)
		{
			throw new ArgumentNullException("list");
		}
		if (comparison == null)
		{
			throw new ArgumentNullException("comparison");
		}
		int count = list.Count;
		for (int i = 1; i < count; i++)
		{
			T val = list[i];
			int num = i - 1;
			while (num >= 0 && comparison(list[num], val) > 0)
			{
				list[num + 1] = list[num];
				num--;
			}
			list[num + 1] = val;
		}
	}
