	public static bool isMouseInChatBox()
	{
		if (!instance)
		{
			return false;
		}
		Vector3 mousePosition = Input.mousePosition;
		mousePosition.y = (float)Screen.height - mousePosition.y;
		foreach (KeyValuePair<int, ChatSession> active_session in instance.active_sessions)
		{
			if (active_session.Value.rc.Contains(mousePosition))
			{
				return true;
			}
		}
		return false;
	}

	public override void Start()
	{
		visible_count = false;
		base.Start();
		chat_behaviour = GetComponent<ChatBehaviour>();
		inspect_ui = GetComponent<InspectUI>();
		emote_container = EmoteContainer.Instance;
		if (emote_container != null)
		{
			emote_size.x = 0f;
			emote_size.y = 0f;
			for (int i = 0; i < emote_container.emotes.Length; i++)
			{
				Vector2 size = emote_container.emotes[i].size;
				if (size.x > emote_size.x)
				{
					emote_size.x = size.x;
				}
				if (size.y > emote_size.y)
				{
					emote_size.y = size.y;
				}
			}
			emote_box_size.x = emote_size.x * 6f;
			emote_box_size.y = emote_size.y * 4f;
		}
		instance = this;
	}

	private void Update()
	{
		if (pending_sessions.Count > 0)
		{
			int num = shown_pending_num;
			for (int i = 0; i < num; i++)
			{
				pending_sessions[i].current_pos = Mathf.MoveTowards(pending_sessions[i].current_pos, pending_sessions[i].target_pos, 10f);
			}
		}
	}

	public void Open(string name)
	{
		addSession(name);
		activateSession(name);
	}

	private void addSession(string name)
	{
		if (!sessions.ContainsKey(name))
		{
			ChatSession chatSession = new ChatSession(name);
			chatSession.scroll_pos = 0f;
			chatSession.rc = __BaseUI._rect(270f, 50f, 600f, 600f);
			sessions.Add(name, chatSession);
		}
	}

	public void activateSession(string name)
	{
		if (!sessions.ContainsKey(name))
		{
			return;
		}
		if (!active_sessions.ContainsKey(sessions[name].id_win))
		{
			sessions[name].id_win = ++id_win_auto_incr;
			active_sessions[sessions[name].id_win] = sessions[name];
			_CalcHeight(name);
			for (int i = 0; i < pending_sessions.Count; i++)
			{
				if (pending_sessions[i].session.opname == name)
				{
					pending_sessions.RemoveAt(i);
					recalc_pending_pos();
					break;
				}
			}
			if (sessions[name].icon == null)
			{
				CSUserIcon_0x240 cSUserIcon_0x = new CSUserIcon_0x240();
				cSUserIcon_0x.name = name;
				RootUI.m_client.Send(576, cSUserIcon_0x, sc_user_icon);
			}
		}
		focuswid = sessions[name].id_win;
	}

	private void sc_user_icon(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCUserIcon_0x240 sCUserIcon_0x = JSProtoBuf.Serializer.Deserialize<SCUserIcon_0x240>(stream);
			if (sessions.ContainsKey(sCUserIcon_0x.name))
			{
				AssetBundle asset = BaseAsset.was.GetAsset("ui_main");
				sessions[sCUserIcon_0x.name].icon = asset.Load("ui_zhu_xitong" + sCUserIcon_0x.icon) as Texture2D;
			}
		}
	}

					pending_sessions.Clear();
				}
			}
			num = shown_pending_num;

	private void DrawTooltip()
	{
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (Input.GetMouseButton(0))
		{
			GUI.tooltip = string.Empty;
		}
		if (Event.current.type == EventType.Repaint && GUI.tooltip != lastTooltip)
		{
			if (GUI.tooltip == string.Empty)
			{
				lastTooltip = string.Empty;
				tooltipUI.HideTooltip("MessengerUI");
			}
			else
			{
				tooltipUI.ShowTooltipCommon(GUI.tooltip, "MessengerUI");
				lastTooltip = GUI.tooltip;
			}
		}
	}

	private void _CalcHeight(string op_name)
	{
		if (!sessions.ContainsKey(op_name))
		{
			return;
		}
		ChatSession chatSession = sessions[op_name];
		if (!active_sessions.ContainsKey(chatSession.id_win))
		{
			return;
		}
		float num = 0f;
		foreach (Message message in chatSession.messages)
		{
			num += 27f;
			if (message.section == null)
			{
				message.section = new ScenarioSection();
				message.section.setParams((int)(242f * __BaseUI._screenScale), 1000, 14);
				message.section.makes_string(((!message.isme) ? "#c=FFFF00" : "#c=FFFFCC") + message.cont, outputStyle);
			}
			num += message.section.height;
		}
		chatSession.height = (int)num;
		if ((float)chatSession.height > __BaseUI._screenScale * 187f)
		{
			chatSession.scroll_pos = chatSession.height - (int)(__BaseUI._screenScale * 187f);
		}
		else
		{
			chatSession.scroll_pos = 0f;
		}
	}

	private void ReceivePrivateMessage(SCChatMessage_0x220 pkt)
	{
		if (pkt.content == null || pkt.content.Length == 0)
		{
			return;
		}
		Message message = new Message();
		message.cont = pkt.content;
		DateTime now = DateTime.Now;
		message.time = now.Hour + ":" + now.Minute + ":" + now.Second;
		string text = null;
		if (pkt.sender == Global.character_.name)
		{
			message.isme = true;
			text = pkt.receiver;
		}
		else
		{
			message.isme = false;
			text = pkt.sender;
		}
		if (!sessions.ContainsKey(text))
		{
			addSession(text);
		}
		sessions[text].messages.Add(message);
		if (!active_sessions.ContainsKey(sessions[text].id_win))
		{
			bool flag = false;
			for (int i = 0; i < pending_sessions.Count; i++)
			{
				if (pending_sessions[i].session.opname == text)
				{
					flag = true;
				}
			}
			if (!flag)
			{
				addPendingSession(sessions[text], text);
			}
		}
		else
		{
			_CalcHeight(text);
		}
	}

	private bool DrawEmoteBox()
	{
		Rect position = __BaseUI._rect(emote_box_pos.x, emote_box_pos.y - emote_box_size.y, emote_box_size.x, emote_box_size.y);
		GUI.Box(new Rect(position.x - 5f, position.y - 5f, position.width + 10f, position.height + 10f), string.Empty, __BaseUI._boxnew10Style);
		GUI.BeginGroup(position);
		for (int i = 0; i < 4; i++)
		{
			for (int j = 0; j < 6; j++)
			{
				int num = i * 6 + j;
				float x = (float)j * emote_size.x + (emote_size.x - emote_container.emotes[num].size.x) / 2f;
				float y = (float)i * emote_size.y + (emote_size.y - emote_container.emotes[num].size.y);
				float x2 = emote_container.emotes[num].size.x;
				float y2 = emote_container.emotes[num].size.y;
				if (num >= emote_container.emotes.Length)
				{
					continue;
				}
				emote_container.DrawIndex(num, __BaseUI._rect(x, y, x2, y2));
				if (GUI.Button(__BaseUI._rect((float)j * emote_size.x, (float)i * emote_size.y, emote_size.x, emote_size.y), string.Empty, styleEmoteSelect))
				{
					if (Global.getTemplate("emote_code", (uint)emote_container.emotes[num].id) is _emote_code emote_code)
					{
						ChatSession chatSession = session_using_emote;
						chatSession.input_string = chatSession.input_string + "#" + emote_code.codeChs;
						GUI.FocusControl("chat input");
					}
					PopupUI.EndPopup("chat_emote");
				}
			}
		}
		GUI.EndGroup();
		Vector3 mousePosition = Input.mousePosition;
		mousePosition.y = (float)Screen.height - mousePosition.y;
		return position.Contains(mousePosition);
	}

	private void CloseEmoteBox()
	{
		show_emote_box = false;
	}

	private void addPendingSession(ChatSession s, string tempname)
	{
		PendingChatSession pendingChatSession = new PendingChatSession(s);
		pendingChatSession.current_pos = pending_w;
		pending_sessions.Add(pendingChatSession);
		recalc_pending_pos();
		FlowerMessUI.FMUI f = new FlowerMessUI.FMUI(b_pending, tempname, callbackOnClickPending);
		BaseUI.GetRootUI().GetComponent<FlowerMessUI>().ADDFM(f);
	}

	private void recalc_pending_pos()
	{
		int num = shown_pending_num;
		float num2 = (pending_w - (float)(57 * num)) / 2f;
		for (int i = 0; i < num; i++)
		{
			pending_sessions[i].target_pos = num2 + (float)(i * 57);
		}
	}
