	private void Update()
	{
		if (clear_tooltip || ownerUI != last_ownerUI || tooltip != last_tooltip)
		{
			if (ownerUI == null || tooltip == string.Empty || tooltip == "Background")
			{
				HideTooltip(string.Empty);
			}
			else
			{
				ownerUI.drawTooltips(tooltip);
			}
			last_ownerUI = ownerUI;
			last_tooltip = tooltip;
		}
		if (clear_tooltip)
		{
			ownerUI = null;
			tooltip = string.Empty;
		}
		tttitle = TooltipFlag;
		if (needadjust && TooltipFlag == "General")
		{
			if (GenInfo.gen_active_skill == string.Empty)
			{
				ttgenskill = false;
			}
			else
			{
				ttgenskill = true;
			}
			if (GenInfo.gen_speicalies == string.Empty)
			{
				ttgenspec = false;
			}
			else
			{
				ttgenspec = true;
			}
			needadjust = false;
		}
		else if (needadjust && TooltipFlag == "Item")
		{
			if (ItemInfo.maxlevel > 0)
			{
				ttitem_isenchantable = true;
			}
			else
			{
				ttitem_isenchantable = false;
			}
			if (ItemInfo.basecount > 0)
			{
				ttitem_base = true;
			}
			else
			{
				ttitem_base = false;
			}
			if (ItemInfo.addcount > 0)
			{
				ttitem_add = true;
			}
			else
			{
				ttitem_add = false;
			}
			if (ItemInfo.specname == string.Empty)
			{
				ttitem_spec = false;
			}
			else
			{
				ttitem_spec = true;
			}
			if (ItemInfo.setname == string.Empty)
			{
				ttitem_set = false;
			}
			else
			{
				ttitem_set = true;
			}
			needadjust = false;
		}
	}

	private void DrawTooltipItem()
	{
		int num = 205;
		int num2 = 334;
		float num3 = tippos.x + 10f;
		float num4 = (float)Screen.height - tippos.y;
		int num5 = 18;
		int num6 = 0;
		if (!ttitem_isenchantable)
		{
			num6++;
		}
		num6 += 4 - ItemInfo.basecount;
		if (ItemInfo.basecount == 0)
		{
			num6++;
		}
		num6 += 3 - ItemInfo.addcount;
		if (ItemInfo.addcount == 0)
		{
			num6++;
		}
		if (!ttitem_spec)
		{
			num6 += 2;
		}
		else
		{
			num2 += (int)text_warp_white.CalcHeight(new GUIContent(ItemInfo.specdesc), 190f);
		}
		if (!ttitem_set)
		{
			num6 += 2;
		}
		else
		{
			num2 += (int)text_warp_white.CalcHeight(new GUIContent(ItemInfo.setdesc), 190f);
		}
		if (!ttitem_spec || !ttitem_set)
		{
			num6++;
		}
		num2 -= num5 * num6;
		if (num3 + (float)num > (float)Screen.width)
		{
			num3 = Screen.width - num;
		}
		if (num4 + (float)num2 > (float)Screen.height)
		{
			num4 = Screen.height - num2;
		}
		GUI.Label(new Rect(num3, num4, num, num2), string.Empty, __BaseUI._bgTooltip);
		GUILayout.BeginArea(new Rect(num3 + 6f, num4 + 5f, num - 15, num2 - 10));
		GUILayout.BeginVertical();
		GUIStyle gUIStyle = new GUIStyle();
		newtextstyle(gUIStyle);
		gUIStyle.normal.textColor = ItemInfo.name_color;
		GUILayout.BeginHorizontal();
		GUILayout.Label(ItemInfo.type, text_left_gold);
		GUILayout.EndHorizontal();
		string text;
		if (ttitem_isenchantable)
		{

	public static string make_ArmySpecString(_army_template army_temp)
	{
		string text = string.Empty;
		if (army_temp.spec1 != 0)
		{
			_speciality speciality = Global.getTemplate("speciality", army_temp.spec1) as _speciality;
			text = text + "【" + speciality.name + "】";
		}
		if (army_temp.spec2 != 0)
		{
			_speciality speciality2 = Global.getTemplate("speciality", army_temp.spec2) as _speciality;
			text = text + "【" + speciality2.name + "】";
		}
		if (army_temp.spec3 != 0)
		{
			_speciality speciality3 = Global.getTemplate("speciality", army_temp.spec3) as _speciality;
			text = text + "【" + speciality3.name + "】";
		}
		if (Global.character_.armystars.ContainsKey(army_temp.id))
		{
			text = string.Empty;
			ushort id = army_temp.id;
			ushort num = Global.character_.armystars[id];
			if (Global.getTemplate("army_star_start", id) is _army_star_start { start_id: var start_id, max_id: var max_id })
			{
				int num2 = start_id + num - 1;
				if (num2 <= max_id)
				{
					int num3 = 0;
					for (int i = start_id; i <= num2; i++)
					{
						if (Global.getTemplate("army_star", (uint)i) is _army_star { type: "spec" } army_star)
						{
							_speciality speciality4 = Global.getTemplate("speciality", (uint)army_star.param) as _speciality;
							num3++;
							text = text + "【" + speciality4.name + "】";
							if (num3 == 3)
							{
								break;
							}
						}
					}
				}
			}
			foreach (object value in Global.getTemplates("talent_template").Values)
			{
				_talent_template talent_template = (_talent_template)value;
				if (talent_template.armyid == id && talent_template.spec == 18 && Global.character_.campaigns_.ContainsKey(talent_template.idcampaign) && Global.character_.campaigns_[talent_template.idcampaign].finished && Global.getTemplate("speciality", (uint)talent_template.param) is _speciality speciality5)
				{
					text = text + "【" + speciality5.name + "】";
				}
			}
		}
		return text;
	}
