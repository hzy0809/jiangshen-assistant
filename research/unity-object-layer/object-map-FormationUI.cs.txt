	private void Update()
	{
		if (targetpos != leftpos)
		{
			float deltaTime = Time.deltaTime;
			float num = ((targetpos <= leftpos) ? ((0f - basestep) * 4f) : (basestep * 4f));
			float num2 = deltaTime * num;
			if ((float)Math.Abs(targetpos - leftpos) < Math.Abs(num2))
			{
				leftpos = targetpos;
				moving = false;
				if (leftpos == 0)
				{
					opened = false;
					switch_style = form_graph.toopen;
				}
				else
				{
					switch_style = form_graph.toclose;
				}
			}
			else
			{
				leftpos += (int)num2;
			}
		}
		if (skilltip_side != -1)
		{
			skilltip_timer += (float)Math.PI * Time.deltaTime;
			if (skilltip_timer > (float)Math.PI)
			{
				skilltip_timer = 0f;
			}
		}
		if (hotpanel_timer > 0f)
		{
			hotpanel_timer -= Time.deltaTime;
			if (hotpanel_timer < 0f && !hotpanel)
			{
				hotpanel_name = string.Empty;
				cur_hottip = string.Empty;
			}
		}
	}

		}

		if (hotpanel && hotpanel_timer < 0f)

			m_dictSkill.Clear();
			foreach (KeyValuePair<string, Rect> hotrect in hotrects)
			{
				if (hotrect.Key.Contains("skill"))
				{
					uint key = uint.Parse(hotrect.Key.Replace("skill", string.Empty));

					{

					}

				if (!GUI.Button(hotrect.Value, string.Empty, __BaseUI.normalStyle))
				{
					continue;
				}
				if (!RootUI.check_mobile() || cur_hottip == hotrect.Key)
				{
					if (hotrect.Key.Contains("formation"))
					{
						int num = int.Parse(hotrect.Key.Replace("formation", string.Empty));
						foreach (FormationInfo item in AllFormation)
						{
							if (item.guid == num && CharFormation.guid != num)
							{
								OnClickFormList(item);
								Global.character_.removehint(1u, (uint)num);
								checkallhint();
							}
						}
					}
					else if (hotrect.Key.Contains("army"))
					{
						uint num2 = uint.Parse(hotrect.Key.Replace("army", string.Empty));
						if (num2 != Global.character_.form_generals_[0].idarmy_template)
						{
							custom_armyselect = num2;

							{

	public override void drawTooltips(string tip)
	{
		if (tip.Contains("CharGeneral"))
		{
			ushort id = ushort.Parse(tip.Replace("CharGeneral", string.Empty));
			tipui.ShowTooltipGeneral(tt_owner, id, tip_refresh);
			if (tip_refresh)
			{
				tip_refresh = false;
			}
		}
		else if (tip.Contains("EnemyNpcGen"))
		{
			int num = int.Parse(tip.Replace("EnemyNpcGen", string.Empty));
			int i;
			for (i = 0; i < EnemyGeneralInfo.Count && EnemyGeneralInfo[i].locx * 4 + EnemyGeneralInfo[i].locy != num; i++)
			{
			}
			if (i < EnemyGeneralInfo.Count)
			{
				tipui.ShowTooltipGeneral(tt_owner, EnemyGeneralInfo[i]);
			}
		}
		else if (tip.Contains("PlayerEnemyGen"))
		{
			long num2 = long.Parse(tip.Replace("PlayerEnemyGen", string.Empty));
			int j;
			for (j = 0; j < EnemyGeneralInfo.Count && EnemyGeneralInfo[j].guid != num2; j++)
			{
			}
			if (j < EnemyGeneralInfo.Count)
			{
				tipui.ShowTooltipGeneral(tt_owner, EnemyGeneralInfo[j]);
			}
		}
		else if (tip.Contains("formation"))
		{
			string text = tip.Replace("formation", string.Empty);
			tipui.ShowTooltipCommon2(text, tt_owner, 340);
		}
		else if (tip.Contains("army"))
		{
			uint id2 = uint.Parse(tip.Replace("army", string.Empty));
			TooltipArmy tta = tipui.make_TooltipArmy(id2, tt_owner);
			tipui.ShowTooltipArmy(tta);
		}
		else if (tip.Contains("skill"))
		{
			uint num3 = uint.Parse(tip.Replace("skill", string.Empty));
			__BaseUI.GetTooltipUI().ShowTooltipSkill(tt_owner, (ushort)num3, string.Empty, string.Empty);
		}
		else if (tip.Contains("score"))
		{
			uint num4 = uint.Parse(tip.Replace("score", string.Empty));
			tipui.ShowTooltipCombatPoint(tt_owner, (ushort)num4);
		}
		else if (tip.Contains("common"))
		{
			string text2 = tip.Replace("common", string.Empty);
			tipui.ShowTooltipCommon2(text2, tt_owner, 250);
		}
		else if (tip.StartsWith("DuelGeneral"))
		{
			string[] array = tip.Replace("DuelGeneral", string.Empty).Split(',');
			if (array.Length == 2)
			{
				tipui.ShowTooltipGR_2(tt_owner, uint.Parse(array[0]), null, locked: false, uint.Parse(array[1]), PresetDuelUI.player_level, _generalbook: true);
			}
		}
	}

	public void SubLoading(Hashtable param)
	{
		m_lastparam = param;
		if (!param.ContainsKey("subui") || !StartCode.Contains(param["subui"].ToString()))
		{
			return;
		}
		drag_gen = -1;
		onfight = false;
		OnClose = null;
		substate = string.Empty;
		if (param["subui"].ToString() == "Formation")
		{
			RightPanel = "GenInfo";
			DefaultState = param["subui"].ToString();
			viewonly = true;
			if (param.ContainsKey("OnClose"))
			{
				OnClose = param["OnClose"] as delegate_OnClose;
			}
			InitNormalUI();
		}
		else if (param["subui"].ToString() == "NPCBattle")
		{
			RightPanel = "Enemy";
			substate = param["substate"] as string;
			DefaultState = param["subui"].ToString();
			Campaignid = long.Parse(param["campaignid"].ToString());
			Troopid = int.Parse(param["troopid"].ToString());
			viewonly = (bool)param["viewonly"];
			Logger.Log("viewonly" + viewonly);
			if (param.ContainsKey("isguide"))
			{
				m_bFromGuide = (bool)param["isguide"];
			}
			etc_data = ((!param.ContainsKey("etc")) ? null : param["etc"]);
			OnAttack = param["OnAttack"] as delegate_OnAttack;
			InitBattleUI();
		}
		else if (param["subui"].ToString() == "RaidBattle")
		{
			RightPanel = "Enemy";
			substate = "guild_raid";
			DefaultState = "NPCBattle";
			viewonly = (bool)param["viewonly"];
		}
		else if (param["subui"].ToString() == "GuildWarBattle")
		{
			RightPanel = "Enemy";
			substate = "guild_war";
			DefaultState = "PlayerBattle";
			viewonly = (bool)param["viewonly"];
		}
		else if (param["subui"].ToString() == "PlayerBattle")
		{
			RightPanel = "Enemy";
			DefaultState = param["subui"].ToString();
			substate = param["substate"] as string;
			viewonly = (bool)param["viewonly"];
			Legion = param["legion"] as BattleLegion;
			OnAttack = param["OnAttack"] as delegate_OnAttack;
			OnClose = param["OnClose"] as delegate_OnClose;
			InitBattleUI();
		}
		else if (param["subui"].ToString() == "SkyTowerFormation")
		{
			RightPanel = "GenInfo";
			DefaultState = "Formation";
			viewonly = true;
			OnClose = param["OnClose"] as delegate_OnClose;
			InitNormalUI();
		}
		else if (param["subui"].ToString() == "Woodwar")
		{
			RightPanel = "GenInfo";
			DefaultState = "NPCBattle";
			viewonly = false;
			OnAttack = param["OnAttack"] as delegate_OnAttack;
			OnClose = param["OnClose"] as delegate_OnClose;
			InitNormalUI();
		}
		else if (param["subui"].ToString() == "PresetDuel")
		{
			RightPanel = "PresetDuel";
			DefaultState = "Formation";
			viewonly = false;
			OnClose = param["OnClose"] as delegate_OnClose;
			InitPresetDuel();
		}
		else if (param["subui"].ToString() == "Ladder")
		{
			RightPanel = "Ladder";
			DefaultState = "PlayerBattle";
			substate = param["substate"] as string;
			viewonly = (bool)param["viewonly"];
			Legion = param["legion"] as BattleLegion;
			OnAttack = param["OnAttack"] as delegate_OnAttack;
			OnClose = param["OnClose"] as delegate_OnClose;
			InitBattleUI();
		}
		showCloseTip = false;
	}

	private void InitialTeamData()
	{
		PitchGenerals.Clear();
		foreach (GeneralInfo item in SquadGeneralInfo)
		{
			bool flag = false;
			for (int i = 0; i < 4; i++)
			{
				for (int j = 0; j < 4; j++)
				{
					if (!flag && item.guid == CharFormation.formgenpos[i, j].genid)
					{
						int index = CharFormation.formgenpos[i, j].genspot - 1;
						item.locx = CharFormation.dbspot[index] / 4;
						item.locy = CharFormation.dbspot[index] % 4;
						item.modx = CharFormation.formgenpos[i, j].modx;
						item.mody = CharFormation.formgenpos[i, j].mody;
						PitchGenerals.Add(item);
						flag = true;
					}
				}
			}
			if (!flag)
			{
				item.locx = 0;
				item.locy = 0;
				item.modx = 0;
				item.modx = 0;
			}
		}
		if (PitchGenerals.IndexOf(SquadGeneralInfo[0]) != -1)
		{
			CurFormSelect = 0;
			SquadSelect = -1;
			EnemySelect = -1;
		}
		else
		{
			SetCurFormSelect(-1);
			SquadSelect = 0;
			EnemySelect = -1;
		}
	}

	private void OnClickCurFormSpot(int i, int j)
	{
		if (!(RightPanel != "Equipment"))
		{
			return;
		}
		if (SquadSelect >= 0)
		{
			if (SquadGeneralInfo[SquadSelect].isBig)
			{
				return;
			}
			int pitchGenIndex = GetPitchGenIndex(i, j);
			if (pitchGenIndex != -1 && PitchGenerals[pitchGenIndex].guid == 0L)
			{
				__BaseUI.GetMessageBox().setMessage(__BaseUI.load_str(20246u), __BaseUI.load_str(20247u), rmg);
				drag_gen = -1;
				return;
			}
			GeneralInfo generalInfo = SquadGeneralInfo[SquadSelect];
			if (pitchGenIndex == -1)
			{
				if (CharFormation.formgenpos[i, j].spot > 0 && CharFormation.formgenpos[i, j].spot <= CharFormation.spotcount)
				{
					generalInfo.locx = i;
					generalInfo.locy = j;
					PitchGenerals.Add(generalInfo);
					SetCurFormSelect(PitchGenerals.Count - 1);
					SetupCharFormGenPos(i, j, generalInfo.guid);
					if (RightPanel != "PresetDuel")
					{
						Join_Form_Equipment_Check((ushort)generalInfo.guid);
					}
					SquadSelect = -1;
					EnemySelect = -1;
					int num = CharFormation.dbspot.IndexOf(i * 4 + j);
				}
			}
			else
			{
				GeneralInfo generalInfo2 = PitchGenerals[pitchGenIndex];
				PitchGenerals.RemoveAt(pitchGenIndex);
				generalInfo.locx = generalInfo2.locx;
				generalInfo.locy = generalInfo2.locy;
				PitchGenerals.Add(generalInfo);
				SetCurFormSelect(PitchGenerals.Count - 1);
				generalInfo2.locx = 0;
				generalInfo2.locy = 0;
				RemoveCharFormGenPos(generalInfo2.guid);
				SetupCharFormGenPos(generalInfo.locx, generalInfo.locy, generalInfo.guid);
				Exchange_Equipment_Check((ushort)generalInfo2.guid, (ushort)generalInfo.guid);
				SquadSelect = -1;
				EnemySelect = -1;
				int num2 = CharFormation.dbspot.IndexOf(generalInfo.locx * 4 + generalInfo.locy);
			}
		}
		else if (CurFormSelect >= 0)
		{
			if (PitchGenerals[CurFormSelect].isBig || PitchGenerals[CurFormSelect].guid == CharFormation.formgenpos[i, j].genid)
			{
				return;
			}
			int pitchGenIndex2 = GetPitchGenIndex(i, j);
			GeneralInfo generalInfo3 = PitchGenerals[CurFormSelect];
			if (pitchGenIndex2 == -1)
			{
				if (CharFormation.formgenpos[i, j].spot > 0 && CharFormation.formgenpos[i, j].spot <= CharFormation.spotcount)
				{
					SetupCharFormGenPos(generalInfo3.locx, generalInfo3.locy, -1L);
					SetupCharFormGenPos(i, j, generalInfo3.guid);
					generalInfo3.locx = i;
					generalInfo3.locy = j;
					int num3 = CharFormation.dbspot.IndexOf(i * 4 + j);
				}
			}
			else
			{
				int locx = PitchGenerals[pitchGenIndex2].locx;
				int locy = PitchGenerals[pitchGenIndex2].locy;
				int locx2 = generalInfo3.locx;
				int locy2 = generalInfo3.locy;
				if (!PitchGenerals[pitchGenIndex2].isBig)
				{
					PitchGenerals[pitchGenIndex2].locx = locx2;
					PitchGenerals[pitchGenIndex2].locy = locy2;
					SetupCharFormGenPos(locx2, locy2, PitchGenerals[pitchGenIndex2].guid);
					generalInfo3.locx = locx;
					generalInfo3.locy = locy;
					SetupCharFormGenPos(locx, locy, generalInfo3.guid);
				}
				int num4 = CharFormation.dbspot.IndexOf(locx * 4 + locy);
			}
			SetCurFormSelect(CurFormSelect);
		}
		else
		{
			int k;
			for (k = 0; k < PitchGenerals.Count && PitchGenerals[k].guid != CharFormation.formgenpos[i, j].genid; k++)
			{
			}
			if (k < PitchGenerals.Count)
			{
				SetCurFormSelect(k);
			}
		}
	}

	public static void calc_init_rage(ref int rage, _speciality spec)
	{
		if (spec.effect1 == "spec_add_rage_before_battle" && rage < int.Parse(spec.effect1_param1))
		{
			rage = int.Parse(spec.effect1_param1);
		}
		if (spec.effect2 == "spec_add_rage_before_battle" && rage < int.Parse(spec.effect2_param1))
		{
			rage = int.Parse(spec.effect1_param2);
		}
		if (spec.effect3 == "spec_add_rage_before_battle" && rage < int.Parse(spec.effect3_param1))
		{
			rage = int.Parse(spec.effect1_param3);
		}
	}

	private void cs_preset_formation(int genid, uint spot)
	{
		CSPresetDuel_0x2A8 cSPresetDuel_0x2A = PresetDuelUI.makeClientPkt();
		cSPresetDuel_0x2A.action = 7u;
		cSPresetDuel_0x2A.form_gen = new CSPresetDuel_0x2A8.FormGen();
		cSPresetDuel_0x2A.form_gen.idgen_temp = genid;
		cSPresetDuel_0x2A.form_gen.location = spot;
		if (genid >= 0)
		{
			for (int i = 0; i < m_curPresetDuelFormGen.general.Count; i++)
			{
				if (m_curPresetDuelFormGen.general[i].location == spot && m_curPresetDuelFormGen.general[i].idgen_temp != genid)
				{
					m_curPresetDuelFormGen.general[i].location = 0u;
					break;
				}
			}
			for (int j = 0; j < m_curPresetDuelFormGen.general.Count; j++)
			{
				if (m_curPresetDuelFormGen.general[j].idgen_temp == genid)
				{
					m_curPresetDuelFormGen.general[j].location = spot;
					break;
				}
			}
		}
		else
		{
			for (int k = 0; k < m_curPresetDuelFormGen.general.Count; k++)
			{
				if (m_curPresetDuelFormGen.general[k].location == spot && m_curPresetDuelFormGen.general[k].idgen_temp != 0)
				{
					m_curPresetDuelFormGen.general[k].location = 0u;
					break;
				}
			}
		}
		RootUI.m_client.Send(680, cSPresetDuel_0x2A, null);
	}

				if (CharFormation.formgenpos[i, j].genid == genid)
				{
					CharFormation.formgenpos[i, j].genid = -1L;
					CharFormation.formgenpos[i, j].modx = 0;
					CharFormation.formgenpos[i, j].mody = 0;
					CharFormation.formgenpos[i, j].genspot = 0;
				}
			}
		}
	}

	private int GetFormSpot(int startx, int starty, int endx, int endy)
	{
		if (endx > 4 || endy > 4)
		{
			return 0;
		}
		for (int i = startx; i < endx; i++)
		{
			for (int j = starty; j < endy; j++)
			{
				if (CharFormation.formgenpos[i, j].spot > 0 && CharFormation.formgenpos[i, j].spot <= CharFormation.spotcount)
				{
					return CharFormation.formgenpos[i, j].spot;
				}
			}
		}
		return 0;
	}

	private int calc_enemy_hp(_army info, int level)
	{
		uint soldier = info.soldier;
		float soldier_add = info.soldier_add;
		int spec_add = 0;
		int spec_perc = 0;
		calc_onespec_effect(info.spec1, ref spec_add, ref spec_perc);
		calc_onespec_effect(info.spec2, ref spec_add, ref spec_perc);
		calc_onespec_effect(info.spec3, ref spec_add, ref spec_perc);
		return (int)((info.soldier_add * (float)level + (float)info.soldier + (float)spec_add) * (100f + (float)spec_perc) / 100f);
	}

	private void calc_spec_hpeffect(string effect, ref int spec_add, ref int spec_perc, string param)
	{
		if (effect == "spec_hp")
		{
			spec_add += int.Parse(param);
		}
		else if (effect == "spec_hp_perc")
		{
			spec_perc += int.Parse(param);
		}
	}

	private void calc_onespec_effect(ushort specid, ref int spec_add, ref int spec_perc)
	{
		if (specid > 0)
		{
			_speciality speciality = Global.getTemplate("speciality", specid) as _speciality;
			calc_spec_hpeffect(speciality.effect1, ref spec_add, ref spec_perc, speciality.effect1_param1);
			calc_spec_hpeffect(speciality.effect2, ref spec_add, ref spec_perc, speciality.effect2_param1);
			calc_spec_hpeffect(speciality.effect3, ref spec_add, ref spec_perc, speciality.effect3_param1);
		}
	}

	public void checkallhint()
	{
		newhint_form = (newhint_army = (newhint_skill = false));
		armyskillhint = 0u;
		foreach (FormationInfo item in AllFormation)
		{
			if (item.available && Global.character_.check_hint(1u, (uint)item.formid))
			{
				newhint_form = true;
				break;
			}
		}
		foreach (ushort key in Global.character_.army_types.Keys)
		{
			if (Global.character_.check_hint(2u, key))
			{
				newhint_army = true;
				break;
			}
		}
		foreach (ushort key2 in Global.character_.skills.Keys)
		{
			if (!Global.character_.check_hint(3u, key2))
			{
				continue;
			}
			_skill skillTemplate = Global.getSkillTemplate(key2);
			_army_template armyTemplate = Global.getArmyTemplate((uint)skillTemplate.limited_army);
			if (Global.character_.form_generals_[0].idarmy_template != skillTemplate.limited_army)
			{
				newhint_armyskill = true;
				armyskillhint = (uint)skillTemplate.limited_army;
				armyskill_msg = __BaseUI.load_str(20251u, armyTemplate.name, skillTemplate.name);
				continue;
			}
			if (newhint_armyskill)
			{
				newhint_armyskill = false;
			}
			if (Global.character_.form_generals_[0].activeskill != key2)
			{
				newhint_skill = true;
			}
			armyskill_msg = __BaseUI.load_str(21250u, skillTemplate.name);
		}
	}

	private void FormEquipEnd()
	{
		if (formequip_state > 0)
		{
			formequip_state--;
		}
		if (formequip_state > 0)
		{
			return;
		}
		foreach (GeneralInfo item in SquadGeneralInfo)
		{
			FormGeneral formGeneral = Global.character_.form_generals_[(ushort)item.guid];
			item.attack = (int)formGeneral.param.attack;
			item.defence = (int)formGeneral.param.defense;
			item.spell = (int)formGeneral.param.spell;
			item.armyhp = (int)formGeneral.param.hp;

		}
		formequip_state = 0;
		Enable();
	}

	private void sc_autoequip(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			Enable();
			return;
		}
		SCAutoEquipGeneral_0x186 sCAutoEquipGeneral_0x = JSProtoBuf.Serializer.Deserialize<SCAutoEquipGeneral_0x186>(stream);
		if (sCAutoEquipGeneral_0x.state)
		{
			FormGeneral formGeneral = Global.character_.form_generals_[(ushort)sCAutoEquipGeneral_0x.genid];
			if (formGeneral.equipments[0] != null)
			{
				formGeneral.equipments[0].idgeneral_template = ushort.MaxValue;
				formGeneral.equipments[0] = null;
			}
			if (formGeneral.equipments[1] != null)
			{
				formGeneral.equipments[1].idgeneral_template = ushort.MaxValue;
				formGeneral.equipments[1] = null;
			}
			if (formGeneral.artifact != null)
			{
				formGeneral.artifact.idgeneral_template = ushort.MaxValue;
				formGeneral.artifact = null;
			}
			if (sCAutoEquipGeneral_0x.weaponid != 0 && Global.character_.equipments.ContainsKey((byte)sCAutoEquipGeneral_0x.weaponid))
			{
				formGeneral.equipments[0] = Global.character_.equipments[(byte)sCAutoEquipGeneral_0x.weaponid];
				formGeneral.equipments[0].idgeneral_template = formGeneral.idgeneral_template;
			}
			if (sCAutoEquipGeneral_0x.armorid != 0 && Global.character_.equipments.ContainsKey((byte)sCAutoEquipGeneral_0x.armorid))
			{
				formGeneral.equipments[1] = Global.character_.equipments[(byte)sCAutoEquipGeneral_0x.armorid];
				formGeneral.equipments[1].idgeneral_template = formGeneral.idgeneral_template;
			}
			if ((bool)arg)
			{
				EquipmentUI component = BaseUI.GetRootUI().GetComponent<EquipmentUI>();
				component.OnFormEquipEnd = FormEquipEnd;
				formequip_state++;
				component.cs_equip_auto_artifact(formGeneral.idgeneral_template);
			}
			FormEquipEnd();
		}
		else
		{
			Enable();
		}
	}

		AddAsset("ui_duel");
		BaseAsset.lds.Visible_sub();
		DisableAll();
		isLoading = true;
		CollectingStart(bCollecting: false);
		StartLoading();
		SetCurFormSelect(-1);
	}

	private void InitPresetDuelInfo()
	{
		InitFormInfo();
		m_curPresetDuelFormGen = ((PresetDuelUI.player_index != 0) ? PresetDuelUI.form_gen1 : PresetDuelUI.form_gen0);
		SquadSelect = 0;
		foreach (SCPresetDuel_0x2A8.General item in m_curPresetDuelFormGen.general)
		{
			if (Global.getTemplate("army", item.idarmy) is _army army)
			{
				GeneralInfo generalInfo = InitByArmy(army.id, army.name, (short)PresetDuelUI.player_level, 0, captain: false, -1);
				_general_template general_template = Global.getTemplate("general_template", item.idgen_temp) as _general_template;
				_army_template army_template = Global.getTemplate("army_template", army.idarmy_template) as _army_template;
				generalInfo.guid = item.idgen_temp;
				if (item.idgen_temp == 0)
				{
					generalInfo.generalName = Global.character_.name;
					generalInfo.faceid = Global.character_.icon;
				}
				else if (general_template != null)
				{
					generalInfo.generalName = general_template.name;
					generalInfo.faceid = general_template.icon;
				}
				if (item.idgen_temp == 0)
				{
					generalInfo.tooltip = "CharGeneral0";
				}
				else
				{
					generalInfo.tooltip = "DuelGeneral" + item.idgen_temp + "," + item.idarmy;
				}
				SquadGeneralInfo.Add(generalInfo);
			}
		}
		for (int i = 0; i < m_curPresetDuelFormGen.idformation.Count; i++)
		{
			int num = (int)m_curPresetDuelFormGen.idformation[i];
			_formation_template formation_template = Global.getTemplate("formation_template", (uint)num) as _formation_template;
			FormationInfo formationInfo = new FormationInfo();
			formationInfo.formationName = formation_template.name;
			formationInfo.available = true;
			formationInfo.desc = formation_template.desc;
			formationInfo.level = 1;
			formationInfo.formid = num;
			formationInfo.spotcount = 8;
			formationInfo.guid = num;
			formationInfo.currentform = false;
			formationInfo.iconid = formation_template.icon;
			for (int j = 0; j < 4; j++)
			{
				for (int k = 0; k < 4; k++)
				{
					formationInfo.formgenpos[j, k].spot = 0;
					formationInfo.formgenpos[j, k].genid = -2L;
					formationInfo.formgenpos[j, k].modx = 0;
					formationInfo.formgenpos[j, k].mody = 0;
					formationInfo.formgenpos[j, k].genspot = 0;
				}
			}
			formationInfo.dbspot.Add(formation_template.location1 - 1);
			formationInfo.dbspot.Add(formation_template.location2 - 1);
			formationInfo.dbspot.Add(formation_template.location3 - 1);
			formationInfo.dbspot.Add(formation_template.location4 - 1);
			formationInfo.dbspot.Add(formation_template.location5 - 1);
			formationInfo.dbspot.Add(formation_template.location6 - 1);
			formationInfo.dbspot.Add(formation_template.location7 - 1);
			formationInfo.dbspot.Add(formation_template.location8 - 1);
			foreach (int item2 in formationInfo.dbspot)
			{
				int num2 = item2 / 4;
				int num3 = item2 % 4;
				formationInfo.formgenpos[num2, num3].spot = formationInfo.dbspot.IndexOf(item2) + 1;
			}
			AllFormation.Add(formationInfo);
			if (m_curPresetDuelFormGen.cur_form == (uint)i)
			{
				CharFormation = formationInfo;
			}
		}
		if (CharFormation == null)
		{
			CharFormation = AllFormation[0];
			CharFormation.currentform = true;
		}
		InitialPresetTeamData();
		duel_op_gen_list.Clear();
		SCPresetDuel_0x2A8.FormGenInfo formGenInfo = ((PresetDuelUI.player_index != 0) ? PresetDuelUI.form_gen0 : PresetDuelUI.form_gen1);
		for (int l = 0; l < formGenInfo.general.Count - 1; l++)
		{
			PresetDuelOpponentGenInfo presetDuelOpponentGenInfo = new PresetDuelOpponentGenInfo();
			presetDuelOpponentGenInfo.gen_temp = Global.getTemplate("general_template", formGenInfo.general[l + 1].idgen_temp) as _general_template;
			presetDuelOpponentGenInfo.army = Global.getTemplate("army", formGenInfo.general[l + 1].idarmy) as _army;
			presetDuelOpponentGenInfo.tooltip = "DuelGeneral" + formGenInfo.general[l + 1].idgen_temp + "," + formGenInfo.general[l + 1].idarmy;
			if (presetDuelOpponentGenInfo.gen_temp != null)
			{
				AddAssetSub("face" + presetDuelOpponentGenInfo.gen_temp.icon);
			}
			duel_op_gen_list.Add(presetDuelOpponentGenInfo);
		}
		duel_op_form_list.Clear();
		for (int m = 0; m < formGenInfo.idformation.Count; m++)
		{
			PresetDuelOpponentFormInfo presetDuelOpponentFormInfo = new PresetDuelOpponentFormInfo();
			presetDuelOpponentFormInfo.form_temp = Global.getTemplate("formation_template", formGenInfo.idformation[m]) as _formation_template;
			if (presetDuelOpponentFormInfo.form_temp != null)
			{
				presetDuelOpponentFormInfo.tooltip = "formation" + presetDuelOpponentFormInfo.form_temp.name + "\n" + presetDuelOpponentFormInfo.form_temp.desc;
			}
			duel_op_form_list.Add(presetDuelOpponentFormInfo);
		}
	}

	private void InitialPresetTeamData()
	{
		PitchGenerals.Clear();
		for (int i = 0; i < 4; i++)
		{
			for (int j = 0; j < 4; j++)
			{
				CharFormation.formgenpos[i, j].genid = -2L;
			}
		}
		for (int k = 0; k < m_curPresetDuelFormGen.general.Count; k++)
		{
			GeneralInfo generalInfo = SquadGeneralInfo[k];
			if (m_curPresetDuelFormGen.general[k].location != 0)
			{
				int num = (int)(m_curPresetDuelFormGen.general[k].location - 1);
				long genid = m_curPresetDuelFormGen.general[k].idgen_temp;
				int num2 = CharFormation.dbspot[num] / 4;
				int num3 = CharFormation.dbspot[num] % 4;
				CharFormation.formgenpos[num2, num3].genid = genid;
				CharFormation.formgenpos[num2, num3].modx = 0;
				CharFormation.formgenpos[num2, num3].mody = 0;
				CharFormation.formgenpos[num2, num3].genspot = num + 1;
				generalInfo.locx = CharFormation.dbspot[num] / 4;
				generalInfo.locy = CharFormation.dbspot[num] % 4;
				generalInfo.modx = CharFormation.formgenpos[num2, num3].modx;
				generalInfo.mody = CharFormation.formgenpos[num2, num3].mody;
				PitchGenerals.Add(generalInfo);
			}
			else
			{
				generalInfo.locx = 0;
				generalInfo.locy = 0;
				generalInfo.modx = 0;
				generalInfo.modx = 0;
			}
		}
		if (PitchGenerals.IndexOf(SquadGeneralInfo[0]) != -1)
		{
			CurFormSelect = 0;
			SquadSelect = -1;
			EnemySelect = -1;
		}
		else
		{
			SetCurFormSelect(-1);
			SquadSelect = 0;
			EnemySelect = -1;
		}
	}

	private void PrepPresetDuelFormAsset()
	{
		AssetBundle asset = BaseAsset.was.GetAsset("ui_formation");
		for (int i = 0; i < duel_op_form_list.Count; i++)
		{
			duel_op_form_list[i].iconStyle.normal.background = asset.Load("icon_zhenxing" + duel_op_form_list[i].form_temp.icon) as Texture2D;
		}
	}
