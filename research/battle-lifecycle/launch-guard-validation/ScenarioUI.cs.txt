	private void Update()
	{
		if (!m_bSectionEnd && Input.GetMouseButtonDown(0) && !rcSkip.Contains(new Vector3(Input.mousePosition.x, (float)Screen.height - Input.mousePosition.y, 0f)))
		{
			m_bIngoreTextSpeed = true;
		}
		if ((m_scenario.sections.Count == 0 || m_bSectionEnd) && Input.GetMouseButtonDown(0) && !m_bHasNewUI && !rcSkip.Contains(new Vector3(Input.mousePosition.x, (float)Screen.height - Input.mousePosition.y, 0f)))
		{
			m_bNext = true;
		}
		if (m_bHasNewUI && !m_bMoving && Input.GetMouseButtonDown(0))
		{
			m_bMoving = true;
		}
		if (m_bMoving)
		{
			for (int i = 0; i < newUIList.Count; i++)
			{
				newUIList[i].curPos = newUIMove(newUIList[i].curPos, newUIList[i].targetPos);
				if (newUIList[i].curPos == newUIList[i].targetPos)
				{
					newUIList[i].bMovingEnd = true;
				}
			}
		}
		int j;
		for (j = 0; j < newUIList.Count && newUIList[j].bMovingEnd; j++)
		{
		}
		if (j == newUIList.Count && newUIList.Count != 0)
		{
			m_bNext = true;
			for (int k = 0; k < newUIList.Count; k++)
			{
				newUIList[k].bMovingEnd = false;
			}
			newUIList.Clear();
			Global.character_.newuis_.Clear();
		}
		if (m_showLight && !m_bMoving)
		{
			if (newUI_anim == 0)
			{
				newUI_alpha -= Time.deltaTime;
				if ((double)newUI_alpha < 0.1)
				{
					newUI_alpha = 0.1f;
					newUI_anim = 1;
				}
			}
			else if (newUI_anim == 1)
			{
				newUI_alpha += Time.deltaTime;
				if (newUI_alpha > 1f)
				{
					newUI_alpha = 1f;
					newUI_anim = 0;
				}
			}
		}
		else
		{
			newUI_anim = 0;
		}
	}

	public void setScenario(Scenario s, RootMessenger prevUI, delegate_OnClose onClose)
	{
		m_prevUI = prevUI;
		m_bIngoreTextSpeed = false;
		m_bSectionEnd = false;
		m_bNext = false;
		OnClose = onClose;
		AddAsset("ui_scenario");
		m_scenario = s;
		if (s.sections == null)
		{
			s.make(__BaseUI._scenarioText);
		}
		foreach (ScenarioSection section in s.sections)
		{
			if (section.texIcon == null && section.idgeneral_template >= 0)
			{
				if (section.idgeneral_template == 0)
				{
					AddAsset("face" + Global.character_.icon, bForce: false);
				}
				else if (Global.getTemplate("general_template", section.idgeneral_template) is _general_template general_template)
				{
					AddAsset("face" + general_template.icon, bForce: false);
				}
			}
		}
		m_iSection = 0;
		m_bFirst = true;
		m_starttime = DateTime.Now;
		m_time = 0.0;
		m_bTrigger = false;
		BaseAsset.lds.Visible_sub();
		CollectingStart(bCollecting: false);
		StartLoading();
	}

	{

		{
			FunctionLoad(null);

			{
				Close();
				return;
			}
			cs_scenario_end();
		}
		if (!m_bMoving)
		{
			GUI.DrawTexture(rc, texNext);
		}
	}

			if (!BeginGUI(Screen.width - 10, Screen.height - 10, __BaseUI.UIDepth_Scenario, bgStyle, null))
			{

		}

				m_time += timeSpan.TotalMilliseconds;

			}
			if (m_bSectionEnd)
			{
				if (!m_bTrigger)
				{
					triggerFunction(scenarioSection);
				}
				if (m_scenario.sections.Count > m_iSection + 1)
				{
					if (m_bNext)

						FunctionLoad(FunctionClose);

						m_starttime = DateTime.Now;
						m_time = 0.0;
						m_bIngoreTextSpeed = false;
						m_bTrigger = false;

				else if (newUIList.Count > 0)
				{
					if (m_scenario.sections.Count == m_iSection + 1 && m_bNext)
					{
						m_iSection++;
						m_bSectionEnd = false;
						m_bNext = false;

			{

			}

			{

			}
			Close(rect);
		}

	private void triggerFunction(ScenarioSection ss)
	{
		m_bTrigger = true;
		if (ss.triggers == null)
		{
			return;
		}
		foreach (_scenario_section_trigger trigger in ss.m_triggers)
		{
			if (!(Global.getTemplate("scenario_trigger", trigger.idtrigger) is _scenario_trigger scenario_trigger))
			{
				continue;
			}
			if (scenario_trigger.type == "button")
			{
				uint result = 0u;
				uint.TryParse(trigger.param0, out result);
				if (!(Global.getTemplate("ui_button", result) is _ui_button ui_button))
				{
					continue;
				}
				if (ui_button.main)
				{
					newUIBtn item = new newUIBtn(ui_button.name);
					newUIList.Add(item);
					if (ui_button.name == "quiz")
					{
						Global.character_.newuis_.Add("huodong");
					}
					else
					{
						Global.character_.newuis_.Add(ui_button.name);
					}
					m_bWantNewUI = true;
				}
				else if (ui_button.name == "mg_yangjian")
				{
					newUIBtn item2 = new newUIBtn("chengjiang");
					newUIList.Add(item2);
					m_bWantNewUI = true;
				}
				else if (ui_button.name == "mg_doumuyuanjun")
				{
					newUIBtn item3 = new newUIBtn("chengjiang");
					newUIList.Add(item3);
					m_bWantNewUI = true;
				}
				Global.character_.addUI(ui_button.name);
				Global.setFirstCheck(ui_button.name, bCheck: true);
				MainUI componentInChildren = BaseAsset.lds.GetCurUI().GetComponentInChildren<MainUI>();
				if (componentInChildren != null)
				{
					componentInChildren.expandUI(ui_button.name);
				}
				if (result == 50)
				{
					Global.character_.grave.buytimes = 0;
					Global.character_.grave.times = 6;
					Global.character_.grave.newgrave = 1;
					if (Global.character_.graves.Count < 1)
					{
						_character_grave character_grave = new _character_grave();
						character_grave.idgrave = 1;
						character_grave.isfinish = false;
						character_grave.isopen = true;
						Global.character_.graves.Add(character_grave.idgrave, character_grave);
					}
				}
			}
			else if (scenario_trigger.type == "guide")
			{
				uint result2 = 0u;
				uint.TryParse(trigger.param0, out result2);
				if (Global.getTemplate("ui_button", result2) is _ui_button ui_button2)
				{
					Global.setGuideCheck(ui_button2.name, bCheck: true);
				}
			}
		}
	}

	private void FunctionLoad(delegate_OnClose onClose)
	{
		if (m_scenario.sections.Count == 0)
		{
			return;
		}
		ScenarioSection scenarioSection = m_scenario.sections[m_scenario.sections.Count - 1];
		if (scenarioSection.triggers == null)
		{
			return;
		}
		foreach (_scenario_section_trigger trigger in scenarioSection.m_triggers)
		{
			if (Global.getTemplate("scenario_trigger", trigger.idtrigger) is _scenario_trigger { type: "button_function" })
			{
				uint result = 0u;
				uint.TryParse(trigger.param0, out result);
				uint result2 = 0u;
				uint.TryParse(trigger.param1, out result2);
				if (Global.getTemplate("ui_button", result) is _ui_button)
				{
					ScenarioFunctionUI component = GetComponent<ScenarioFunctionUI>();
					component.MyLoading(result, result2 == 1, onClose);
					Hide();
				}
			}
		}
	}

	private void FunctionClose()
	{
		Visible();
	}

	private void cs_scenario_end()
	{
		CSScenarioEnd_0x410 cSScenarioEnd_0x = new CSScenarioEnd_0x410();
		cSScenarioEnd_0x.idscenario = m_scenario.id;
		RootUI.m_client.Send(1040, cSScenarioEnd_0x, sc_scenario_end);
	}

	private void sc_scenario_end(MemoryStream stream, object arg)
	{
		Close();
		if (stream == null)
		{
			return;
		}
		SCScenarioEnd_0x410 sCScenarioEnd_0x = JSProtoBuf.Serializer.Deserialize<SCScenarioEnd_0x410>(stream);
		Scenario scenario = m_scenario;
		if (scenario == null || scenario.sections.Count <= 0)
		{
			return;
		}
		foreach (ScenarioSection section in scenario.sections)
		{
			if (section.triggers == null)
			{
				continue;
			}
			foreach (_scenario_section_trigger trigger in section.m_triggers)
			{
				if (Global.getTemplate("scenario_trigger", trigger.idtrigger) is _scenario_trigger scenario_trigger)
				{
					if (scenario_trigger.type == "tutorial")
					{
						int result = 0;
						int.TryParse(trigger.param0, out result);
					}
					else if (!(scenario_trigger.type == "quest") && !(scenario_trigger.type == "formation") && !(scenario_trigger.type == "formation_location") && scenario_trigger.type == "newbie")
					{
						int result2 = 0;
						int.TryParse(trigger.param0, out result2);
						NewbieQuest item = new NewbieQuest((ushort)result2);
						Global.character_.newbiequests.Add(item);
					}
				}
			}
		}
	}

	public void scenarioSkip(Rect rcNext)
	{
		for (int i = m_iSection; i < m_scenario.sections.Count; i++)
		{
			if (!m_bTrigger)
			{
				triggerFunction(m_scenario.sections[i]);
			}
			m_bTrigger = false;
			m_iSection++;
		}
		m_bTrigger = true;
		if (newUIList.Count > 0)
		{
			m_iSection++;
			m_bSectionEnd = false;
			m_bNext = false;
			setNewUITargetPos();
			setNewUIParma();
			if (!m_bMoving)
			{
				GUI.DrawTexture(rcNext, texNext);
			}
		}
		else
		{
			FunctionLoad(null);
			if (m_scenario.param0 == "system")
			{
				cs_scenario_end();
			}
			else
			{
				Close();
			}
		}
	}

	public void setNewUITargetPos()
	{
		MainUI componentInChildren = BaseAsset.lds.GetCurUI().GetComponentInChildren<MainUI>();
		for (int i = 0; i < newUIList.Count; i++)
		{
			if (newUIList[i].name == "quiz")
			{
				newUIList[i].targetPos = componentInChildren.getButtonPos("huodong");
			}
			else if (newUIList[i].name == "chengjiang")
			{
				if (MainUI.topBarMood == MainUI.TOPBARMOOD.NORMAL)
				{
					newUIList[i].targetPos = componentInChildren.getButtonPos("divine_general");
				}
				else if (MainUI.topBarMood == MainUI.TOPBARMOOD.SHRINK)
				{
					newUIList[i].targetPos = new Vector2(Screen.width - 260 - 87, 11f);
				}
			}
			else if (!componentInChildren.isButtonShrinked(newUIList[i].name))
			{
				newUIList[i].targetPos = componentInChildren.getButtonPos(newUIList[i].name);
			}
			else
			{
				newUIList[i].targetPos = new Vector2(Screen.width - 260 - 87, 11f);
			}
		}
	}

			int index = newUIList.Count / 2;

	public Vector2 newUIMove(Vector2 current, Vector2 target)
	{
		if (target == current)
		{
			return current;
		}
		Vector2 vector = target - current;
		Vector2 normalized = vector.normalized;
		current += normalized * Time.deltaTime * 500f;
		if (vector.magnitude < 15f)
		{
			current = target;
		}
		return current;
	}
