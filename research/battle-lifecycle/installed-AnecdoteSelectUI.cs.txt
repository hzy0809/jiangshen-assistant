	public override void Update()
	{
		if (m_campaign != null && m_campaign.mesh != null)
		{
			GameObject gameObject = null;
			InGUIBehaviour component = m_campaign.mesh.GetComponent<InGUIBehaviour>();
			Ray ray = component.InGUI_Camera.ScreenPointToRay(Input.mousePosition);
			foreach (CampaignTroop value in m_campaign.troops_.Values)
			{
				if (!(value.mesh == null) && value.mesh.collider.Raycast(ray, out var _, 9999f))
				{
					gameObject = value.mesh;
				}
			}
			if (gameObject != HoverMesh)
			{
				if (HoverMesh != null)
				{
					HoverMesh.BroadcastMessage("OnMouseExit", SendMessageOptions.DontRequireReceiver);
				}
				if (gameObject != null)
				{
					gameObject.BroadcastMessage("OnMouseEnter", SendMessageOptions.DontRequireReceiver);
				}
				HoverMesh = gameObject;
			}
			if (gameObject != null)
			{
				if (Input.GetMouseButtonDown(0))
				{
					gameObject.BroadcastMessage("OnMouseDown", SendMessageOptions.DontRequireReceiver);
				}
				if (Input.GetMouseButtonUp(0))
				{
					gameObject.BroadcastMessage("OnMouseUp", SendMessageOptions.DontRequireReceiver);
				}
			}
		}
		m_fTime += Time.deltaTime;
		if (m_fTime > 2f)
		{
			m_fTime = 0f;
		}
		if (m_fTime < 0.75f)
		{
			m_fNewAlpha = m_fTime / 0.75f;
		}
		else if (m_fTime < 1.25f)
		{
			m_fNewAlpha = 1f;
		}
		else
		{
			m_fNewAlpha = (2f - m_fTime) / 0.75f;
		}
	}

		foreach (CampaignTroop value in m_campaign.troops_.Values)
		{
			value.arrow_mesh = null;
			value.mesh = null;
		}
		m_campaign = null;
		GetComponent<CampaignTroopUI>().Hide();
		EnableAll();
	}

	private void LoadChapter(Database.Campaign c)
	{
		if (c == null)
		{
			return;
		}
		if (c.texIcon == null && c.m_troopsArray.Count > 0)
		{
			_campaign_troop campaign_troop = c.m_troops[c.m_troopsArray[c.m_troopsArray.Count - 1]] as _campaign_troop;
			c.texIcon = BaseAsset.LoadTexture("face" + campaign_troop.icon);
		}
		foreach (_Treasure drop in c.drops)
		{
			if (drop.texIcon == null)
			{
				drop.texIcon = BaseAsset.LoadTexture(drop.getIconName());
			}
		}
	}

	protected override void AssetLoaded(string name)
	{
		if (m_campaign == null)

			_ChapterCampaigns chapter = m_chapter;

			{
				_AnecdoteCampaign anecdoteCampaign = chapter.campaigns[num];
				LoadChapter(anecdoteCampaign.campaign);
			}
		}
	}

			int num2 = (int)(scrollPos.y / 130f);
			if (num2 >= chapter.campaigns.Count)
			{
				num2 = chapter.campaigns.Count - 1;
			}
			if (num2 < 0)
			{
				num2 = 0;
			}
			int num3 = ((num2 + 4 <= chapter.campaigns.Count) ? (num2 + 4) : chapter.campaigns.Count);
			for (int i = num2; i < num3; i++)
			{
				_AnecdoteCampaign ac = chapter.campaigns[i];
				drawAnecdote2(i + 1, __BaseUI._rect(0f, i * 130, 590f, 140f), ac);
			}
			GUIHacks.EndScrollView();
			__BaseUI.Label(__BaseUI._rect(0f, 0f, 590f, 16f), string.Empty, ui_mengban_top);
			__BaseUI.Label(__BaseUI._rect(0f, h - 80f - 16f, 590f, 16f), string.Empty, ui_mengban_down);

		}
		if (pktNew.ids.Count > 0)
		{
			RootUI.m_client.Send(1033, pktNew, null);
		}
	}

	public override void drawTooltips(string tip)

		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();

		{
			string[] array = tip.Replace("bonus", string.Empty).Split(',');
			ushort id = ushort.Parse(array[0]);
			int index = int.Parse(array[1]);
			Database.Campaign campaign = Global.character_.getCampaign(id);

		else if (tip != null)
		{
			tooltipUI.ShowTooltipContent(string.Empty, tip, 300, __BaseUI._fontSize);
		}
	}

	public void _makeCampaigns()
	{
		m_campaign = null;
		_ChapterCampaigns chapterCampaigns = (m_chapter = new _ChapterCampaigns());
		List<Database.Campaign> list = new List<Database.Campaign>();
		Hashtable templates = Global.getTemplates("campaign");
		_campaign campaign = null;
		foreach (DictionaryEntry item in templates)
		{
			_campaign campaign2 = item.Value as _campaign;
			if (Global.getTemplate("campaign_type", campaign2.idcampaign_type) is _campaign_type { chapter: not false } && campaign2.start)
			{
				campaign = campaign2;
				break;
			}
		}
		list.Add(Global.character_.getCampaign(campaign.id));
		for (int i = 0; i < list.Count; i++)
		{
			Database.Campaign campaign3 = list[i];
			foreach (ushort next_campaign in campaign3.m_next_campaigns)
			{
				Database.Campaign campaign4 = Global.character_.getCampaign(next_campaign);
				if (campaign4 != null && !list.Contains(campaign4))
				{
					list.Add(campaign4);
				}
			}
		}
		campaign_cur = -1;
		int num = 0;
		int num2 = 0;
		Database.Campaign campaign5 = null;
		foreach (Database.Campaign item2 in list)
		{
			if (item2.idcampaign_type != 2 || !item2.unlocked)
			{
				continue;
			}
			item2.tmp_first = item2.first;
			_AnecdoteCampaign anecdoteCampaign = new _AnecdoteCampaign();
			anecdoteCampaign.name = item2.name;
			anecdoteCampaign.campaign = item2;
			anecdoteCampaign.idcampaign = item2.id;
			anecdoteCampaign.campaign_elite = item2;
			chapterCampaigns.campaigns.Add(anecdoteCampaign);
			item2.real_unlocked = true;
			if (campaign5 != null && !campaign5.finished)
			{
				item2.real_unlocked = false;
			}
			else if (item2.finished)
			{
				chapterCampaigns.m_iFinishCount++;
			}
			if (!item2.finished && campaign_cur == -1)
			{
				campaign_cur = num2;
			}
			if (campaign_cur != -1)
			{
				num++;
				if (num >= 5)
				{
					break;
				}
			}
			campaign5 = item2;
			num2++;
			if (item2.m_next_campaigns.Count <= 0)
			{
				continue;
			}
			anecdoteCampaign.campaign_hero1 = Global.character_.getCampaign(item2.m_next_campaigns[0]);
			if (anecdoteCampaign.campaign_hero1 == null)
			{
				continue;
			}
			anecdoteCampaign.campaign_hero1.star_level = 1;
			if (anecdoteCampaign.campaign.finished)
			{
				anecdoteCampaign.campaign = anecdoteCampaign.campaign_hero1;
				anecdoteCampaign.campaign.unlocked = (anecdoteCampaign.campaign.real_unlocked = true);
			}
			if (anecdoteCampaign.campaign_hero1.m_next_campaigns.Count <= 0)
			{
				continue;
			}
			anecdoteCampaign.campaign_hero2 = Global.character_.getCampaign(anecdoteCampaign.campaign_hero1.m_next_campaigns[0]);
			if (anecdoteCampaign.campaign_hero2 == null)
			{
				continue;
			}
			anecdoteCampaign.campaign_hero2.star_level = 2;
			if (anecdoteCampaign.campaign.finished)
			{
				anecdoteCampaign.campaign = anecdoteCampaign.campaign_hero2;
				anecdoteCampaign.campaign.unlocked = (anecdoteCampaign.campaign.real_unlocked = true);
			}
			if (anecdoteCampaign.campaign_hero2.m_next_campaigns.Count <= 0)
			{
				continue;
			}
			anecdoteCampaign.campaign_hero3 = Global.character_.getCampaign(anecdoteCampaign.campaign_hero2.m_next_campaigns[0]);
			if (anecdoteCampaign.campaign_hero3 != null)
			{
				anecdoteCampaign.campaign_hero3.star_level = 3;
				if (anecdoteCampaign.campaign.finished)
				{
					anecdoteCampaign.campaign = anecdoteCampaign.campaign_hero3;
					anecdoteCampaign.campaign.unlocked = (anecdoteCampaign.campaign.real_unlocked = true);
					anecdoteCampaign.campaign.unlock_general = anecdoteCampaign.campaign_elite.unlock_general;
				}
			}
		}
		chapterCampaigns.m_iPage = (chapterCampaigns.campaigns.Count - 1) / 3;
	}

	public void SetCampaigns()
	{
		_makeCampaigns();
		cs_anecdote_records_0x40A();
		MyLoading();
	}

	private void cs_anecdote_records_0x40A()
	{
		CSAnecdoteRecords_0x40A cSAnecdoteRecords_0x40A = new CSAnecdoteRecords_0x40A();
		foreach (_AnecdoteCampaign campaign in m_chapter.campaigns)
		{
			byte b = campaign.campaign.m_troopsArray[campaign.campaign.m_troopsArray.Count - 1];
			cSAnecdoteRecords_0x40A.ids.Add((uint)((campaign.campaign.id << 16) | (b << 8)));
		}
		RootUI.m_client.Send(1034, cSAnecdoteRecords_0x40A, sc_anecdote_records_0x40A);
	}

	public static void sc_anecdote_records_0x40A(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCAnecdoteRecords_0x40A sCAnecdoteRecords_0x40A = JSProtoBuf.Serializer.Deserialize<SCAnecdoteRecords_0x40A>(stream);
		foreach (AnecdoteRecord record in sCAnecdoteRecords_0x40A.records)
		{
			if (Global.character_.campaigns_.ContainsKey((ushort)record.idcampaign))
			{
				Database.Campaign campaign = Global.character_.campaigns_[(ushort)record.idcampaign];
				string[] array = record.min_score.Split(',');
				if (array.Length > 0)
				{
					campaign.min_score_name = array[0];
				}
				array = record.min_time.Split(',');
				if (array.Length > 0)
				{
					campaign.min_time_name = array[0];
				}
			}
		}
	}

	private void makeChapters()
	{
		_ChapterCampaigns chapter = m_chapter;
		for (int num = chapter.campaigns.Count - 1; num >= 0; num--)
		{
			_AnecdoteCampaign anecdoteCampaign = chapter.campaigns[num];
			makeChapter(anecdoteCampaign.campaign);
		}
		if (!m_bInit)
		{
			scrollPos.y = campaign_cur * 130;
			if (campaign_cur > chapter.campaigns.Count - 3)
			{
				scrollPos.y = (chapter.campaigns.Count - 3) * 130;
			}
			if (scrollPos.y < 0f)
			{
				scrollPos.y = 0f;
			}
			ScrollControl.infos[anecdote_scrollid].SetStartPos(scrollPos);
		}
		if (asset_names.Count > 0)
		{
			Disable();
			CollectingStart(bCollecting: false);
			StartLoading();
		}
		else
		{
			Show();
		}
	}

	private void makeChapter(Database.Campaign c)
	{
		if (c == null)
		{
			return;
		}
		if (c.texIcon == null && c.m_troopsArray.Count > 0)
		{
			_campaign_troop campaign_troop = c.m_troops[c.m_troopsArray[c.m_troopsArray.Count - 1]] as _campaign_troop;
			AddAssetSub("face" + campaign_troop.icon);
		}
		foreach (_Treasure drop in c.drops)
		{
			AddAssetSub(drop.getIconName());
		}
	}

	private void makeCampaign(Database.Campaign c)
	{
		m_campaign = c;
		AddAsset("campaign/arrow_red");
		LoadCampaignChapter();
		setTroops();
		BaseAsset.lds.Visible_sub();
		DisableAll();
		CollectingStart(bCollecting: false);
		StartLoading();
	}
