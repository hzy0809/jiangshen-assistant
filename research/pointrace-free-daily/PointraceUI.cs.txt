	public void checkGameObject(GameObject obj)
	{
		int i;
		for (i = 0; i < mModelList.Count && !(mModelList[i] == obj); i++)
		{
		}
		int num = 0;
		Cooldown cooldown = Global.character_.getCooldown(6);
		foreach (Database.PointraceChallenger challenger in m_pointrace.challengers)
		{
			if (!challenger.killed && num == i)
			{
				m_challengeIdx = num;
				DisableAll();
				if (m_pointrace.challenge_count >= 10 + m_pointrace.challenge_buy)
				{
					if (m_pointrace.challenge_buy >= 10)
					{
						__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20010u), BaseAsset.lds.GetCurUI(), EnableAll);
					}
					else
					{
						_buy_template buyTemplate = Global.getBuyTemplate(_buy_template.TYPE.POINTRACE, m_pointrace.challenge_buy + 1);
						int gold = buyTemplate.gold;
						__BaseUI.GetConsumeUI().setConsume(gold, 80, ChallengeBuyOK);
					}
				}
				else
				{
					__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(21373u, challenger.name), BaseAsset.lds.GetCurUI(), ChallengeOK, _bCancel: true);
				}
			}
			num++;
		}
	}

	private void ChallengeOK(bool bOK, object[] objs)
	{
		if (bOK)
		{
			Cooldown cooldown = Global.character_.getCooldown(6);
			if (cooldown != null && cooldown.time > TimeUtil.Now)
			{
				m_bWantChallenge = true;
				WantClearCD(bOK: true, objs);
			}
			else
			{
				WantChallenge();
			}
		}
		else
		{
			EnableAll();
		}
	}

	{
		for (int i = 0; i < 5; i++)

			reference = Camera.main.WorldToScreenPoint(position);

		}
	}

	{
		if (m_pointrace.challenge_count >= 10 + m_pointrace.challenge_buy)

			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20010u), BaseAsset.lds.GetCurUI(), EnableAll);

	public void resetModel()
	{

			UnityEngine.Object.Destroy(mModel);
		}

		{
			assetBundle = ((!challenger.sex) ? BaseAsset.was.GetAsset("char_model/heroine1_walk") : BaseAsset.was.GetAsset("char_model/hero1_walk"));
			if (assetBundle != null)
			{
				GameObject gameObject = UnityEngine.Object.Instantiate(assetBundle.mainAsset) as GameObject;
