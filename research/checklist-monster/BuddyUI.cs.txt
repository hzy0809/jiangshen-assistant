	private void Update()
	{
		if (moveynz)
		{
			if (update_x > 125f && update_x2 < 125f)
			{
				update_x -= update_speed;
				update_x2 += update_speed;
				if (update_x < 125f)
				{
					update_x = 125f;
				}
				if (update_x2 > 125f)
				{
					update_x2 = 125f;
				}
			}
			else
			{
				moveynz = false;
				zyyn = false;
				hideyn = false;
			}
		}
		if (!moveyny)
		{
			return;
		}
		zyyn = true;
		hideyn = true;
		if (update_x < 250f && update_x2 > 0f)
		{
			update_x += update_speed;
			update_x2 -= update_speed;
			if (update_x > 250f)
			{
				update_x = 250f;
			}
			if (update_x2 < 0f)
			{
				update_x2 = 0f;
			}
		}
		else
		{
			moveyny = false;
		}
	}

	protected override void Web_OnGUI()
	{
		if (!BeginGUI(527f, 441f, __BaseUI.UIDepth_Sub))
		{
			return;
		}
		if (m_bShowXianhua || m_detailBuddy != null || subreaddwindow)
		{
			Disable();
		}
		else
		{
			Enable();
		}
		if (m_showAddBuddy != 0 || m_detailBuddy != null || subreaddwindow)
		{
			Disable();
		}
		else

		}
		float x = update_x;
		if (hideyn)
		{
			__BaseUI.BeginGroup(__BaseUI._rect(x, 0f, 266f, 441f));

	private void Drawxianhua()
	{
		int num = Math.Max(6, Global.character_.FlowerEvents.Count + 1) * 55;
		if (num > 340)
		{
			scrollFlowers = ScrollControl.infos[scroll_index_flowers].scrollpos;
			scrollFlowers = GUIHacks.BeginScrollView(__BaseUI._rect(17f, 63f, 221f, 340f), scrollFlowers, __BaseUI._rect(0f, 0f, 221f, num), useHorizontal: false, useVertical: true, new GUIStyle(), __BaseUI._vertScroll1[0], 55, 0, scroll_index_flowers);
			ScrollControl.infos[scroll_index_flowers].scrollpos = scrollFlowers;
		}
		else
		{
			__BaseUI.BeginGroup(__BaseUI._rect(17f, 63f, 221f, 340f));
		}
		if (Global.character_.FlowerEvents.Count > 0)
		{
			__BaseUI.Label(__BaseUI._rect(xianhuapos.x, xianhuapos.y, 214f, 44f), string.Empty, bar4);
			for (int i = 0; i < Global.character_.FlowerEvents.Count; i++)
			{
				EventTimeline eventTimeline = Global.character_.FlowerEvents[i];
				if (eventTimeline.eventid == 4)
				{
					Drawxianhua(eventTimeline, i);
				}
			}
		}
		if (num > 340)
		{
			GUIHacks.EndScrollView();
		}
		else
		{

	protected override void AssetLoaded(string AssetName)
	{
		if ("face" + Global.character_.icon == AssetName)

			Global.character_.texIcon2 = BaseAsset.LoadTexture("face" + Global.character_.icon);
		}
		for (int i = 0; i < Buddys.Count; i++)
		{
			if (AssetName == "face" + Buddys[i].icon)
			{
				Buddys[i].faceicon = BaseAsset.LoadTexture(AssetName);
			}
		}
		for (int j = 0; j < Fans.Count; j++)
		{
			if (AssetName == "face" + Fans[j].icon)
			{
				Fans[j].faceicon = BaseAsset.LoadTexture(AssetName);
			}
		}
		for (int k = 0; k < Blacklist.Count; k++)
		{
			if (AssetName == "face" + Blacklist[k].icon)
			{
				Blacklist[k].faceicon = BaseAsset.LoadTexture(AssetName);
			}
		}
		for (int l = 0; l < Contacts.Count; l++)
		{
			if (AssetName == "face" + Contacts[l].icon)
			{
				Contacts[l].faceicon = BaseAsset.LoadTexture(AssetName);
			}
		}
	}
