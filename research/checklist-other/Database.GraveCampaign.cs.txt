using game.net.data.StoC;

namespace Database;

public class GraveCampaign
{
	public _character_grave_troop data_ = new _character_grave_troop();

	public GraveTroop[] troops = new GraveTroop[6];

	public _campaign m_campaign;

	public ushort idgrave;

	public byte finish;

	public int soldier;

	public bool showboss;

	public GraveCampaign(SCGraveLoadCampaign_0x341 d)
	{
		finish = (byte)d.finish;
		for (int i = 0; i < 6; i++)
		{
			troops[i] = new GraveTroop();
		}
		data_.idtroop1 = (ushort)d.idtroop1;
		data_.idtroop2 = (ushort)d.idtroop2;
		data_.idtroop3 = (ushort)d.idtroop3;
		data_.idtroop4 = (ushort)d.idtroop4;
		data_.idtroop5 = (ushort)d.idtroop5;
		data_.idboss = (ushort)d.idboss;
		m_campaign = Global.getTemplate("campaign", d.idcampaign) as _campaign;
		soldier = d.soldier;
		showboss = d.showboss;
		idgrave = (ushort)d.idgrave;
		troops[0].Set(m_campaign.m_troops[m_campaign.m_troopsArray[0]] as _campaign_troop);
		troops[1].Set(m_campaign.m_troops[m_campaign.m_troopsArray[1]] as _campaign_troop);
		troops[2].Set(m_campaign.m_troops[m_campaign.m_troopsArray[2]] as _campaign_troop);
		troops[3].Set(m_campaign.m_troops[m_campaign.m_troopsArray[3]] as _campaign_troop);
		troops[4].Set(m_campaign.m_troops[m_campaign.m_troopsArray[4]] as _campaign_troop);
		foreach (object value in m_campaign.m_troops.Values)
		{
			_campaign_troop campaign_troop = (_campaign_troop)value;
			if (campaign_troop.rank == 2)
			{
				troops[5].Set(campaign_troop);
				break;
			}
		}
		troops[0].idtroop = 1;
		troops[1].idtroop = 2;
		troops[2].idtroop = 3;
		troops[3].idtroop = 4;
		troops[4].idtroop = 5;
		troops[5].idtroop = 6;
		for (int j = 0; j < 6; j++)
		{
			if (finish > j)
			{
				troops[j].killed = true;
			}
			else
			{
				troops[j].killed = false;
			}
			troops[j].genPos();
			troops[j].setArmy(d);
		}
		Global.character_.grave.times = (short)d.times;
	}

	public GraveCampaign(SCGraveLoad_0x343 d)
	{
		finish = (byte)d.finish;
		for (int i = 0; i < 6; i++)
		{
			troops[i] = new GraveTroop();
		}
		data_.idtroop1 = (ushort)d.idtroop1;
		data_.idtroop2 = (ushort)d.idtroop2;
		data_.idtroop3 = (ushort)d.idtroop3;
		data_.idtroop4 = (ushort)d.idtroop4;
		data_.idtroop5 = (ushort)d.idtroop5;
		data_.idboss = (ushort)d.idboss;
		m_campaign = Global.getTemplate("campaign", d.idcampaign) as _campaign;
		soldier = d.soldier;
		showboss = d.showboss;
		idgrave = (ushort)d.idgrave;
		troops[0].idtroop = 1;
		troops[1].idtroop = 2;
		troops[2].idtroop = 3;
		troops[3].idtroop = 4;
		troops[4].idtroop = 5;
		troops[5].idtroop = 6;
	}
}
