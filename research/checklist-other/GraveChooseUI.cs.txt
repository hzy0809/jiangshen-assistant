	private void checkGrave()
	{
		curPageGraves.Clear();
		_grave grave = Global.getTemplate("grave", 1u) as _grave;
		if (grave == null)
		{
			return;
		}
		curPageGraves.Add(grave);
		while (grave.nextid != 0)
		{
			grave = Global.getTemplate("grave", grave.nextid) as _grave;
			if (grave == null)
			{
				break;
			}
			curPageGraves.Add(grave);
		}
	}

	private bool checkGraveBoss()
	{
		if (Global.character_.grave_campaign.showboss && Global.character_.grave_campaign.finish == 5 && Global.character_.grave_campaign.soldier > 0 && Global.character_.grave.reseted != 1)

	private bool checkGraveFinish()
	{
		if (Global.character_.grave.reseted == 1)
		{
			return true;
		}
		if (Global.character_.grave_campaign.soldier <= 0)
		{
			return true;
		}
		if (!Global.character_.grave_campaign.showboss && Global.character_.grave_campaign.finish == 5)
		{
			return true;
		}
		if (Global.character_.grave_campaign.showboss && Global.character_.grave_campaign.finish == 6)
		{
			return true;
		}
		return false;
	}
