	private void cs_buysilver()
	{
		CSBuySilver_0x177 pkt = new CSBuySilver_0x177();
		Global.m_client.Send(375, pkt, sc_buysilver);
	}

	private void sc_buysilver(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			Enable();
			return;
		}
		SCBuySilver_0x177 sCBuySilver_0x = JSProtoBuf.Serializer.Deserialize<SCBuySilver_0x177>(stream);
		if (sCBuySilver_0x.state)
		{
			Global.character_.sliver_pick++;
			Global.character_.gold -= sCBuySilver_0x.cost;
			__BaseUI.drawStateBubble("gold-" + sCBuySilver_0x.cost);
			Global.character_.buy_silverpick++;
			Enable();
			ToDig(digging_type.sliver);
		}
		else
		{
			Enable();
		}
	}

	private void ToDig(digging_type type)
	{
		if (type == digging_type.sliver)
		{
			digTimer = DateTime.Now.AddSeconds(0.5);
			timecd = 0.5f;
		}
		else
		{
			digTimer = DateTime.Now.AddSeconds(1.0);
			timecd = 1f;
		}
		m_digging = type;
	}

		GUIStyle gUIStyle = new GUIStyle();
		gUIStyle.font = __BaseUI.ArialFont;
		gUIStyle.normal.textColor = Color.white;
		gUIStyle.alignment = TextAnchor.MiddleCenter;
		gUIStyle.fontStyle = FontStyle.Bold;
		_vip_template vip_template = Global.getTemplate("vip_template", Global.character_.VIPLEVEL) as _vip_template;
		string text = GUI.TextField(__BaseUI._rect(327f, 112f, 27f, 27f), m_iTimes.ToString(), gUIStyle);
		int num = vip_template.goldenchantharvest_times - Global.character_.gold_pick_times;
		if (text == string.Empty)
		{
			m_iTimes = 1;
		}
		if (int.TryParse(text, out var result))
		{
			if (result <= num && result > 0)
			{
				m_iTimes = result;
			}
			else if (result > num)
			{
				m_iTimes = num;
			}
		}
		if (__BaseUI._btnShang.Button(result < num, __BaseUI._rect(360f, 116f, 19f, 9f), string.Empty))
		{
			m_iTimes++;
		}
		if (__BaseUI._btnXia.Button(m_iTimes > 1, __BaseUI._rect(360f, 126f, 19f, 9f), string.Empty))
		{
			m_iTimes--;
		}
		uint goldCost = GetGoldCost(m_iTimes);
		__BaseUI.Label(__BaseUI._rect(371f, 156f, 47f, 17f), goldCost.ToString(), __BaseUI._numLeft, __BaseUI._clrWhite);
		if (button_piliang3.Button(m_iTimes > 0 && CheckGoldTimes(m_iTimes), __BaseUI._rect(196f, 213f, 110f, 40f), string.Empty))
		{
			__BaseUI.GetConsumeUI().setConsume((int)goldCost, 35, OnGoldPickLot);
		}
		if (__BaseUI._btnClose.Button(bBtn: true, __BaseUI._rect(364f, 9f, 150f, 41f), string.Empty))
		{
			vipdig = false;
		}
		GUI.EndGroup();
	}
