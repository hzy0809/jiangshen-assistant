	private void sc_dummy_challenge_info(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCDummyChallengeInfo_0x264 sCDummyChallengeInfo_0x = JSProtoBuf.Serializer.Deserialize<SCDummyChallengeInfo_0x264>(stream);
			if (sCDummyChallengeInfo_0x != null)
			{

					StartLoading();
					return;
				}
			}
		}
		BaseAsset.lds.Hide();
		EnableAll();
	}

	private void sc_dummy_challenge_fight(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCDummyChallengeFight_0x265 sCDummyChallengeFight_0x = (OK_fight = JSProtoBuf.Serializer.Deserialize<SCDummyChallengeFight_0x265>(stream));
			if (sCDummyChallengeFight_0x != null)
			{
				if (sCDummyChallengeFight_0x.report != null && sCDummyChallengeFight_0x.report.Count > 0 && m_battleUI != null)
				{
					_report_list = sCDummyChallengeFight_0x.report;
					_turn_index = sCDummyChallengeFight_0x.start_turn_index;
					current_report_index = 0;
					current_turn_index = 0;
					display_turn_index = 0;
					start_total_troop_num = 0;
					m_battleUI.setPrevUI(BaseAsset.lds.GetCurUI());
					m_battleUI.SetTopMode(BattleUI.TopUIMode.Woodwar);
					m_battleUI.GetComponent<BattleWoodwarTopUI>().ResetKillCount();
					NextFight();
					Hide();
					return;
				}
				if (sCDummyChallengeFight_0x.reward_status != null && sCDummyChallengeFight_0x.reward_status.player.can_take)
				{
					Global.character_.dummy_challenge_status = sCDummyChallengeFight_0x.reward_status;
					PS_state = PlayState.ok;
				}
			}
		}
		BaseAsset.lds.Hide();
		EnableAll();
	}

	private void NextFight()
	{
		if (current_report_index < _report_list.Count)
		{
			current_turn_index = _turn_index[current_report_index];
			is_init_turn = true;
			if (current_report_index < _report_list.Count - 1)
			{
				m_battleUI.setReport(_report_list[current_report_index], wait_for_end: true, use_battle_end: false, use_battle_surrender: false, null, isopponent: false);
			}
			else
			{
				m_battleUI.setReport(_report_list[current_report_index], wait_for_end: true, use_battle_end: true, use_battle_surrender: false, null, isopponent: false);
			}
			if (current_report_index == 0)
			{
				start_total_troop_num = m_battleUI.getArmyTotalNum(0);
			}
			m_battleUI.SetTopMode(BattleUI.TopUIMode.Woodwar);
			m_battleUI.SetNewTurnCallback(_on_new_turn);
			m_battleUI.SetHandleCallback(BattleReport.GameTurn.Action.ActionType.dead, m_battleUI.GetComponent<BattleWoodwarTopUI>().update_enemy_kill_count);
			m_battleUI.ViewType = enumBattleViewType.viewDefault;
			if (current_report_index < _report_list.Count - 1)
			{
				m_battleUI.SetResultFunc(NextFight);
			}
			else
			{
				m_battleUI.GetComponent<BattleWoodwarResultUI>().total_lose_num = start_total_troop_num - (m_battleUI.getArmyTotalNum(0) - m_battleUI.getArmyLoseNum(0));
				m_battleUI.setClose(_woodwar_battle_close);
				m_battleUI.GetComponent<BattleWoodwarResultUI>().SetClose(m_battleUI.battle_exit);
				m_battleUI.SetResultFunc(m_battleUI.GetComponent<BattleWoodwarResultUI>().PrepareOpen);
			}
			current_report_index++;
			BaseUI.GetRootUI().battle_enter(fslds: false);
		}
	}

	public override void drawTooltips(string tip)
	{
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (tip == "help")
		{
			tooltipUI.ShowTooltipCommonH();
		}
		else if (tip.StartsWith("xiangxi:"))
		{
			string s = tip.Substring("xiangxi:".Length);
			int index = int.Parse(s);
			List<CharacterDummyChallenge> first_ = Global.character_.dummy_challenge_status.first_10;
			string text = string.Format(__BaseUI.load_str(20546u), (int)first_[index].level, (int)calcChallengeUILevel(first_[index].challenge_level), (int)first_[index].challenge_level, (int)first_[index].enemy_count, (int)first_[index].total_damage);
			tooltipUI.ShowTooltipContent(null, first_[index].name + text, 300, __BaseUI._fontSize);
		}
		else if (tip == "baoxiang")
		{
			CharacterDummyChallenge player = Global.character_.dummy_challenge_status.player;
			int num = 0;
			if (player.bonuses.Count >= 2)
			{
				num += player.bonuses[0].param1 + player.bonuses[1].param1;
			}
			if (player.bonuses.Count >= 4)
			{
				num += player.bonuses[2].param1;
			}
			string text2 = string.Empty;
			if (player.bonuses.Count >= 2)
			{
				text2 += string.Format(__BaseUI.load_str(20582u), (int)calcChallengeUILevel(player.challenge_level), player.bonuses[0].param1);
				text2 += string.Format(__BaseUI.load_str(20583u), (int)player.wave_count, player.bonuses[1].param1);
			}
			if (player.bonuses.Count >= 4)
			{
				text2 += string.Format(__BaseUI.load_str(20581u), (int)player.rank, player.bonuses[2].param1, player.bonuses[3].param1);
				text2 += string.Format(__BaseUI.load_str(20584u), num, player.bonuses[3].param1);
			}
			else
			{
				text2 += string.Format(__BaseUI.load_str(20585u), num);
			}
			tooltipUI.ShowTooltipContent(null, text2, 300, __BaseUI._fontSize);
		}
		else if (tip.StartsWith("common:"))
		{
			tooltipUI.ShowTooltipCommon(tip.Replace("common:", string.Empty), "woodwar");
		}
	}

	protected override void Web_OnGUI()
	{
		if (!BeginGUI(719f, 404f, __BaseUI.UIDepth_Sub, __BaseUI._bgStyle, textitle))
		{
			return;
		}
		uint t = 0u;
		if (temptime > TimeUtil.Now)
		{
			t = (uint)(temptime - TimeUtil.Now);
		}
		if (Global.character_.dummy_challenge_status.player.rank == 0)
		{
			PS_state = PlayState.init;
		}
		else
		{
			PS_state = PlayState.no;
		}
		if (Global.character_.dummy_challenge_status.player.can_take)
		{
			PS_state = PlayState.ok;
		}
		__BaseUI.Label(__BaseUI._rect(33f, 60f, 659f, 189f), string.Empty, bgframe);
		__BaseUI.Label(__BaseUI._rect(255f, 24f, 207f, 15f), string.Empty, zi9);
		switch (PS_state)
		{
		case PlayState.init:
			DrawInit();
			break;
		case PlayState.no:
			DrawNO();
			__BaseUI.Label(__BaseUI._rect(612f, 300f, 100f, 20f), Global.getFormatTime((int)t), __BaseUI._numLeft, __BaseUI._clrRed);
			break;
		case PlayState.ok:
			DrawOK();
			break;
		}
		Drawpaihang();
		if (PS_state == PlayState.no || PS_state == PlayState.init)
		{
			if (btn_start.Button(IsEnable(), __BaseUI._rect(200f, 345f, 146f, 36f), string.Empty))
			{
				BattleFormation();
			}
			bool flag = Global.character_.dummy_challenge_status.player.fast_challenge_left_times != 0;
			string empty = string.Empty;
			empty = ((!flag) ? ("common:" + __BaseUI.get_string(20636u)) : ("common:" + __BaseUI.get_string(20643u)));
			if (btn_kuaisu.Button(IsEnable() && flag, __BaseUI._rect(370f, 345f, 146f, 40f), new GUIContent(string.Empty, empty)))
			{
				cs_dummy_challenge_fast_complete();
			}
		}
		else if (btn_lingqu2.Button(IsEnable(), __BaseUI._rect(285f, 345f, 146f, 36f), new GUIContent(string.Empty, "baoxiang")))
		{
			cs_dummy_challenge_get_reward();
		}
		btn_help.Button(IsEnable(), __BaseUI._rect(53f, 188f, 38f, 57f), new GUIContent(string.Empty, "help"));
		if (EndGUI())
		{
			MainUI.SetSubUI(null);
		}
	}

	private void Drawpaihang()
	{
		int num = Math.Max(5, Global.character_.dummy_challenge_status.first_10.Count + 1) * 20;
		if (num > 105)
		{
			scrollpaihang = ScrollControl.infos[scroll_index_paihang].scrollpos;
			scrollpaihang = GUIHacks.BeginScrollView(__BaseUI._rect(302f, 110f, 345f, 105f), scrollpaihang, __BaseUI._rect(0f, 0f, 348f, num), useHorizontal: false, useVertical: true, new GUIStyle(), __BaseUI._vertScroll1[0], 20, 0, scroll_index_paihang);
			ScrollControl.infos[scroll_index_paihang].scrollpos = scrollpaihang;
		}
		else
		{
			__BaseUI.BeginGroup(__BaseUI._rect(302f, 110f, 345f, 105f));
		}
		List<CharacterDummyChallenge> first_ = Global.character_.dummy_challenge_status.first_10;
		if (first_.Count > 0)
		{
			for (int i = 0; i < first_.Count; i++)
			{
				Drawpaihang(first_[i], (int)(first_[i].rank - 1));
			}
		}
		if (num > 105)
		{
			GUIHacks.EndScrollView();
		}
		else
		{
			__BaseUI.EndGroup();
		}
	}

	private void sc_dummy_challenge_get_reward(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCDummyChallengeGetReward_0x266 sCDummyChallengeGetReward_0x = JSProtoBuf.Serializer.Deserialize<SCDummyChallengeGetReward_0x266>(stream);
			if (sCDummyChallengeGetReward_0x != null && sCDummyChallengeGetReward_0x.bonuses.Count > 0)
			{
				List<_Treasure> list = new List<_Treasure>();
				for (int i = 0; i < sCDummyChallengeGetReward_0x.bonuses.Count; i++)
				{
					_Treasure treasure = new _Treasure();
					treasure.make(sCDummyChallengeGetReward_0x.bonuses[i]);
					list.Add(treasure);
				}
				__BaseUI.GetBattleResultUI().Set(null, string.Empty, string.Empty, 0, 0, list, closeRewardResult);
				__BaseUI.GetBattleResultUI().MyLoading();
			}
		}
		else
		{
			closeRewardResult();
		}
	}

	private void cs_dummy_challenge_fast_complete()
	{
		Disable();
		CSDummyChallengeFastComplete_0x268 pkt = new CSDummyChallengeFastComplete_0x268();
		RootUI.m_client.Send(616, pkt, sc_dummy_challenge_fast_complete);
	}

	private void sc_dummy_challenge_fast_complete(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCDummyChallengeFastComplete_0x268 sCDummyChallengeFastComplete_0x = JSProtoBuf.Serializer.Deserialize<SCDummyChallengeFastComplete_0x268>(stream);
			if (sCDummyChallengeFastComplete_0x == null)
			{
				return;
			}
			if (sCDummyChallengeFastComplete_0x.state == SCDummyChallengeFastComplete_0x268.State.FAILED)
			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.get_string(20641u), rmg);
				return;
			}
			Global.character_.dummy_challenge_status = sCDummyChallengeFastComplete_0x.status;
			if (sCDummyChallengeFastComplete_0x.state == SCDummyChallengeFastComplete_0x268.State.REWARD)
			{
				PS_state = PlayState.ok;
				Enable();
				return;
			}
			BattleWoodwarResultUI component = m_battleUI.GetComponent<BattleWoodwarResultUI>();
			component.resultStatus = sCDummyChallengeFastComplete_0x.result;
			component.resultChallengeLevelFinished = sCDummyChallengeFastComplete_0x.level_finished;
			component.used_all_turns = sCDummyChallengeFastComplete_0x.used_all_turns;
			component.total_lose_num = sCDummyChallengeFastComplete_0x.lose_num;
			component.SetClose(_fast_finish_close);
			component.PrepareOpen();
			Enable();
		}
		else
		{
			Enable();
		}
	}

	private void closeRewardResult()
	{
		cs_dummy_challenge_info();
	}

	public static uint calcChallengeUILevel(uint clevel)
	{
		Hashtable templates = Global.getTemplates("dummy_challenge_setting");
		IDictionaryEnumerator dictionaryEnumerator = templates.GetEnumerator();
		try
		{
			if (dictionaryEnumerator.MoveNext())
			{
				_dummy_challenge_setting dummy_challenge_setting = (_dummy_challenge_setting)((DictionaryEntry)dictionaryEnumerator.Current).Value;
				if (clevel < dummy_challenge_setting.start_level)
				{
					return 0u;
				}
				return (clevel - dummy_challenge_setting.start_level) / dummy_challenge_setting.level_increase + 1;
			}
		}
		finally
		{
			IDisposable disposable = dictionaryEnumerator as IDisposable;
			if (disposable != null)
			{
				disposable.Dispose();
			}
		}
		return 0u;
	}

	private void _woodwar_battle_close()
	{
		m_battleUI.SetTopMode(BattleUI.TopUIMode.Normal);
		BattleWoodwarResultUI component = m_battleUI.GetComponent<BattleWoodwarResultUI>();
		if (component.resultChallengeLevelFinished)
		{
			if (component.resultStatus.challenge_level == component.resultStatus.next_challenge_level)
			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.get_string("no_dummy_troop"), rmg, reshow_window);
			}
			else
			{
				__BaseUI.GetMessageBox().setMessage(picmode: true, zi10, new Rect(28f, 55f, 299f, 64f), null, new Rect(0f, 0f, 0f, 0f), null, rmg, reshow_window);
			}
		}
		else
		{
			reshow_window();
		}
	}

	private void _fast_finish_close()
	{
		BattleWoodwarResultUI component = m_battleUI.GetComponent<BattleWoodwarResultUI>();
		if (component.resultChallengeLevelFinished)
		{
			if (component.resultStatus.challenge_level == component.resultStatus.next_challenge_level)
			{
				__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.get_string("no_dummy_troop"), null);
			}
			else
			{
				__BaseUI.GetMessageBox().setMessage(picmode: true, zi10, new Rect(28f, 55f, 299f, 64f), null, new Rect(0f, 0f, 0f, 0f), null, rmg, null);
			}
		}
		else
		{
			Enable();
		}
	}
