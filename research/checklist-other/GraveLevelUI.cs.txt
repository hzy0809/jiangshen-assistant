			for (int i = 0; i < 6; i++)

				asset = BaseAsset.was.GetAsset("battle/" + text);

				{

				BaseAsset.SetLayerRecursively(graveTroop.mesh, 9);

					graveTroop.mesh.animation["dead"].time = graveTroop.mesh.animation["dead"].length;
					graveTroop.mesh.animation["dead"].enabled = true;
					graveTroop.mesh.animation.Sample();
				}

					if (Global.character_.grave_campaign.showboss)
					{
						graveTroop.mesh.SetActiveRecursively(state: true);

		}
		for (int j = 0; j < 6; j++)
		{
			GraveTroop graveTroop2 = Global.character_.grave_campaign.troops[j];
			if (graveTroop2.mesh != null && j < Global.character_.grave_campaign.finish && graveTroop2.mesh.animation["dead"] != null)

	public void makeArrow(Database.GodTroop t, int layer)
	{
		t.arrow_mesh = (GameObject)Object.Instantiate(BaseAsset.was.GetAsset("campaign/arrow_red").mainAsset);
		if (!(t.arrow_mesh != null))
		{
			return;
		}
		float y = 18f;
		if (t.mesh != null)
		{
			BoxCollider boxCollider = t.mesh.collider as BoxCollider;
			if (boxCollider != null)
			{
				y = boxCollider.extents.y * 2f * 2f + 0.5f;
			}
		}
		t.arrow_mesh.transform.position = t.pos + new Vector3(0f, y, 0f);
		float num = 3f;
		t.arrow_mesh.transform.localScale = new Vector3(num, num, num);
		t.arrow_mesh.transform.parent = mesh.transform;
		BaseAsset.SetLayerRecursively(t.arrow_mesh, layer);
	}

	public override void Destroy()
	{
		base.Destroy();
		if (mesh != null)
		{
			Object.Destroy(mesh);
			mesh = null;
		}
		stone_mesh = null;
		if (Global.character_.grave_campaign == null || Global.character_.grave_campaign.troops == null)
		{
			return;
		}
		for (int i = 0; i < Global.character_.grave_campaign.troops.Length; i++)
		{
			GraveTroop graveTroop = Global.character_.grave_campaign.troops[i];
			if (graveTroop != null)
			{
				graveTroop.mesh = null;
				graveTroop.arrow_mesh = null;
			}
		}
	}

	private void _return()
	{
		RootMessenger oldMainUI = BaseAsset.lds.GetOldMainUI();
		BaseAsset.lds.SetOldUI(rmg);
		if (oldMainUI == null || oldMainUI.LevelID == "Login")
		{
			GameObject.Find("/Root/RootUI").GetComponent<RootUI>().LoadLevel("Campaign", rmg);
			return;
		}
		GameObject.Find("/Root/RootUI").GetComponent<RootUI>().LoadLevel(oldMainUI.LevelID, rmg);
		RootBase.add_UIFunc(MainUI._grave);
	}

	public void wantCombat(GraveTroop troop)
	{
		m_troop = troop;
		if (troop.idtroop == Global.character_.grave_campaign.finish + 1)
		{
			Hashtable hashtable = new Hashtable();
			__BaseUI.subui = "NPCBattle";
			hashtable.Add("subui", __BaseUI.subui);
			hashtable.Add("campaignid", troop.idtroop);
			hashtable.Add("troopid", 0);
			hashtable.Add("substate", "grave_troop");
			hashtable.Add("viewonly", false);
			hashtable.Add("OnAttack", new FormationUI.delegate_OnAttack(OnAttack));
			SendMessage("SubLoading", hashtable, SendMessageOptions.DontRequireReceiver);
			FormationUI component = GetComponent<FormationUI>();
			MainUI.SetSubUI(component);
		}
	}

	private void cs_grave_surrender_0x34D()

	private void sc_grave_surrender_0x34D(MemoryStream stream, object arg)
	{
		if (stream != null)
		{
			SCGraveSurrender_0x34D sCGraveSurrender_0x34D = JSProtoBuf.Serializer.Deserialize<SCGraveSurrender_0x34D>(stream);
			if (sCGraveSurrender_0x34D.operate == SCGraveSurrender_0x34D.Operate.OK)
			{
				Global.character_.grave.reseted = 1;
				_return();
			}
		}
	}

	private void cs_grave_battle_0x342()
	{
		BaseAsset.lds.Visible_sub();
		DisableAll();
		CSGraveBattle_0x342 cSGraveBattle_0x = new CSGraveBattle_0x342();
		cSGraveBattle_0x.idtroop = m_troop.idtroop;
		RootUI.m_client.Send(834, cSGraveBattle_0x, sc_grave_battle_0x342);
	}

	private void sc_grave_battle_0x342(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			BaseAsset.lds.Hide();
			EnableAll();
			return;
		}
		SCGraveBattle_0x342 sCGraveBattle_0x = JSProtoBuf.Serializer.Deserialize<SCGraveBattle_0x342>(stream);
		BattleUI battleUI = BaseUI.GetBattleUI();
		battleUI.setPrevUI(rmg);
		battleUI.setReport(sCGraveBattle_0x.report, wait_for_end: false, use_battle_end: true, this, isopponent: false);
		battleUI.ViewType = enumBattleViewType.viewDefault;
		if (sCGraveBattle_0x.iswin)
		{
			m_ts = sCGraveBattle_0x.bonuses;
		}
		else
		{
			tempsoldier = sCGraveBattle_0x.soldiers;
		}
		tempendgrave = sCGraveBattle_0x.endgrave;
		tempsoldiernum = sCGraveBattle_0x.soldier;
		tempshowboss = sCGraveBattle_0x.showboss;
		BaseUI.GetRootUI().battle_enter(fslds: false);
		BaseAsset.lds.SetBattling(bBattling: true);
		MainUI.SetSubUI(null);
	}

	public void BattleEnd(SCBattleEnd_0x408 pkt)
	{
		idnewgrave = 0;
		Global.character_.grave_campaign.soldier -= tempsoldiernum;
		if (pkt.iswin)
		{
			m_troop.killed = true;
			List<_Treasure> list = new List<_Treasure>();
			_Treasure treasure = new _Treasure();
			if (m_ts.bonuses != null)
			{
				foreach (Treasure bonuse in m_ts.bonuses)
				{
					if (bonuse.idtreasure_type == 3)
					{
						for (int i = 0; i < bonuse.param1; i++)
						{
							treasure = new _Treasure();
							treasure.make((byte)bonuse.idtreasure_type, bonuse.param0, 1);
							list.Add(treasure);
						}
					}
					else
					{
						treasure = new _Treasure();
						treasure.make((byte)bonuse.idtreasure_type, bonuse.param0, bonuse.param1);
						list.Add(treasure);
					}
				}
			}
			Global.character_.grave_campaign.finish++;
			BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
			if (tempshowboss)
			{
				Global.character_.grave_campaign.showboss = true;
				Global.character_.grave_campaign.troops[5].mesh.transform.parent = mesh.transform;
				if (Global.character_.grave_campaign.finish == 5)
				{
					isOver = true;
					Global.character_.grave.finishtoday = 1;
				}
			}
			else if (Global.character_.grave_campaign.finish == 5)
			{
				isOver = true;
				Global.character_.grave.finishtoday = 1;
			}
			if (Global.character_.grave_campaign.finish == 6)
			{
				if (Global.character_.graves.ContainsKey(Global.character_.grave_campaign.idgrave))
				{
					Global.character_.graves[Global.character_.grave_campaign.idgrave].isfinish = true;
				}
				if (Global.getTemplate("grave", Global.character_.grave_campaign.idgrave) is _grave grave)
				{
					Global.character_.shop_open = (ushort)grave.shop;
					if (Global.getTemplate("grave", grave.nextid) is _grave && !Global.character_.graves.ContainsKey(grave.nextid))
					{
						_character_grave character_grave = new _character_grave();
						character_grave.idgrave = grave.nextid;
						character_grave.isopen = true;
						character_grave.isfinish = false;
						Global.character_.graves.Add(grave.nextid, character_grave);
						Global.character_.grave.newgrave = character_grave.idgrave;
						idnewgrave = character_grave.idgrave;
					}
				}
				isOver = true;
				Global.character_.grave.finishtoday = 1;
			}
			if (Global.character_.grave_campaign.finish == 5)
			{
			}
			BattleUI battleUI = BaseUI.GetBattleUI();
			battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, BonusClose);
			if (list.Count > 0)
			{
				battleUI.setClose(BonusOpen);
			}
			else
			{
				battleUI.setClose(BonusClose);
			}
			return;
		}
		for (int j = 0; j < tempsoldier.Count; j++)
		{
			foreach (_character_grave_army item in m_troop.armys_)
			{
				if (tempsoldier[j].location == item.location)
				{
					item.soldier = tempsoldier[j].soldier;
					break;
				}
			}
		}
		FailedOpen();
	}

	private void BonusOpen()
	{
		BaseAsset.lds.GetCurUI().ToDisable();

	}

	private void BonusClose()
	{
		if (tempshowboss)
		{
			DisableAll();
			__BaseUI.GetMessageBox().setMessage(picmode: true, ui_zi_jingxing, new Rect(64f, 19f, 235f, 32f), ui_icon_shouhushen, new Rect(94f, 45f, 152f, 122f), null, rmg, Endboss);
		}
		else if (isOver)
		{
			if (idnewgrave > 0)
			{
				if (Global.getTemplate("grave", idnewgrave) is _grave grave && GraveChooseUI.texDicName.ContainsKey(grave.title))
				{
					__BaseUI.GetMessageBox().setMessage(picmode: true, ui_zi_faxain, new Rect(104f, 19f, 142f, 32f), ui_icon_cangbaotu, new Rect(100f, 34f, 152f, 122f), GraveChooseUI.texDicName[grave.title], rmg, Over);
				}
				isOver = false;
			}
			else
			{
				isOver = false;
				Over();
			}
		}
		else if (Global.character_.grave_campaign.finish == 5)
		{
			EnableAll();
			_return();
		}
	}

	private void Endboss()
	{
		EnableAll();
	}

	private string stamina_bar(string key)
	{
		_buy_template buy_template = Global.get_buy_template(_buy_template.TYPE.STAMINA, Global.character_.stamina_buy_count + 1);
		return string.Format(key, Global.character_.stamina, 20, Global.character_.stamina_buy_count, buy_template.gold);
	}

	public override void drawTooltips(string tip)
	{
		TooltipUI tooltipUI = __BaseUI.GetTooltipUI();
		if (tip == "stamina")
		{
			tooltipUI.ShowTooltipCommon2(stamina_bar(__BaseUI.load_str(5u)), string.Empty, 200);
		}
		else if (!(tip == "fangqi"))
		{
			if (tip != string.Empty)
			{
				tooltipUI.ShowTooltipCommon2(tip, string.Empty, 300);
			}
			else
			{
				tooltipUI.HideTooltip(string.Empty);
			}
		}
	}
