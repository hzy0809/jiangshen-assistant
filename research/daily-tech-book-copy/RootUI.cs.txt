	public static void HandleExchangeParam(ExchangeParam param)
	{
		RootBase._drawStateBubble("money" + (param.money64 - Global.character_.money));
		Global.character_.money = param.money64;
		RootBase._drawStateBubble("money" + (param.gold - Global.character_.gold));
		Global.character_.gold = param.gold;
		Global.character_.exchange.today_times = param.times;
	}

	public static void reconnect_full()
	{
		tcp_base.connection_state = tcp_base.CONNECTION_STATE.OK;
	}

	public static void reconnect_fast()
	{
		tcp_base.connection_state = tcp_base.CONNECTION_STATE.WAIT;
		Global.m_client.TryConnect(reconnect_back);

	public static void reconnect_back(bool state)
	{
		if (state)
		{
			Logger.Log("reconnect_back true");
			LoginUI.cs_user_reconnect();
		}
		else
		{
			Logger.Log("reconnect_back false");
			tcp_base.connection_state = tcp_base.CONNECTION_STATE.PREPARE;
			RetryConnectUI.tryconnect_timer = (float)(11 - RetryConnectUI.retry_count) * 1f;
		}
	}

	public static void connect_lost()
	{
		if (tcp_base.connection_state == tcp_base.CONNECTION_STATE.OK)
		{
			tcp_base.connection_state = tcp_base.CONNECTION_STATE.PREPARE;
			Logger.Log("rootUI connect_lost");
			LoadScreen component = GameObject.Find("/LoadUI").GetComponent<LoadScreen>();
			if (component.GetCurUI() != null)
			{
				component.GetCurUI().ToDisable();
			}
			if (RetryConnectUI.phase != 2)
			{
				BaseUI.GetRootUI().GetComponent<RetryConnectUI>().Myloading();
			}
			else
			{
				RetryConnectUI.RetryFailed();
			}
		}
	}

	public static void QuitApplication()

		Application.Quit();
	}

	private void Update()
	{
		if (m_client != null)
		{
			m_client.Run();
		}
		TimeUtil.Update();
		update_UIFuncs();
		if (Input.GetMouseButton(0))
		{
			med_countdown = 120f;
		}
		else
		{
			med_countdown -= Time.deltaTime;
		}
		if (Global.character_.meditation_start == 0)
		{
			MeditationUI component = BaseUI.GetRootUI().GetComponent<MeditationUI>();
			if (med_countdown < 0f)
			{
				component.StartMeditation(showui: false);
				med_countdown = 120f;
			}
			else if (component.IsVisible())
			{
				component.Hide();
			}
		}
		if (Global.character_.meditation_start != 0)
		{
			MeditationUI component2 = BaseUI.GetRootUI().GetComponent<MeditationUI>();
			if (!component2.IsVisible())
			{
				component2.SubLoading(showui: false);
			}
		}
		if (Input.GetKeyDown("escape"))
		{
			if ((DateTime.Now - last_esc).TotalSeconds < 1.0)
			{
				EscCount++;
			}
			else
			{
				EscCount = 1;
			}
			last_esc = DateTime.Now;
			if (EscCount < QuitCount)
			{
				MainUI.SetSubUI(null);
			}
		}
		RootBase.Update();
		m_bannerAnim.Update();
		if (BaseAsset.lds != null && BaseAsset.lds.IsBattling())
		{
			m_bannerAnim.Clear();
		}
		if (Global.m_bInit)
		{
			if ((DateTime.Now - m_prevTime).TotalSeconds > 30.0)
			{
				m_prevTime = DateTime.Now;
			}
			if (Global.character_.next_stamina_time <= TimeUtil.Now && !m_bRefreshCharacter)
			{
				m_bRefreshCharacter = true;
				CSPlayerRefresh_0x23 pkt = new CSPlayerRefresh_0x23();
				m_client.Send(35, pkt, null);
			}
			if (Global.character_.next_resource_time <= TimeUtil.Now && !m_bRefreshFive)
			{
				m_bRefreshFive = true;
				CSPlayerRefreshFiveClock_0x24 pkt2 = new CSPlayerRefreshFiveClock_0x24();
				m_client.Send(36, pkt2, null);
			}
			if (playtimer >= 0f)
			{
				playtimer -= Time.deltaTime;
				return;
			}
			playtimer = 60f;
			PlayerPrefs.SetString("LastStart", DateTime.Now.ToString());
		}
	}

	private void sc_assault_stronghold_load(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCAssaultStrongholdLoad_0x3AC sCAssaultStrongholdLoad_0x3AC = JSProtoBuf.Serializer.Deserialize<SCAssaultStrongholdLoad_0x3AC>(stream);
		if (sCAssaultStrongholdLoad_0x3AC.result != 1)
		{
			return;
		}
		if (sCAssaultStrongholdLoad_0x3AC.assaultstronghold != null)
		{
			Global.character_.assaultstronghold.Set(sCAssaultStrongholdLoad_0x3AC.assaultstronghold);
			if (BaseAsset.lds.GetCurUI().LevelID == "StrongHold")
			{
				for (ushort num = 0; num < Global.character_.assaultstronghold.troops.Count; num++)
				{
					AssaultStrongholdTroop assaultStrongholdTroop = Global.character_.assaultstronghold.troops[num];
					if (assaultStrongholdTroop.mesh != null)
					{
						UnityEngine.Object.Destroy(assaultStrongholdTroop.mesh);
					}
					if (assaultStrongholdTroop.arrow_mesh != null)
					{
						UnityEngine.Object.Destroy(assaultStrongholdTroop.arrow_mesh);
					}
				}
			}
			Global.character_.assaultstronghold.troops.Clear();
			foreach (db_character_assault_stronghold_troop item2 in sCAssaultStrongholdLoad_0x3AC.assaultstrongholdtroop)
			{
				AssaultStrongholdTroop assaultStrongholdTroop2 = new AssaultStrongholdTroop();
				assaultStrongholdTroop2.Set(item2);
				Global.character_.assaultstronghold.troops.Add(assaultStrongholdTroop2.id, assaultStrongholdTroop2);
				if (Global.getTemplate("assault_stronghold_monster", assaultStrongholdTroop2.idmonster) is _assault_stronghold_monster assault_stronghold_monster)
				{
					for (int i = 0; i < assault_stronghold_monster.wave; i++)
					{
						AssaultStrongholdWaveTroop item = new AssaultStrongholdWaveTroop();
						Global.character_.assaultstronghold.troops[assaultStrongholdTroop2.id].wavetroops.Add(item);
					}
				}
			}
			foreach (db_character_assault_stronghold_troop_army item3 in sCAssaultStrongholdLoad_0x3AC.assaultstrongholdtrooparmy)
			{
				AssaultStrongholdTroopArmy assaultStrongholdTroopArmy = new AssaultStrongholdTroopArmy();
				assaultStrongholdTroopArmy.Set(item3);
				if (Global.character_.assaultstronghold.troops.ContainsKey(assaultStrongholdTroopArmy.idtroop) && Global.character_.assaultstronghold.troops[assaultStrongholdTroopArmy.idtroop].wavetroops.Count >= assaultStrongholdTroopArmy.idwave)
				{
					Global.character_.assaultstronghold.troops[assaultStrongholdTroopArmy.idtroop].wavetroops[assaultStrongholdTroopArmy.idwave - 1].armys.Add(assaultStrongholdTroopArmy);
				}
			}
			Global.character_.assaultstronghold.gens.Clear();
			foreach (db_character_assault_stronghold_gen item4 in sCAssaultStrongholdLoad_0x3AC.assaultstrongholdgen)
			{
				AssaultStrongholdGeneral assaultStrongholdGeneral = new AssaultStrongholdGeneral();
				assaultStrongholdGeneral.Set(item4);
				Global.character_.assaultstronghold.gens.Add(assaultStrongholdGeneral);
			}
		}
		Global.character_.assault_stamina = sCAssaultStrongholdLoad_0x3AC.stamina;
		if (Global.character_.assaultstronghold.idtroop_battling != -1)
		{
			BaseUI.GetRootUI().StrongHoldFormation_enter((uint)Global.character_.assaultstronghold.idtroop_battling, Global.character_.assaultstronghold.troops[(ushort)Global.character_.assaultstronghold.idtroop_battling].idmonster);

		{
			for (ushort num2 = 0; num2 < Global.character_.assaultstronghold.troops.Count; num2++)

		RootMessenger rootMessenger = arg as RootMessenger;

			Global.character_.idcampaign = (ushort)sCCampaignEnter_0x.idcampaign;
		}

			foreach (game.net.data.StoC.Campaign campaign2 in sCCampaignEnter_0x.campaigns)
			{
				Global.character_.addCampaign(campaign2);

		{
			Global.character_.addCampaign(sCCampaignEnter_0x.campaign);
		}
		BaseAsset.lds.SetOldUI(BaseAsset.lds.GetCurUI());
		if (campaign.idcampaign_type == 0)
		{
			CampaignUI component = GameObject.Find("/Campaign/CampaignUI").GetComponent<CampaignUI>();

		{
			CampaignUI component3 = GameObject.Find("/Anecdote/AnecdoteUI").GetComponent<CampaignUI>();

	private void update_UIFuncs()
	{
		if (BaseAsset.lds.isVisible())
		{
			return;
		}
		CampaignTroop campaignTroop = Global.character_.getCampaign(70)?.troops_[1];
		if (LoginMessageUI.need_show && __BaseUI.UICount_Visible == 0 && campaignTroop != null && campaignTroop.killed)
		{
			if (!LoginMessageUI.isshowing)
			{
				LoginMessageUI component = GetComponent<LoginMessageUI>();
				component.MyLoading();
				LoginMessageUI.isshowing = true;
			}
		}
		else if (FlowerMessUI.need_show)
		{
			GetComponent<FlowerMessUI>().Prep(string.Empty);
			FlowerMessUI.need_show = false;
		}
		else
		{
			if ((anti_addict_ui != null && anti_addict_ui.HasMessage) || !(level_rmg != null) || __BaseUI.UICount_Visible != 0 || BaseAsset.lds.isVisible())
			{
				return;
			}
			Hashtable templates = Global.getTemplates("client_version");
			if (Global.character_.client_version < templates.Count)
			{
				BaseAsset.lds.GetCurUI().ToDisable();
				_client_version client_version = templates[(uint)templates.Count] as _client_version;
				string text = "#b=1版本#c=99FF00" + client_version.version + "\n\n#b=0#c=FFFFFF";
				text += Global.getText(client_version.idtext);
				__BaseUI.GetMessageBox().setRect(new Rect(0f, 0f, 450f, 400f));
				__BaseUI.GetMessageBox().setMessage("版本更新", text, null, VersionOK);
				Global.character_.client_version = (ushort)client_version.id;
				CSPlayerClientVersion_0x25 pkt = new CSPlayerClientVersion_0x25();
				m_client.Send(37, pkt, null);
			}
			else if (Global.character_.scenarios_.Count > 0)
			{
				if (!PlayScenario(Global.character_.scenarios_[0], ScenarioEnd))
				{
					Global.character_.scenarios_.RemoveAt(0);
				}
			}
			else if (RootBase.vctUIFuncs.Count > 0)
			{
				delegate_OnClose delegate_OnClose = RootBase.vctUIFuncs[0];
				RootBase.vctUIFuncs.RemoveAt(0);
				delegate_OnClose();
			}
			else
			{
				CraftUI.trialWarning(Global.craft_);
				CraftUI.trialWarningTactic(Global.craft_tactic);
			}
		}
	}

	public void LoadLevel(string level, RootMessenger rmg)
	{
		if (!rmg || !(rmg.LevelID == level))
		{
			Logger.Log("load level " + level);
			switch (level)
			{
			case "Campaign":
				cs_campaign_enter(0, rmg);
				break;
			case "Capital":
				cs_capital_enter(rmg);
				BgLoadPrep(999u);
				break;
			case "CapitalFriend":
				cs_capital_friend_enter(rmg);
				break;
			case "SealDevil":
				cs_sealdevil_enter(rmg);
				break;
			case "Convoy":
				cs_convoy_enter(rmg);
				break;
			case "PointRace":
				cs_pointrace_enter(rmg, OnClose);
				break;
			case "TrialMaze":
			{
				BaseAsset.lds.SetOldUI(BaseAsset.lds.GetCurUI());
				RootMessenger component = GameObject.Find("/TrialMaze").GetComponent<RootMessenger>();
				component.GetComponentInChildren<MazeUI>().first_visible = true;
				component.GetComponentInChildren<MazeDebugUI>().first_visible = false;
				component.LevelStart();
				level_rmg = component;
				BaseAsset.lds.SetCurUI(component);
				break;
			}
			case "StrongHold":
				load_stronghold();
				break;
			}
		}
	}

	public void sc_lake_temple_enter(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			EnableAll();
			return;
		}
		SCLakeTempleEnter_0x73C sCLakeTempleEnter_0x73C = JSProtoBuf.Serializer.Deserialize<SCLakeTempleEnter_0x73C>(stream);
		if (sCLakeTempleEnter_0x73C.operate == SCLakeTempleEnter_0x73C.Operate.OK)
		{
			LakeTempleFormationUI component = GameObject.Find("/LakeTempleFormation/LakeTempleFormationUI").GetComponent<LakeTempleFormationUI>();
			if (sCLakeTempleEnter_0x73C.first_time)
			{
				if (Global.character_.skytower != null)
				{
					Global.character_.skytower.challenge_times--;
				}
				if (sCLakeTempleEnter_0x73C.laketemple != null)
				{
					Global.character_.laketemple.Set(sCLakeTempleEnter_0x73C.laketemple);
					Global.character_.laketemple.m_armys.Clear();
					foreach (db_character_lake_temple_army laketemple_army in sCLakeTempleEnter_0x73C.laketemple_armys)
					{
						LakeTempleArmy lakeTempleArmy = new LakeTempleArmy();
						lakeTempleArmy.Set(laketemple_army);
						Global.character_.laketemple.m_armys.Add(lakeTempleArmy.location, lakeTempleArmy);
					}
					Global.character_.laketemple.m_bonuses.Clear();
					foreach (db_character_lake_temple_bonus laketemple_bonuse in sCLakeTempleEnter_0x73C.laketemple_bonuses)
					{
						_Treasure treasure = new _Treasure();
						treasure.make(91, laketemple_bonuse.element_id, laketemple_bonuse.element_count);
						Global.character_.laketemple.m_bonuses.Add(treasure);
					}
				}
				component.Setfirstin(_isfirst: true);
			}
			BaseAsset.lds.SetOldUI(BaseAsset.lds.GetCurUI());
			RootMessenger component2 = GameObject.Find("/LakeTempleFormation").GetComponent<RootMessenger>();
			component2.LevelStart();
			level_rmg = component2;
			BaseAsset.lds.SetCurUI(component2);
		}
		else if (sCLakeTempleEnter_0x73C.operate != SCLakeTempleEnter_0x73C.Operate.Error)
		{
		}
		EnableAll();
	}

	private void OnApplicationQuit()
	{
		if (show_setup_warning_on_quit)
		{
			show_update_msgbox();
			Application.CancelQuit();
			if (!warning_showed)
			{
				warning_showed = true;
				StartCoroutine(delayed_quit());
			}
		}
		else if (Application.platform == RuntimePlatform.WindowsPlayer)
		{
			string text = ((!openLauncherAfterQuit) ? new_setup_path : Global.launcher_exe);
			if (text.Length > 0)
			{
				Process process = new Process();
				process.EnableRaisingEvents = false;
				process.StartInfo.FileName = text;
				process.StartInfo.Arguments = string.Empty;
				int length = text.LastIndexOf("\\");
				process.StartInfo.WorkingDirectory = text.Substring(0, length);
				process.Start();
			}
		}
	}

	private IEnumerator delayed_quit()
	{
		yield return new WaitForSeconds(5f);
		show_setup_warning_on_quit = false;
		Application.Quit();
	}
