namespace Database;

public class SevenDaysLogin : _character_seven_days_login
{
	public bool hasGotLoginBonus(int day)
	{
		return day switch
		{
			1 => loginbonus1, 
			2 => loginbonus2, 
			3 => loginbonus3, 
			4 => loginbonus4, 
			5 => loginbonus5, 
			6 => loginbonus6, 
			7 => loginbonus7, 
			_ => true, 
		};
	}

	public bool canGetBonus(int day)
	{
		return Global.character_.login_days >= day && !hasGotLoginBonus(day);
	}

	public bool canGetBonus()
	{
		for (int i = 1; i <= 7; i++)
		{
			if (canGetBonus(i))
			{
				return true;
			}
		}
		return false;
	}

	public int showBonusDay()
	{
		for (int i = 1; i <= 7; i++)
		{
			if (canGetBonus(i))
			{
				return i;
			}
		}
		for (int j = 1; j <= 7; j++)
		{
			if (!hasGotLoginBonus(j))
			{
				return j;
			}
		}
		return 0;
	}

	public bool isFinish()
	{
		for (int i = 1; i <= 7; i++)
		{
			if (!hasGotLoginBonus(i))
			{
				return false;
			}
		}
		return true;
	}
}
