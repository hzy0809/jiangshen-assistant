	public override void drawTooltips(string tip)
	{
		if (!tip.StartsWith("tips"))
		{
			return;
		}
		byte id = byte.Parse(tip.Replace("tips", string.Empty));
		_Treasure treasure = new _Treasure();
		_treasure_type treasure_type = null;
		_treasure_type treasure_type2 = null;
		_return_gift return_gift = Global.getTemplate("return_gift", id) as _return_gift;
		switch (return_gift.treasure_type1)
		{
		case 0:
			__BaseUI.GetTooltipUI().ShowTooltipGR("ReturnGift", return_gift.type1_param0, null, locked: false);
			return;
		case 54:
		case 60:
			treasure.make(return_gift.treasure_type1, (int)return_gift.type1_param0, (int)return_gift.type1_param1);
			treasure.PrepTooltip();
			return;
		}
		treasure_type = Global.getTemplate("treasure_type", return_gift.treasure_type1) as _treasure_type;
		if (return_gift.treasure_type2 == 0)
		{
			__BaseUI.GetTooltipUI().ShowTooltipCommon(treasure_type.desc + " + " + return_gift.type1_param1, "ReturnGift");
			return;
		}
		treasure_type2 = Global.getTemplate("treasure_type", return_gift.treasure_type2) as _treasure_type;
		__BaseUI.GetTooltipUI().ShowTooltipCommon(treasure_type.desc + " + " + return_gift.type1_param1 + ", " + treasure_type2.desc + " + " + return_gift.type2_param1, "ReturnGift");
	}

	private void cs_return_gift_0x809()
	{
		if (mCanSendMsg)
		{
			mCanSendMsg = false;
			CSReturnGift_0x809 pkt = new CSReturnGift_0x809();
			RootUI.m_client.Send(2057, pkt, sc_return_gift_0x809);
		}
	}

	private void sc_return_gift_0x809(MemoryStream stream, object arg)
	{
		mCanSendMsg = true;
		if (stream == null)
		{
			return;
		}
		SCReturnGift_0x809 sCReturnGift_0x = JSProtoBuf.Serializer.Deserialize<SCReturnGift_0x809>(stream);
		if (sCReturnGift_0x.isOk)
		{
			Global.character_.return_gift_status = (short)sCReturnGift_0x.return_gift_status;
			Global.character_.calcReturnGiftRedCount();
			List<_Treasure> list = new List<_Treasure>();
			foreach (Treasure bonu in sCReturnGift_0x.bonus)
			{
				_Treasure treasure = new _Treasure();
				treasure.make(bonu);
				list.Add(treasure);
			}
			if (list.Count > 0)
			{
				BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
				battleResultUI.setBonus(list);
				battleResultUI.MyLoading();
			}
		}
		EnableAll();
	}
