	protected override void Web_OnGUI()
	{
		if (!BeginGUI(800f, 480f, __BaseUI.UIDepth_Sub, __BaseUI._bgStyle, titleStyle))
		{
			return;
		}
		if (ui_di[pageList[curPage]] != null)
		{
			__BaseUI.Label(__BaseUI._rect(169f, 61f, 610f, 397f), string.Empty, ui_di[pageList[curPage]]);
		}
		if (drawPage[pageList[curPage]] != null)
		{
			drawPage[pageList[curPage]]();
		}
		int num = 0;
		pagemax = (pageList.Count - 1) / 7;
		GUI.BeginGroup(__BaseUI._rect(20f, 86f, 171f, 343f));
		int num2 = -1 * pagenum * 49 * 7;
		if (DownTimer > DateTime.Now)
		{
			double num3 = (1.0 - (DownTimer - DateTime.Now).TotalSeconds / 0.3) * 49.0 * 7.0;
			num2 -= (int)num3;
		}
		else if (UpTimer > DateTime.Now)
		{
			double num4 = (1.0 - (UpTimer - DateTime.Now).TotalSeconds / 0.3) * 49.0 * 7.0;
			num2 += (int)num4;
		}
		else if (downMoving)
		{
			downMoving = false;
			pagenum++;
			curPage = pagenum * 7;
		}
		else if (upMoving)
		{
			upMoving = false;
			pagenum--;
			curPage = pagenum * 7;
		}
		GUI.BeginGroup(__BaseUI._rect(0f, num2, 171f, 147f));
		for (int i = 0; i < pageList.Count; i++)
		{
			if (i != curPage)
			{
				if (ui_bar.Button(IsEnable(), __BaseUI._rect(0f, 49 * num, 171f, 50f), string.Empty))
				{
					curPage = i;
				}
			}
			else
			{
				__BaseUI.Label(__BaseUI._rect(0f, 49 * num, 171f, 50f), string.Empty, ui_bar_down);
			}
			__BaseUI.Label(__BaseUI._rect(15f, 14 + 49 * num, 92f, 20f), string.Empty, ui_zi[pageList[i]]);
			if (checkAward[pageList[i]] != null && checkAward[pageList[i]]())
			{
				if (pageList[i] == pageidx[2])
				{
					__BaseUI.Box(__BaseUI._rect(119f, 0 + 49 * num, 28f, 28f), ui_dui);
				}
				else
				{
					__BaseUI.Label(__BaseUI._rect(119f, 0 + 49 * num, 28f, 28f), string.Empty, __BaseUI._ui_jiang);
				}
			}
			num++;
		}
		GUI.EndGroup();
		GUI.EndGroup();
		if (DownTimer < DateTime.Now && UpTimer < DateTime.Now)
		{
			if (pagenum < pagemax && ui_down.Button(bBtn: true, __BaseUI._rect(61f, 428f, 56f, 38f), string.Empty))
			{
				DownTimer = DateTime.Now.AddSeconds(0.3);
				downMoving = true;
			}
			if (pagenum > 0 && ui_up.Button(bBtn: true, __BaseUI._rect(61f, 51f, 56f, 38f), string.Empty))
			{
				UpTimer = DateTime.Now.AddSeconds(0.3);
				upMoving = true;
			}
		}
		if (EndGUI(hasClose: true, hasBlock: true, isX: false, hasFrameAnimation: false))
		{
			MainUI.SetSubUI(null);
		}
	}

	private void sc_vip_daily_shop_buy(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCVipDailyShopBuy_0x3B3 sCVipDailyShopBuy_0x3B = JSProtoBuf.Serializer.Deserialize<SCVipDailyShopBuy_0x3B3>(stream);
		if (sCVipDailyShopBuy_0x3B.result == 1)
		{
			if (sCVipDailyShopBuy_0x3B.bonus != null)
			{
				List<_Treasure> list = new List<_Treasure>();
				_Treasure treasure = new _Treasure();
				treasure.make(sCVipDailyShopBuy_0x3B.bonus);
				list.Add(treasure);
				BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
				battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, BonusClose);
				battleResultUI.MyLoading();
			}
			Global.character_.gold -= (int)sCVipDailyShopBuy_0x3B.goldcost;
			__BaseUI.drawStateBubble("gold-" + sCVipDailyShopBuy_0x3B.goldcost);
			VipDailyShopItem vipDailyShopItem = new VipDailyShopItem();
			vipDailyShopItem.Set(sCVipDailyShopBuy_0x3B.item);
			Global.character_.vipdailyshopitems.Add(vipDailyShopItem.iditem, vipDailyShopItem);
		}
	}

	private void sc_vip_daily_gift_get(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCVIPDailyGiftGet_0x71C sCVIPDailyGiftGet_0x71C = JSProtoBuf.Serializer.Deserialize<SCVIPDailyGiftGet_0x71C>(stream);
		if (sCVIPDailyGiftGet_0x71C.operate != SCVIPDailyGiftGet_0x71C.Operate.OK)
		{
			return;
		}
		Global.character_.vip_gift_today = true;
		List<_Treasure> list = new List<_Treasure>();
		foreach (Treasure bonu in sCVIPDailyGiftGet_0x71C.bonus)
		{
			_Treasure treasure = new _Treasure();
			treasure.make(bonu);
			list.Add(treasure);
		}
		BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
		battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, EnableAll);
		battleResultUI.MyLoading();
	}
