	public void wantCombat(SealMonsterTroop troop)
	{
		if (!troop.finish())
		{
			m_troop = troop;
			Hashtable hashtable = new Hashtable();
			__BaseUI.subui = "NPCBattle";
			hashtable.Add("subui", __BaseUI.subui);
			hashtable.Add("campaignid", troop.idtroop);
			hashtable.Add("troopid", 0);
			hashtable.Add("substate", "sealmonster_troop");
			hashtable.Add("viewonly", false);
			hashtable.Add("OnAttack", new FormationUI.delegate_OnAttack(OnAttack));
			SendMessage("SubLoading", hashtable, SendMessageOptions.DontRequireReceiver);
			FormationUI component = GetComponent<FormationUI>();
			MainUI.SetSubUI(component);
		}
	}

	private bool OnAttack()
	{
		DisableAll();
		if (m_troop.getCD() > TimeUtil.Now)
		{
			string message = __BaseUI.format("battle_cd", new object[3]
			{
				m_troop.getName(),
				m_troop.getTimes(),
				Global.getFormatTime((int)(m_troop.getCD() - TimeUtil.Now))
			});
			__BaseUI.GetMessageBox().setMessage("鎴樻枟鍐峰嵈", message, null, EnableAll);
			return false;
		}
		cs_sealmonster_battle_0x368();
		return true;
	}

	private void cs_sealmonster_battle_0x368()
	{
		BaseAsset.lds.Visible_sub();
		DisableAll();
		CSSealMonsterBattle_0x368 cSSealMonsterBattle_0x = new CSSealMonsterBattle_0x368();
		cSSealMonsterBattle_0x.idtroop = m_troop.idtroop;
		RootUI.m_client.Send(872, cSSealMonsterBattle_0x, sc_sealmonster_battle_0x368);
	}

	private void sc_sealmonster_battle_0x368(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			BaseAsset.lds.Hide();
			EnableAll();
			return;
		}
		SCSealMonsterBattle_0x368 sCSealMonsterBattle_0x = JSProtoBuf.Serializer.Deserialize<SCSealMonsterBattle_0x368>(stream);
		BattleUI battleUI = BaseUI.GetBattleUI();
		battleUI.setPrevUI(rmg);
		battleUI.setReport(sCSealMonsterBattle_0x.report, wait_for_end: true, use_battle_end: true, this, isopponent: false);
		battleUI.ViewType = enumBattleViewType.viewFull;
		if (sCSealMonsterBattle_0x.iswin)
		{
			m_exp = sCSealMonsterBattle_0x.exp;
			m_buff = sCSealMonsterBattle_0x.buff;
		}
		else
		{
			m_cd = sCSealMonsterBattle_0x.cd;
		}
		BaseUI.GetRootUI().battle_enter(fslds: false);
		BaseAsset.lds.SetBattling(bBattling: true);
		MainUI.SetSubUI(null);
	}

		}
	}

	private void starScaleFunc(float ts, ref float scale)
	{
		if (ts < 1f)
		{
			scale = 1f + ts * 0.7f;
		}
		else
		{
			scale = 0f;
		}
	}

	public void BattleEnd(SCBattleEnd_0x408 pkt)
	{
		if (pkt.iswin)
		{
			m_troop.setFinish(bfinish: true);
			Global.character_.sealmonsterinfo.buff = (int)m_buff;
			List<_Treasure> list = new List<_Treasure>();
			_Treasure treasure = new _Treasure();
			treasure.make(8, 0, (int)m_exp);
			list.Add(treasure);
			BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
			BattleUI battleUI = BaseUI.GetBattleUI();
			battleResultUI.Set(rmg, string.Empty, string.Empty, 0, 0, list, BonusClose);
			battleUI.setClose(BonusOpen);
		}
		else
		{
			m_troop.setCD(m_cd);
			m_troop.addTimes();
			BonusClose();
		}
	}

	public void sc_sealmonster_refresh(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCSealMonsterRefresh_0x369 sCSealMonsterRefresh_0x = JSProtoBuf.Serializer.Deserialize<SCSealMonsterRefresh_0x369>(stream);
		Global.character_.sealmonsterinfo.Set(sCSealMonsterRefresh_0x.sealmonsterinfo);
		for (byte b = 0; b < Global.character_.sealmonsterinfo.m_troops.Length; b++)
		{
			if (Global.character_.sealmonsterinfo.m_troops[b].mesh != null)
			{
				UnityEngine.Object.Destroy(Global.character_.sealmonsterinfo.m_troops[b].mesh);
			}
			Global.character_.sealmonsterinfo.m_troops[b] = new SealMonsterTroop();
			Global.character_.sealmonsterinfo.m_troops[b].idtroop = (byte)(b + 1);
			Global.character_.sealmonsterinfo.m_troops[b].m_armys.Clear();
			Global.character_.sealmonsterinfo.m_troops[b].setCD(0u);
			Global.character_.sealmonsterinfo.m_troops[b].setTimes(0);
		}
		foreach (db_character_sealmonster_army item in sCSealMonsterRefresh_0x.sealmonsterarmy)
		{

	protected override void AssetLoaded(string AssetName)
	{
		for (byte b = 1; b <= 6; b++)
		{
			byte troopid = Global.character_.sealmonsterinfo.getTroopid(b);
			if (troopid != 0)
			{
				_campaign campaign = Global.getTemplate("campaign", 268u) as _campaign;
				_campaign_troop campaign_troop = (_campaign_troop)campaign.m_troops[troopid];
				string text = (Global.getTemplate("troop_animation", campaign_troop.idmodel) as _troop_animation).name;
				if (AssetName == "battle/" + text)
				{
					AssetBundle asset = BaseAsset.was.GetAsset("battle/" + text);
					if (asset != null)
					{
						SealMonsterTroop sealMonsterTroop = Global.character_.sealmonsterinfo.m_troops[b - 1];
						sealMonsterTroop.mesh = UnityEngine.Object.Instantiate(asset.mainAsset) as GameObject;
						sealMonsterTroop.mesh.transform.parent = base.gameObject.transform;
						BaseAsset.SetLayerRecursively(sealMonsterTroop.mesh, 9);
						sealMonsterTroop.mesh.transform.position = pos_container.Positions[b - 1].position;
						sealMonsterTroop.mesh.transform.rotation = pos_container.Positions[b - 1].rotation;
						sealMonsterTroop.mesh.transform.parent = mesh.transform;
						sealMonsterTroop.mesh.AddComponent<SealMonsterTroopScript>().setTroop(this, sealMonsterTroop);
						if (sealMonsterTroop.finish())
						{
							sealMonsterTroop.mesh.animation.Play("dead");
						}
					}
				}
			}
		}
	}
