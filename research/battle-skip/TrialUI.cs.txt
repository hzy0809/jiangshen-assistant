	private void callBoss()
	{
		Hashtable hashtable = new Hashtable();
		__BaseUI.subui = "NPCBattle";
		hashtable.Add("subui", __BaseUI.subui);
		hashtable.Add("campaignid", Global.character_.trialinfo.bossid);
		hashtable.Add("troopid", 0);
		hashtable.Add("substate", "trial_boss");
		hashtable.Add("viewonly", false);
		hashtable.Add("OnAttack", new FormationUI.delegate_OnAttack(OnAttack));
		FormationUI component = GameObject.Find("Trial/TrialUI").GetComponent<FormationUI>();
		component.SubLoading(hashtable);
		MainUI.SetSubUI(component);
	}

	private bool OnAttack()
	{
		int num = (int)(Global.character_.trialinfo.bosscd - TimeUtil.Now);
		if (num > 0)
		{
			return false;
		}
		DisableAll();
		cs_trial_battle_boss();
		return true;
	}

	private void cs_trial_battle_boss()
	{
		BaseAsset.lds.Visible_sub();
		DisableAll();
		CSTrialBossBattle_0x336 pkt = new CSTrialBossBattle_0x336();
		RootUI.m_client.Send(822, pkt, sc_trial_battle_boss);
	}

	private void sc_trial_battle_boss(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			BaseAsset.lds.Hide();
			EnableAll();
			return;
		}
		SCTrialBossBattle_0x336 sCTrialBossBattle_0x = JSProtoBuf.Serializer.Deserialize<SCTrialBossBattle_0x336>(stream);
		BattleUI battleUI = BaseUI.GetBattleUI();
		battleUI.setPrevUI(BaseAsset.lds.GetCurUI());
		battleUI.setReport(sCTrialBossBattle_0x.report, wait_for_end: true, use_battle_end: true, this, isopponent: false);
		battleUI.ViewType = enumBattleViewType.viewFull;
		if (sCTrialBossBattle_0x.iswin)
		{
		}
		Global.character_.trialinfo.bosscd = sCTrialBossBattle_0x.cooldown;
		BaseUI.GetRootUI().battle_enter(fslds: false);
		BaseAsset.lds.SetBattling(bBattling: true);
		MainUI.SetSubUI(null);
	}

	public void BattleEnd(bool bWin)
	{
		BattleUI battleUI = BaseUI.GetBattleUI();
		if (bWin)
		{
			Global.character_.trialinfo.bosskilled = 1;
			Global.character_.trialinfo.bosscd = TimeUtil.Now;
			battleUI.setClose(BonusClose);
		}
		else
		{
			battleUI.setClose(FailedOpen);
		}
	}

	private void FailedOpen()
	{
		MainUI.SetSubUI(BaseUI.GetBattleFailedUI());
		BaseUI.GetBattleFailedUI().MyLoading();
	}

		{
			if (Global.character_.trialinfo.icon[i] != 0)

		}

	public void sc_trial_chest(MemoryStream stream, object arg)
	{
		if (stream == null)
		{
			return;
		}
		SCTrialChest_0x335 sCTrialChest_0x = JSProtoBuf.Serializer.Deserialize<SCTrialChest_0x335>(stream);
		if (sCTrialChest_0x.operate == SCTrialChest_0x335.Operate.OK)
		{
			List<_Treasure> list = new List<_Treasure>();
			_Treasure treasure = new _Treasure();
			treasure.make(8, 0, (int)sCTrialChest_0x.exp);
			list.Add(treasure);
			if (sCTrialChest_0x.stone != 0)
			{
				treasure = new _Treasure();
				treasure.make(30, 0, (int)sCTrialChest_0x.stone);
				list.Add(treasure);
			}
			if (sCTrialChest_0x.enchantpick != 0)
			{
				treasure = new _Treasure();
				treasure.make(37, 0, (int)sCTrialChest_0x.enchantpick);
				list.Add(treasure);
			}
			if (sCTrialChest_0x.money != 0)
			{
				treasure = new _Treasure();
				treasure.make(4, 0, (int)sCTrialChest_0x.money);
				list.Add(treasure);
			}
			if (sCTrialChest_0x.gold != 0)
			{
				treasure = new _Treasure();
				treasure.make(7, 0, (int)sCTrialChest_0x.gold);
				list.Add(treasure);
			}
			Global.character_.trialinfo.chest_got++;
			BattleResultUI battleResultUI = __BaseUI.GetBattleResultUI();
			string empty = string.Empty;
			battleResultUI.Set(rmg, string.Empty, empty, 0, 0, list, EnableAll);
			__BaseUI.GetBattleResultUI().PrepareLoading("nocollecting");
			__BaseUI.GetBattleResultUI().StartLoading();
			if (Global.character_.trialinfo.chest_got == 5)
			{
				ServerOpenTrialAward serverOpenTrialAward = new ServerOpenTrialAward();
				int num = (byte)MainUI.CalcServerOpenRefreshDays();
				if (Global.getTemplate("server_open_trial_award", (uint)num) is _server_open_trial_award)
				{
					serverOpenTrialAward.finished = 1;
					serverOpenTrialAward.day = (byte)num;
					if (!Global.character_.trialawards.ContainsKey(serverOpenTrialAward.day))
					{
						Global.character_.trialawards.Add(serverOpenTrialAward.day, serverOpenTrialAward);
					}
				}
			}
		}
		if (Global.character_.trialinfo.bosskilled == 0)
		{
			MyLoading();
		}
		else
		{
			__BaseUI.GetMessageBox().setMessage(string.Empty, __BaseUI.load_str(20571u), rmg);
		}
	}

				ServerOpenTrialAward serverOpenTrialAward = new ServerOpenTrialAward();
				int num2 = (byte)MainUI.CalcServerOpenRefreshDays();
				if (Global.getTemplate("server_open_trial_award", (uint)num2) is _server_open_trial_award)
				{
					serverOpenTrialAward.finished = 1;
					serverOpenTrialAward.day = (byte)num2;
					if (!Global.character_.trialawards.ContainsKey(serverOpenTrialAward.day))
					{
						Global.character_.trialawards.Add(serverOpenTrialAward.day, serverOpenTrialAward);
					}
				}
			}
		}
		else if (sCTrialRapid_0x.operate == SCTrialRapid_0x337.Operate.Nostamina)
		{
			MainUI componentInChildren = BaseAsset.lds.GetCurUI().GetComponentInChildren<MainUI>();
			componentInChildren._stamina();
		}
		sending = false;
	}

		_buy_template buy_template = Global.get_buy_template(_buy_template.TYPE.STAMINA, Global.character_.stamina_buy_count + 1);
		return string.Format(key, Global.character_.stamina, 20, Global.character_.stamina_buy_count, buy_template.gold);
	}
