	private EnemyArmyDisplayData createEnemyDisplayData(uint idtroop)
	{
		uint id = 318u;
		_general_book_template general_book_template = Global.getTemplate("general_book_template", troopid) as _general_book_template;
		if (Global.getTemplate("campaign", id) is _campaign campaign && general_book_template != null && campaign.m_troops.ContainsKey(general_book_template.enemyid))
		{
			_campaign_troop campaign_troop = campaign.m_troops[general_book_template.enemyid] as _campaign_troop;
			EnemyArmyDisplayData enemyArmyDisplayData = new EnemyArmyDisplayData();
			enemyArmyDisplayData.troop = new CampaignTroop(campaign_troop, _anecdote: false, _fairy: false);
			enemyArmyDisplayData.troop.limit_count = -1;
			AddAssetSub("face" + enemyArmyDisplayData.troop.icon);
			{
				foreach (_campaign_troop_army army2 in campaign_troop.m_armys)
				{
					if (Global.getTemplate("army", army2.idarmy) is _army army)
					{
						EnemyLegionDisplayData enemyLegionDisplayData = new EnemyLegionDisplayData();
						enemyLegionDisplayData.name = army2.army_name;
						enemyLegionDisplayData.location = army2.location;
						enemyLegionDisplayData.icon = army.icon;
						enemyLegionDisplayData.skillid = army.activeskill;
						AddAssetSub("face" + enemyLegionDisplayData.icon);
						if (Global.getTemplate("army_template", army.idarmy_template) is _army_template army_template)
						{
							enemyLegionDisplayData.army_temp = army_template;
							AddAssetSub("battle/" + army_template.model_name_male);
						}
						enemyArmyDisplayData.legion_layout.Add(enemyLegionDisplayData);
					}
				}
				return enemyArmyDisplayData;
			}
		}
		return null;
	}

				Object.Destroy(item.gameObject);
			}

		float[] array3 = new float[3] { 0f, -0.7f, 0.7f };

			_army_template army_temp = item2.army_temp;

			int num2 = item2.location % 4;
			GameObject gameObject = new GameObject(item2.name);
			gameObject.transform.parent = m_ArmyFormation[1].transform;
			gameObject.transform.localPosition = new Vector3((-4.5f + (float)num2 * 3f) * -1f, 0f, 4.5f - (float)num * 3f);

	public void PrepareGroup()
	{
		if (!base.enabled)
		{
			return;
		}
		List<byte> list = new List<byte>();
		foreach (KeyValuePair<byte, GeneralDisplayData> item in group_layout)
		{
			if (m_gens[item.Key - 1] == -1)
			{
				Object.Destroy(item.Value.obj);
				list.Add(item.Key);
			}
		}
		for (int i = 0; i < list.Count; i++)
		{
			group_layout.Remove(list[i]);
		}
		for (int j = 0; j < 16; j++)
		{
			if (m_gens[j] != -1)
			{
				if (group_layout.ContainsKey((byte)(j + 1)))
				{
					group_layout[(byte)(j + 1)].Init(m_gens[j]);
				}
				else
				{
					group_layout[(byte)(j + 1)] = new GeneralDisplayData(m_gens[j]);
				}
			}
		}
		foreach (KeyValuePair<byte, GeneralDisplayData> item2 in group_layout)
		{
			if (item2.Value.need_update)
			{
				if (item2.Value.asset_name == string.Empty)
				{
					Object.Destroy(item2.Value.obj);
					setup_general_obj(item2.Key, item2.Value, null);
				}
				else
				{
					AddAssetSub(item2.Value.asset_name);
				}
			}
		}
		current_npc_level = Global.character_.convoy.npc_level;
	}

	private void setup_general_obj(byte location, GeneralDisplayData gdata, GameObject gen_obj)
	{
		if (gen_obj == null)
		{
			gen_obj = new GameObject("genid:" + gdata.gen_id);
		}
		if (gen_obj.collider == null)
		{
			BoxCollider boxCollider = gen_obj.AddComponent<BoxCollider>();
			boxCollider.size = new Vector3(0.8f, 1.5f, 0.8f);
			boxCollider.center = new Vector3(0f, 0.75f, 0f);
			boxCollider.isTrigger = true;
		}
		AnimationTest component = gen_obj.GetComponent<AnimationTest>();
		if (component != null)
		{
			component.enabled = false;
		}
		BaseAsset.SetLayerRecursively(gen_obj, m_ArmyFormation[0].layer);
		gen_obj.transform.parent = m_ArmyFormation[0].transform;
		int num = (location - 1) / 4;
		int num2 = (location - 1) % 4;
		gen_obj.transform.localPosition = new Vector3(-4.5f + (float)num2 * 3f, 0f, 4.5f - (float)num * 3f);
		gen_obj.transform.localRotation = Quaternion.identity;
		gdata.obj = gen_obj;
		gdata.need_update = false;
	}

	private void create_loaded_asset(string AssetName)
	{
		foreach (KeyValuePair<byte, GeneralDisplayData> item in group_layout)
		{
			if (item.Value.need_update && item.Value.asset_name == AssetName)
			{
				Object.Destroy(item.Value.obj);
				setup_general_obj(item.Key, item.Value, BaseAsset.Instantiate_mainAsset(AssetName) as GameObject);
			}
		}
		float[] array = new float[5] { 0f, -0.7f, 0.7f, -0.7f, 0.7f };
		float[] array2 = new float[5] { 0f, 0.7f, 0.7f, -0.7f, -0.7f };
		float[] array3 = new float[3] { 0f, -0.7f, 0.7f };
		float[] array4 = new float[3] { 0.7f, -0.7f, -0.7f };
		foreach (EnemyLegionDisplayData item2 in enemy_layout.legion_layout)
		{
			if (item2.collider.transform.childCount != 0 || !(AssetName == "battle/" + item2.army_temp.model_name_male))
			{
				continue;
			}
			for (int i = 0; i < item2.army_temp.model_num; i++)
			{
				GameObject gameObject = BaseAsset.Instantiate_mainAsset(AssetName) as GameObject;
				if (gameObject != null)
				{
					Vector3 localPosition = gameObject.transform.localPosition;
					Quaternion localRotation = gameObject.transform.localRotation;
					gameObject.transform.parent = item2.collider.transform;
					gameObject.transform.localPosition = localPosition + new Vector3((item2.army_temp.model_num != 3) ? array[i] : array3[i], 0f, (item2.army_temp.model_num != 3) ? array2[i] : array4[i]);
					gameObject.transform.localRotation = localRotation;
					AnimationTest component = gameObject.GetComponent<AnimationTest>();
					if (component != null)
					{
						component.enabled = false;
					}
				}
			}
		}
		if (newgen_asset_name == AssetName)
		{
			GameObject gameObject2 = BaseAsset.Instantiate_mainAsset(AssetName) as GameObject;
			if (gameObject2 != null)
			{
				BaseAsset.SetLayerRecursively(gameObject2, 9);
				gameObject2.transform.parent = m_ArmyFormation[0].transform;
				gameObject2.transform.position = new_gen_obj.transform.position;
				gameObject2.transform.localRotation = Quaternion.identity;
				AnimationTest component2 = gameObject2.GetComponent<AnimationTest>();
				if (component2 != null)
				{
					component2.enabled = false;
				}
				Object.Destroy(new_gen_obj);
				new_gen_obj = gameObject2;
			}
			newgen_asset_name = null;
		}
		if (!AssetName.Contains("face"))
		{
			return;
		}
		foreach (_general_book_user_army gen in gens)
		{
			if (gen.idgenernal_template == 0 && !armyfaces.ContainsKey(gen.id))
			{
				armyfaces.Add(gen.id, BaseAsset.LoadTexture("face" + Global.character_.icon));
			}
			else if (Global.getTemplate("army", gen.idarmy) is _army army && !armyfaces.ContainsKey(gen.id) && AssetName == "face" + army.icon)
			{
				armyfaces.Add(gen.id, BaseAsset.LoadTexture(AssetName));
			}
		}
	}

	protected override void Show()
	{
		base.Show();
		sended = false;
		InitUIResource();
		InitScene();
		Visible();
		Enable();
		Hashtable templates = Global.getTemplates("general_book_user_army");
		gens.Clear();
		_general_book_template general_book_template = Global.getTemplate("general_book_template", troopid) as _general_book_template;
		foreach (object value in templates.Values)
		{
			_general_book_user_army general_book_user_army = (_general_book_user_army)value;
			if (general_book_user_army.idgen != troopid)
			{
				continue;
			}
			gens.Add(general_book_user_army);
			if (general_book_user_army.idgenernal_template != 0)
			{
				continue;
			}
			bool flag = false;
			int[] array = m_gens;
			foreach (int num in array)
			{
				if (num == general_book_user_army.id)
				{
					flag = true;
					break;
				}
			}
			if (!flag)
			{
				m_gens[0] = general_book_user_army.id;
			}
			mainchar = general_book_user_army;
		}
		if (general_book_template != null)
		{
			GeneralSelectMaxnum = general_book_template.gennum;
		}
		PrepareGroup();
	}

	private void Update()
	{
		if (!Mathf.Approximately(gen_select_alpha, gen_select_target_alpha))
		{
			gen_select_alpha = Mathf.MoveTowards(gen_select_alpha, gen_select_target_alpha, Time.deltaTime * 5f);
		}
		onfield_obj_tooltip = string.Empty;
		switch (drag_stat)
		{
		case DragStat.Picking:
			pickOnFieldGen();
			break;
		case DragStat.DraggingField:
			dragOnFieldGen();
			break;
		case DragStat.DraggingNew:
			dragNewGeneral();
			break;
		}
	}

	private void pickOnFieldGen()
	{
		bool flag = false;
		if (Camera.main == null)
		{
			return;
		}
		Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
		foreach (KeyValuePair<byte, GeneralDisplayData> item in group_layout)
		{
			if (!(item.Value.obj != null) || !(item.Value.obj.collider != null) || !item.Value.obj.collider.Raycast(ray, out var _, float.PositiveInfinity))
			{
				continue;
			}
			if (Input.GetMouseButtonDown(0))
			{
				startDragOnFieldGeneral(item.Key);
				break;
			}
			if (item.Value.idgeneral_template > 0)
			{
				onfield_obj_tooltip = "MineG:" + item.Key;
			}
			else if (item.Value.idgeneral_template == 0)
			{
				onfield_obj_tooltip = "MineG:" + item.Key;
			}
			else if (item.Value.idgeneral_template != -2)
			{
			}
			flag = true;
			break;
		}
		foreach (EnemyLegionDisplayData item2 in enemy_layout.legion_layout)
		{
			if (item2.collider.Raycast(ray, out var _, float.PositiveInfinity))
			{
				onfield_obj_tooltip = "FieldGen:" + item2.name + "|" + enemy_layout.troop.level + "|" + item2.icon + "|" + item2.color + "|" + item2.army_temp.id + "|" + item2.skillid;
				break;
			}
		}
	}

	private void startDragNewGeneral(int idtemp)
	{
		if (!IsEnable() || !(Global.getTemplate("general_book_user_army", (uint)idtemp) is _general_book_user_army general_book_user_army))
		{
			return;
		}
		id_newgen_temp = idtemp;
		if (!(Global.getTemplate("army", general_book_user_army.idarmy) is _army army))
		{
			return;
		}
		if (Global.getTemplate("army_template", army.idarmy_template) is _army_template army_template)
		{
			newgen_asset_name = "battle/" + army_template.model_name_male;
			if (new_gen_obj != null)
			{
				Object.Destroy(new_gen_obj);
			}
			new_gen_obj = new GameObject("new_gen");
			AddAssetSub(newgen_asset_name);
			StartLoadingSub();
		}
		gen_select_target_alpha = 0f;
		drag_stat = DragStat.DraggingNew;
	}

	private void startDragOnFieldGeneral(byte pos)
	{
		if (IsEnable())
		{
			drag_stat = DragStat.DraggingField;
			dragging_onfield_pos = pos;
		}
	}

	private void dragOnFieldGen()
	{
		if (dragging_onfield_pos < 0)
		{
			return;
		}
		if (Input.GetMouseButtonUp(0))
		{
			dropOnFieldGeneral();
			return;
		}
		byte key = (byte)dragging_onfield_pos;
		if (group_layout.ContainsKey(key))
		{
			dragging_obj(group_layout[key].obj, drag_onfield_gen_show_pos);
		}
	}

	private void dragNewGeneral()
	{
		if (Input.GetMouseButtonUp(0))
		{
			if (tempcount != -1 && group_layout.ContainsKey((byte)tempcount) && group_layout[(byte)tempcount].idgeneral_template == 0)
			{
				mc = true;
				tempcount = -1;
			}
			dropNewGeneral();
		}
		else
		{
			dragging_obj(new_gen_obj, drag_new_gen_show_pos);
		}
	}

	private bool drag_new_gen_show_pos(int collide_location)
	{
		List<byte> list = new List<byte>();
		foreach (KeyValuePair<byte, GeneralDisplayData> item in group_layout)
		{
			if (item.Value.gen_id > 0)
			{
				list.Add(item.Key);
			}
		}
		tempcount = collide_location;
		if (list.Count >= GeneralSelectMaxnum || group_layout.ContainsKey((byte)collide_location))
		{
			if (group_layout.ContainsKey((byte)collide_location) && group_layout[(byte)collide_location].idgeneral_template == 0)
			{
				return false;
			}
			return list.Contains((byte)collide_location);
		}
		return !group_layout.ContainsKey((byte)collide_location);
	}

	private bool drag_onfield_gen_show_pos(int collide_location)
	{
		return true;
	}

	private void dragging_obj(GameObject obj, delegate_showPosEffect show_pos_effect)
	{
		if (obj == null)
		{
			return;
		}
		Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
		int num = 0;
		for (num = 0; num < 16; num++)
		{
			if (m_drag_indicator[num].Raycast(ray, out var hitInfo, float.PositiveInfinity))
			{
				obj.transform.position = new Vector3(hitInfo.point.x, (!(hitInfo.point.y < m_ArmyFormation[0].transform.position.y)) ? hitInfo.point.y : m_ArmyFormation[0].transform.position.y, hitInfo.point.z);
				if (show_pos_effect == null || show_pos_effect(num + 1))
				{
					m_pos_eff.SetActiveRecursively(state: true);
					m_pos_eff.transform.position = m_drag_indicator[num].transform.position;
				}
				else
				{
					m_pos_eff.SetActiveRecursively(state: false);
				}
				break;
			}
		}
		if (num >= 16)
		{
			Vector3 point = ray.GetPoint((m_ArmyFormation[0].transform.position.y - ray.origin.y) / ray.direction.y);
			obj.transform.position = point;
			m_pos_eff.SetActiveRecursively(state: false);
		}
	}

	private void dropOnFieldGeneral()
	{
		if (dragging_onfield_pos < 0)

			return;
		}
		byte b = (byte)dragging_onfield_pos;
		if (!group_layout.ContainsKey(b))

		}
		Ray ray = Camera.main.ScreenPointToRay(Input.mousePosition);
		int num = 0;
		for (num = 0; num < 16; num++)
		{
			if (!m_drag_indicator[num].Raycast(ray, out var _, float.PositiveInfinity))
			{
				continue;
			}
			group_layout[b].obj.transform.position = new Vector3(m_drag_indicator[num].transform.position.x, m_ArmyFormation[0].transform.position.y, m_drag_indicator[num].transform.position.z);
			byte b2 = (byte)(num + 1);
			if (b == b2)
			{
				break;
			}
			if (group_layout.ContainsKey(b2))
			{
				GeneralDisplayData generalDisplayData = group_layout[b2];
				group_layout[b2] = group_layout[b];
				if (generalDisplayData != null)
				{
					int num2 = (b - 1) / 4;
					int num3 = (b - 1) % 4;
					generalDisplayData.obj.transform.localPosition = new Vector3(-4.5f + (float)num3 * 3f, 0f, 4.5f - (float)num2 * 3f);
				}
				group_layout[b] = generalDisplayData;
				int num4 = m_gens[b - 1];
				m_gens[b - 1] = generalDisplayData.gen_id;
				m_gens[b2 - 1] = num4;
			}
			else
			{
				group_layout[b2] = group_layout[b];
				group_layout.Remove(b);
				m_gens[b2 - 1] = group_layout[b2].gen_id;
				m_gens[b - 1] = -1;
			}
			break;
		}
		if (num >= 16)

			int num5 = (b - 1) / 4;
			int num6 = (b - 1) % 4;
			group_layout[b].obj.transform.localPosition = new Vector3(-4.5f + (float)num6 * 3f, 0f, 4.5f - (float)num5 * 3f);

	}

	public void BattleEnd(bool bWin)
	{
		BattleUI battleUI = BaseUI.GetBattleUI();
		if (bWin)
		{
			Global.character_.generalbooks[(ushort)troopid] = temp_book;
			temp_book.cd = 0u;
			temp_book.finish = true;
			Hashtable templates = Global.getTemplates("general_book_template");
			Global.character_.generalbookcount = 0;
			foreach (DictionaryEntry item in templates)
			{
				_general_book_template general_book_template = item.Value as _general_book_template;
				if (CheckGeneralbook(general_book_template.id))
				{
					Global.character_.addParam("generalbookcount", 0, 1);
				}
			}
			battleUI.setClose(WinOpen);
		}
		else
		{
			Global.character_.generalbooks[(ushort)troopid] = temp_book;
			battleUI.setClose(FailedOpen);
		}
	}

	private void WinOpen()
	{
		Hashtable templates = Global.getTemplates("general_group");
		int num = -1;
		bool flag = true;
		foreach (DictionaryEntry item in templates)
		{
			if (item.Value is _general_group general_group && general_group.idgen == troopid)
			{
				num = general_group.idgroup;
				break;
			}
		}
		if (num != -1)
		{
			foreach (DictionaryEntry item2 in templates)
			{
				if (item2.Value is _general_group general_group2 && general_group2.idgroup == num)
				{
					if (!Global.character_.generalbooks.ContainsKey(general_group2.idgen))
					{
						flag = false;
						break;
					}
					if (!Global.character_.generalbooks[general_group2.idgen].finish)
					{
						flag = false;
						break;
					}
				}
			}
		}
		GeneralBookUI component = BaseUI.GetRootUI().GetComponent<GeneralBookUI>();
		if (flag && num != -1)
		{
			component.MyLoading(LabelTypeBack, book_gidBack, num, flag);
		}
		else
		{
			component.MyLoading(LabelTypeBack, book_gidBack);
		}
		MainUI.SetSubUI(component);
		BattleEndReturn();
	}

	}

	private bool check_recruit_gen(int general_id)
	{
		_general_origin general_origin = Global.getTemplate("general_origin", (uint)general_id) as _general_origin;
		List<ushort> list = new List<ushort>();
		if (general_origin != null)
		{
			Hashtable templates = Global.getTemplates("general_origin");
			foreach (DictionaryEntry item in templates)
			{
				_general_origin general_origin2 = item.Value as _general_origin;
				if (general_origin2.origin == general_origin.origin && general_origin2.form > general_origin.form)
				{
					list.Add(general_origin2.idgeneral_template);
				}
			}
		}
		list.Add((ushort)general_id);
		foreach (KeyValuePair<ushort, FormGeneral> item2 in Global.character_.form_generals_)
		{
			if (list.Contains(item2.Value.idgeneral_template))
			{
				return true;
			}
		}
		foreach (GeneralsSimple item3 in Global.character_.GeneralsOutSquad)
		{
			if (list.Contains((ushort)item3.idtemplate))
			{
				return true;
			}
		}
		return false;
	}

	public static int GeneralBookCount()
	{
		CheckGeneralbook(out var allgens);
		int num = 0;
		foreach (int item in allgens)
		{
			if (!(Global.getTemplate("general_book_template", (uint)item) is _general_book_template))
			{
				continue;
			}
			if (Global.character_.generalbooks.ContainsKey((ushort)item))
			{
				if (!Global.character_.generalbooks[(ushort)item].finish)
				{
					num++;
				}
			}
			else
			{
				num++;
			}
		}
		return num;
	}

	public static void CheckGeneralbook(out List<int> allgens)
	{
		allgens = new List<int>();
		AnecdoteSelectUI component = GameObject.Find("/Anecdote/AnecdoteUI").GetComponent<AnecdoteSelectUI>();
		if (component != null)
		{
			component._makeCampaigns();
		}
		List<uint> all_chapters = component.get_all_chapters();
		CampaignChapterUI component2 = GameObject.Find("/Campaign/CampaignUI").GetComponent<CampaignChapterUI>();
		if (component2 != null)
		{
			component2._makeCampaigns();
		}
		List<uint> all_chapets = component2.get_all_chapets();
		List<uint> list = new List<uint>();
		foreach (uint item in all_chapters)
		{
			list.Add(item);
		}
		foreach (uint item2 in all_chapets)
		{
			list.Add(item2);
		}
		Hashtable templates = Global.getTemplates("general_template");
		foreach (DictionaryEntry item3 in templates)
		{
			_general_template general_template = item3.Value as _general_template;
			if (general_template.active && general_template.special_quest == 0 && general_template.special_event == 0 && general_template.special_nation == 0 && Global.character_.check_genshow(general_template) && list.Contains(general_template.idcampaign))
			{
				if (!allgens.Contains(general_template.id))
				{
					allgens.Add(general_template.id);
				}
			}
			else if (general_template.special_event > 0 && general_template.active && Global.character_.check_special(general_template.id))
			{
				if (!allgens.Contains(general_template.id))
				{
					allgens.Add(general_template.id);
				}
			}
			else if (general_template.special_nation > 0 && general_template.active && Global.character_.get_nation_rank_self() >= general_template.special_nation && !allgens.Contains(general_template.id))
			{
				allgens.Add(general_template.id);
			}
		}
	}

	public static bool CheckGeneralbook(int go_id)
	{
		CheckGeneralbook(out var allgens);
		if (allgens.Contains(go_id))
		{
			if (!Global.character_.generalbooks.ContainsKey((ushort)go_id))
			{
				return true;
			}
			if (!Global.character_.generalbooks[(ushort)go_id].finish)
			{
				return true;
			}
		}
		return false;
	}
