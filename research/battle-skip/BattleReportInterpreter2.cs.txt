	private class FlyEffect
	{
		public GameObject effect_obj;

		private List<Vector3> target_pos;

		private int cur_target;

		private float speed;

		private float destroy_delay;

		private Vector3 velocity;

		public uint effect_id;

		public FlyEffect(uint id, GameObject obj, List<Vector3> target_list, float _speed, float _destroy_delay)
		{
			effect_id = id;
			effect_obj = obj;
			target_pos = target_list;
			speed = _speed;
			destroy_delay = _destroy_delay;
			cur_target = 0;
			effect_obj.transform.LookAt(target_list[0]);
			effect_obj.BroadcastMessage("Emit");
		}

		public float GetFlyDuration()
		{
			velocity = target_pos[cur_target] - effect_obj.transform.position;
			velocity.Normalize();
			velocity *= speed;
			return (target_pos[cur_target] - effect_obj.transform.position).magnitude / speed;
		}

		public bool NextTarget()
		{
			cur_target++;
			if (cur_target < target_pos.Count)
			{
				effect_obj.transform.LookAt(target_pos[cur_target]);
				return true;
			}
			UnityEngine.Object.Destroy(effect_obj, destroy_delay);
			return false;
		}

		public void UpdatePosition()
		{
			effect_obj.transform.position += velocity * Time.deltaTime;
		}

		public bool Flying()
		{
			return cur_target < target_pos.Count;
		}

		public void Destroy()
		{
			effect_obj.BroadcastMessage("Stop");
			UnityEngine.Object.Destroy(effect_obj, destroy_delay);
		}
	}

	private class EffectTimer
	{
		public float start_time;

		public float time_span;

		public int index;

		public bool attacker;

		public uint effect_id;

		public EffectTimer(int loc, bool att, uint id, float span)
		{
			start_time = Time.time;
			time_span = span;
			index = loc;
			attacker = att;
			effect_id = id;
		}

		public bool isFinished()
		{
			if (time_span < 0f)
			{
				return false;
			}
			return Time.time - start_time > time_span;
		}

		public bool isForever()
		{
			return time_span < 0f;
		}
	}

	private class Banner
	{
		private string msg;

		private Vector2 cur_pos;

		private Vector2 end_pos;

		private float speed;

		private float slideInTime;

		private float holdTime;

		private float fadeTime;

		private float start_time;

		private Color color = new Color(1f, 1f, 1f, 1f);

		private bool fading;

		public Banner(string _msg, Vector2 _start, Vector2 _end, float _slide, float _hold, float _fade)
		{
			msg = _msg;
			cur_pos = _start;
			end_pos = _end;
			slideInTime = _slide;
			holdTime = _hold;
			fadeTime = _fade;
			start_time = Time.time;
		}

		public bool Update()
		{
			float num = Time.time - start_time;
			if (num < slideInTime)
			{
				cur_pos.x = Mathf.SmoothDamp(cur_pos.x, end_pos.x, ref speed, slideInTime - num);
			}
			else if (num > slideInTime + holdTime)
			{
				fading = true;
				color.a = 1f - (num - slideInTime - holdTime) / fadeTime;
			}
			return num < slideInTime + holdTime + fadeTime;
		}

		public void Draw(GUIStyle style, GUIStyle text_style)
		{
			if (fading)
			{
				GUI.color = color;
			}
			else
			{
				GUI.color = Color.white;
			}
			GUI.Box(new Rect(cur_pos.x, cur_pos.y, 0f, 0f), string.Empty, style);
			GUI.Label(new Rect(cur_pos.x, cur_pos.y + 4f, 0f, 0f), msg, text_style);
		}
	}

	public void setReport(byte[] report_string)
	{
		clearAnimationTable();
		m_report = null;
		m_turn_enumerator = null;
		m_action_enumerator = null;
		m_finished = false;
		Logger.Log("report");
		if (report_string == null)
		{
			Logger.Log("NULL");
		}
		else
		{
			Logger.Log(report_string.Length.ToString());
		}
		if (report_string != null)
		{
			MemoryStream stream = new MemoryStream(report_string);
			try
			{
				DeflateStream inputStream = new DeflateStream(stream, CompressionMode.Decompress);
				m_report = JSProtoBuf.Serializer.Deserialize<BattleReport>(inputStream);
				Logger.Log(m_report.battle_field.ToString());
			}
			catch (Exception ex)
			{
				Logger.Log("catch zlib exception");
				Logger.Log(ex.Message);
			}
			clearAllCallbacks();
		}
	}

	public void Update()
	{
		updateEffectTimer();
		updateFlyEffect();
		updateBanners();
		if (m_cameraShake && m_shakeCamera != null)
		{
			if (Time.time < m_cameraShakeStart + m_cameraShakeDuration)
			{
				AccumulateCameraShakeVertical(0.3f * Mathf.Sin((float)Math.PI * 2f / m_cameraShakeDuration * (float)m_cameraShakeTimes * (Time.time - m_cameraShakeStart)));
			}
			else
			{
				m_cameraShake = false;
			}
		}
	}

	public void LateUpdate()
	{
		if (m_shakeCamera == null)
		{
			getShakeCamera();
		}
		if (m_shakeCamera != null)
		{
			if (m_shakeCameraOffset.sqrMagnitude > 0.001f)
			{
				m_shakeCamera.transform.position = m_cameraPositionOrigin + m_shakeCameraOffset;
			}
			else
			{
				m_shakeCamera.transform.position = m_cameraPositionOrigin;
			}
		}
		m_shakeCameraOffset = Vector3.zero;
	}

	private void getShakeCamera()
	{
		if (m_Battleground != null)
		{
			Camera[] componentsInChildren = m_Battleground.GetComponentsInChildren<Camera>();
			if (componentsInChildren.Length == 1)
			{
				m_shakeCamera = componentsInChildren[0];
				m_cameraPositionOrigin = m_shakeCamera.transform.position;
			}
		}
	}

	private void AccumulateCameraShake(Vector3 offset)
	{
		if (m_shakeCameraOffset.sqrMagnitude < 0.001f)
		{
			m_shakeCameraOffset = offset;
		}
	}

	public void AccumulateCameraShakeVertical(float v)
	{
		if (m_shakeCamera != null)
		{
			AccumulateCameraShake(m_shakeCamera.transform.up * v);
		}
	}

	public void AccumulateCameraShakeHorizontal(float v)
	{
		if (m_shakeCamera != null)
		{
			AccumulateCameraShake(m_shakeCamera.transform.right * v);
		}
	}

	public bool fight()
	{
		ExecLegionAction();
		while (!_MainActionQueue.isWaiting())
		{
			if (_MainActionQueue.isActionQueued())
			{
				_MainActionQueue.ExecQueuedAction();
				continue;
			}
			if (!m_finished && !bLegionBlocking)
			{
				TryMoveNext();
			}
			if (m_finished)
			{
				_MainActionQueue.AddAction(new WaitAction2(1.5f));
				break;
			}
			if (bLegionBlocking = isBlocked())
			{
				break;
			}
			float num = Handle();
			switch (CurrentStepMode)
			{
			case StepMode.SingleStep:
				_MainActionQueue.Wait(num);
				break;
			case StepMode.Skill:
				_MainActionQueue.Wait(num);
				break;
			case StepMode.Batch:
				if (num > BatchWaitMax)
				{
					BatchWaitMax = num;
				}
				break;
			}
		}
		return !m_finished;
	}

	public bool TryMoveNext()
	{
		if (m_turn_enumerator == null)
		{
			m_finished = true;
			return false;
		}
		while (m_action_enumerator == null || !m_action_enumerator.MoveNext())
		{
			if (m_turn_enumerator.MoveNext())
			{
				OnNewTurn(m_turn_enumerator.Current as BattleReport.GameTurn);
				m_action_enumerator = (m_turn_enumerator.Current as BattleReport.GameTurn).actions.GetEnumerator();
				continue;
			}
			m_finished = true;
			return false;
		}
		return true;
	}

	public virtual bool isBlocked()
	{
		BattleReport.GameTurn.Action currentAction = CurrentAction;
		bool flag = currentAction.side == BattleReport.GameTurn.Action.LegionSide.LEGION_LEFT;
		int location = currentAction.location;
		switch (currentAction.type)
		{
		case BattleReport.GameTurn.Action.ActionType.skillActionEnd:
		case BattleReport.GameTurn.Action.ActionType.targetHandleBegin:
		case BattleReport.GameTurn.Action.ActionType.targetHandleEnd:
		case BattleReport.GameTurn.Action.ActionType.one_pass_begin:
		case BattleReport.GameTurn.Action.ActionType.one_pass_end:
		case BattleReport.GameTurn.Action.ActionType.block:
		case BattleReport.GameTurn.Action.ActionType.beHit:
		case BattleReport.GameTurn.Action.ActionType.addRage:
		case BattleReport.GameTurn.Action.ActionType.buffOperation:
		case BattleReport.GameTurn.Action.ActionType.dead:
		case BattleReport.GameTurn.Action.ActionType.revive:
			return false;
		case BattleReport.GameTurn.Action.ActionType.move:
			if (isLegionBlocked(flag, location))
			{
				return true;
			}
			if ((flag && right_turn_move == 0 && !currentAction.is_sub_skill) || (!flag && left_turn_move == 0 && !currentAction.is_sub_skill))
			{
				return false;
			}
			if (currentAction.target_locations.Count > 0)
			{
				for (int i = 0; i < currentAction.target_locations.Count; i++)
				{
					if (isLegionBlocked(currentAction.target_side == BattleReport.GameTurn.Action.LegionSide.LEGION_LEFT, currentAction.target_locations[i]))
					{
						return true;
					}
				}
			}
			return false;
		case BattleReport.GameTurn.Action.ActionType.skillHandleEnd:
			if ((flag && right_turn_move == 0 && !currentAction.is_sub_skill) || (!flag && left_turn_move == 0 && !currentAction.is_sub_skill))
			{
				return false;
			}
			return isLegionBlocked(flag, location);
		default:
			return isLegionBlocked(flag, location);
		}
	}

	public void SetArmyToFinalState()
	{
		if (_LegionActionQueue[0] != null)
		{
			_LegionActionQueue[0].Clear();
		}
		if (_LegionActionQueue[1] != null)
		{
			_LegionActionQueue[1].Clear();
		}
		_MainActionQueue.Clear();
		for (int i = 0; i < m_report.left_army.legions.Count; i++)
		{
			if (m_ArmyBehaviour[0] != null)
			{
				m_ArmyBehaviour[0].SetToFinal(m_report.left_army.legions[i].location, m_report.left_army.legions[i]);
			}
		}
		for (int j = 0; j < m_report.right_army.legions.Count; j++)
		{
			if (m_ArmyBehaviour[1] != null)
			{
				m_ArmyBehaviour[1].SetToFinal(m_report.right_army.legions[j].location, m_report.right_army.legions[j]);
			}
		}
	}

	public virtual void Destroy()
	{
		RemoveFlyEffect();
		ClearActionQueue();
		clearEffectTimer();
		soundEffectDict.Clear();
		if (m_ArmyBehaviour[0] != null && m_ArmyBehaviour[0].gameObject != null)
		{
			UnityEngine.Object.Destroy(m_ArmyBehaviour[0].gameObject);
		}
		if (m_ArmyBehaviour[1] != null && m_ArmyBehaviour[1].gameObject != null)
		{
			UnityEngine.Object.Destroy(m_ArmyBehaviour[1].gameObject);
		}
		m_ArmyBehaviour[0] = null;
		m_ArmyBehaviour[1] = null;
		UnityEngine.Object.Destroy(m_Battleground);
		foreach (KeyValuePair<string, GameObject> item in m_EffectGround)
		{
			UnityEngine.Object.Destroy(item.Value);
		}
		m_EffectGround.Clear();
	}

	public float Handle()
	{
		BattleReport.GameTurn.Action currentAction = CurrentAction;
		bool b_attacker = currentAction.side == BattleReport.GameTurn.Action.LegionSide.LEGION_LEFT;
		int location = currentAction.location;
		float result = 0f;
		if (StepHandleDictionary.ContainsKey(currentAction.type))
		{
			result = StepHandleDictionary[currentAction.type].handle(b_attacker, location, currentAction);
			if (StepHandleDictionary[currentAction.type].callback != null)
			{
				StepHandleDictionary[currentAction.type].callback(b_attacker, location, currentAction);
			}
		}
		return result;
	}

	public _skill_animation getSkillAnimation(bool b_attacker, int actor_loc, uint skill_id)
	{
		int num = ((!b_attacker) ? 1 : 0);
		if (!skill_animation_table[num].ContainsKey(actor_loc))
		{
			skill_animation_table[num][actor_loc] = new Dictionary<uint, _skill_animation>();
		}
		if (!skill_animation_table[num][actor_loc].ContainsKey(skill_id))
		{
			BattleReport.ArmyInfo army_info = ((!b_attacker) ? m_report.right_army : m_report.left_army);
			skill_animation_table[num][actor_loc][skill_id] = getSkillAnimation(army_info, actor_loc, skill_id);
		}
		if (skill_animation_table[num][actor_loc][skill_id] == null)
		{
			return null;
		}
		_skill_animation skill_animation = new _skill_animation();
		if (!ShowEffect)
		{
			skill_animation.Set(Global.getTemplate("skill_animation", 0u) as _skill_animation);
			skill_animation.skill_handle_animation = skill_animation_table[num][actor_loc][skill_id].skill_handle_animation;
			skill_animation.skill_move_type = skill_animation_table[num][actor_loc][skill_id].skill_move_type;
		}
		else
		{
			skill_animation.Set(skill_animation_table[num][actor_loc][skill_id]);
		}
		return skill_animation;
	}

	public static _skill_animation getSkillAnimation(BattleReport.ArmyInfo army_info, int actor_loc, uint skill_id)
	{
		if (actor_loc == 0)
		{
			return getSkillAnimation(army_info.beast, skill_id);
		}
		for (int i = 0; i < army_info.legions.Count; i++)
		{
			if (army_info.legions[i].location == actor_loc)
			{
				return getSkillAnimation(army_info.legions[i], skill_id);
			}
		}
		return null;
	}

	private static _skill_animation getSkillAnimation(BattleReport.ArmyInfo.LegionInfo legion, uint skill_id)
	{
		if (legion.isMainCharacter)
		{
			Hashtable templates = Global.getTemplates("skill_animation_override");
			if (!legion.male_model)
			{
				foreach (DictionaryEntry item in templates)
				{
					_skill_animation_override skill_animation_override = item.Value as _skill_animation_override;
					if (skill_animation_override.idskill == skill_id && skill_animation_override.variation == 2)
					{
						return Global.getTemplate("skill_animation", skill_animation_override.idskill_animation) as _skill_animation;
					}
				}
			}
		}
		if (!(Global.getTemplate("skill", skill_id) is _skill skill))
		{
			return null;
		}
		return Global.getTemplate("skill_animation", skill.idskill_animation) as _skill_animation;
	}

	private static _skill_animation getSkillAnimation(BattleReport.ArmyInfo.SpiritBeastInfo beast, uint skill_id)
	{
		Hashtable templates = Global.getTemplates("skill_animation_override");
		foreach (DictionaryEntry item in templates)
		{
			_skill_animation_override skill_animation_override = item.Value as _skill_animation_override;
			if (skill_animation_override.idskill == skill_id && skill_animation_override.variation == 3)
			{
				return Global.getTemplate("skill_animation", skill_animation_override.idskill_animation) as _skill_animation;
			}
		}
		if (!(Global.getTemplate("skill", skill_id) is _skill skill))
		{
			return null;
		}
		return Global.getTemplate("skill_animation", skill.idskill_animation) as _skill_animation;
	}

	private void addArmyAsset(bool b_attacker, BattleReport.ArmyInfo army_info, bool brawl_mode)
	{
		List<uint> list = new List<uint>();
		List<int> skill_id_list = new List<int>();
		foreach (BattleReport.ArmyInfo.LegionInfo legion in army_info.legions)
		{
			if (Global.getTemplate("army_template", (uint)legion.army_id) is _army_template army_template)
			{
				string text = ((!legion.male_model && army_template.model_name_female.Length != 0) ? army_template.model_name_female : army_template.model_name_male);
				assetBehaviour.AddAsset("battle/" + text, bForce: false);
			}
			else
			{
				Logger.Log("can't find army template " + legion.army_id);
			}
			assetBehaviour.AddAsset("face" + legion.icon, bForce: false);
			for (int i = 0; i < legion.used_armys.Count; i++)
			{
				if (Global.getTemplate("army_template", (uint)legion.used_armys[i]) is _army_template army_template2)
				{
					string text2 = ((!legion.male_model && army_template2.model_name_female.Length != 0) ? army_template2.model_name_female : army_template2.model_name_male);
					assetBehaviour.AddAsset("battle/" + text2, bForce: false);
				}
			}
			if (!brawl_mode)
			{
				for (int j = 0; j < legion.used_skills.Count; j++)
				{
					addSkillEffectAsset(b_attacker, legion.location, legion.used_skills[j], list, skill_id_list);
				}
				for (int k = 0; k < legion.used_buffs.Count; k++)
				{
					addBuffEffectAsset(legion.used_buffs[k], list);
				}
				for (int l = 0; l < legion.used_damages.Count; l++)
				{
					addDamageEffectAsset(legion.used_damages[l], list);
				}
			}
		}
		spirit_tt[(!b_attacker) ? 1u : 0u] = null;
		if (army_info.beast != null)
		{
			if (Global.getTemplate("spirit_beast", army_info.beast.id) is _spirit_beast spirit_beast)
			{
				assetBehaviour.AddAsset("battle/" + spirit_beast.model_name, bForce: false);
			}
			if (!brawl_mode)
			{
				for (int m = 0; m < army_info.beast.skill_id.Count; m++)
				{
					addSkillEffectAsset(b_attacker, 0, army_info.beast.skill_id[m], list, skill_id_list);
				}
				for (int n = 0; n < army_info.beast.used_buffs.Count; n++)
				{
					addBuffEffectAsset(army_info.beast.used_buffs[n], list);
				}
				for (int num = 0; num < army_info.beast.used_damages.Count; num++)
				{
					addDamageEffectAsset(army_info.beast.used_damages[num], list);
				}
			}
			spirit_tt[(!b_attacker) ? 1u : 0u] = new SpiritBeastTooltip(army_info.beast);
			SpiritBeastTooltip spiritBeastTooltip = spirit_tt[(!b_attacker) ? 1u : 0u];
			for (int num2 = 0; num2 < spiritBeastTooltip.skills.Count; num2++)
			{
				if (spiritBeastTooltip.skills[num2] != null)
				{
					assetBehaviour.AddAssetSub(spiritBeastTooltip.skills[num2].icon_name);
				}
			}
		}
		if (!list.Contains(27u))
		{
			list.Add(27u);
		}
		if (!list.Contains(28u))
		{
			list.Add(28u);
		}
		for (int num3 = 0; num3 < list.Count; num3++)
		{
			if (Global.getTemplate("battle_effect", list[num3]) is _battle_effect battle_effect && battle_effect.name.Length > 0)
			{
				assetBehaviour.AddAsset("battle/" + battle_effect.name, bForce: false);
			}
		}
	}

	private void addSkillEffectAsset(bool b_attacker, int actor_loc, int skill_id, List<uint> effect_id_list, List<int> skill_id_list)
	{
		if (skill_id_list.Contains(skill_id))
		{
			return;
		}
		skill_id_list.Add(skill_id);
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)skill_id);
		if (skillAnimation != null)
		{
			if (!effect_id_list.Contains(skillAnimation.skill_action_effect))
			{
				effect_id_list.Add(skillAnimation.skill_action_effect);
			}
			if (!effect_id_list.Contains(skillAnimation.skill_handle_effect))
			{
				effect_id_list.Add(skillAnimation.skill_handle_effect);
			}
			if (!effect_id_list.Contains(skillAnimation.skill_fly_effect_id))
			{
				effect_id_list.Add(skillAnimation.skill_fly_effect_id);
			}
			if (!effect_id_list.Contains(skillAnimation.skill_attack_effect))
			{
				effect_id_list.Add(skillAnimation.skill_attack_effect);
			}
			if (!effect_id_list.Contains(skillAnimation.skill_hit_effect))
			{
				effect_id_list.Add(skillAnimation.skill_hit_effect);
			}
			if (!effect_id_list.Contains(skillAnimation.skill_transform_effect))
			{
				effect_id_list.Add(skillAnimation.skill_transform_effect);
			}
			if (skillAnimation.ground_setting != string.Empty)
			{
				assetBehaviour.AddAsset("battle/" + skillAnimation.ground_setting, bForce: false);
			}
		}
	}

	private void addBuffEffectAsset(uint buff_id, List<uint> effect_id_list)
	{
		if (Global.getTemplate("buff_table", buff_id) is _buff_table buff_table)
		{
			if (!effect_id_list.Contains(buff_table.effect_id))
			{
				effect_id_list.Add(buff_table.effect_id);
			}
			if (!effect_id_list.Contains(buff_table.trigger_effect_id))
			{
				effect_id_list.Add(buff_table.trigger_effect_id);
			}
			if (!effect_id_list.Contains(buff_table.hit_effect_id))
			{
				effect_id_list.Add(buff_table.hit_effect_id);
			}
		}
	}

	private void addDamageEffectAsset(uint damage_type, List<uint> effect_id_list)
	{
		switch (damage_type)
		{
		case 4u:
			if (!effect_id_list.Contains(4u))
			{
				effect_id_list.Add(4u);
			}
			break;
		case 5u:
			if (!effect_id_list.Contains(24u))
			{
				effect_id_list.Add(24u);
			}
			break;
		}
	}

	public virtual void InitTroop(bool invert_side)
	{
		UnityEngine.Object.Destroy(m_Battleground);
		foreach (KeyValuePair<string, GameObject> item in m_EffectGround)
		{
			UnityEngine.Object.Destroy(item.Value);
		}
		m_EffectGround.Clear();
		RemoveFlyEffect();
		ClearActionQueue();
		clearEffectTimer();
		current_pass_troop_anim = null;
		string text = "battle" + m_report.battle_field;
		m_Battleground = BaseAsset.Instantiate_mainAsset("battle/" + text) as GameObject;
		if (m_Battleground == null)
		{
			m_Battleground = new GameObject();
		}
		if (m_Battleground.GetComponent<CutSceneHelper>() != null)
		{
			m_Battleground.GetComponent<CutSceneHelper>().enabled = false;
		}
		m_Battleground.transform.parent = this.gameObject.transform;
		Camera componentInChildren = m_Battleground.GetComponentInChildren<Camera>();
		if (componentInChildren != null)
		{
			if ((bool)componentInChildren.gameObject.GetComponent<AudioListener>())
			{
				componentInChildren.gameObject.GetComponent<AudioListener>().enabled = false;
			}
			GameObject gameObject = new GameObject("listener");
			gameObject.transform.parent = componentInChildren.transform;
			gameObject.transform.localRotation = Quaternion.identity;
			gameObject.transform.localPosition = new Vector3(0f, 0f, 60f);
			gameObject.AddComponent<AudioListener>();
		}
		for (int i = 0; i < 2; i++)
		{
			m_ArmyBehaviour[i] = new GameObject().AddComponent<ArmyBehaviour>();
			m_ArmyBehaviour[i].transform.parent = m_Battleground.transform;
			int num = ((!invert_side) ? i : (1 - i));
			m_ArmyBehaviour[i].SetSide(num);
			m_ArmyBehaviour[i].transform.localPosition = new Vector3(0f, 0f, 8 * ((num == 0) ? 1 : (-1)));
			m_ArmyBehaviour[i].transform.localEulerAngles = new Vector3(0f, (num == 0) ? 180 : 0, 0f);
			m_ArmyBehaviour[i].gameObject.layer = 9;
			if (getSpiritTooltip(i) != null)
			{
				getSpiritTooltip(i).skill_used = 0;
			}
		}
		Camera componentInChildren2 = m_Battleground.GetComponentInChildren<Camera>();
		if (componentInChildren2 != null)
		{
			m_cameraPositionOrigin = componentInChildren2.transform.position;
		}
		m_ArmyBehaviour[0].Init(packet.left_army, 0, brawl_mode: false);
		m_ArmyBehaviour[1].Init(packet.right_army, 0, brawl_mode: false);
		bLegionBlocking = false;
		playedTurnCount = 0;
		Reset();
	}

	public virtual void AddEffect(int location, bool isAttacker, uint id, float span, bool attach_to_unit)
	{
		if (!bShowEffect || !(Global.getTemplate("battle_effect", id) is _battle_effect battle_effect) || battle_effect.name.Length <= 0)
		{
			return;
		}
		int num = hasEffectTimed(location, isAttacker, id);
		if (num != -1)
		{
			if (m_EffectControlList[num].time_span + m_EffectControlList[num].start_time < Time.time + span)
			{
				m_EffectControlList[num].time_span = span;
				m_EffectControlList[num].start_time = Time.time;
			}
		}
		else
		{
			m_EffectControlList.Add(new EffectTimer(location, isAttacker, id, span));
		}
		float size_scale = 1f;
		m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].AddEffect(location, battle_effect.name, battle_effect.speed_scale, size_scale, battle_effect.area_type, attach_to_unit);
	}

	public virtual void AddEffect(int location, bool isAttacker, uint id, float span, bool attach_to_unit, string se_name)
	{
		float size_scale = 1f;
		_battle_effect battle_effect = null;
		if (bShowEffect)
		{
			battle_effect = Global.getTemplate("battle_effect", id) as _battle_effect;
		}
		int num = hasEffectTimed(location, isAttacker, id);
		if (num != -1)
		{
			if (m_EffectControlList[num].time_span + m_EffectControlList[num].start_time < Time.time + span)
			{
				m_EffectControlList[num].time_span = span;
				m_EffectControlList[num].start_time = Time.time;
			}
		}
		else
		{
			m_EffectControlList.Add(new EffectTimer(location, isAttacker, id, span));
		}
		object obj;
		if (se_name == null)
		{
			obj = null;
		}
		else if (soundEffectDict.ContainsKey(se_name))
		{
			obj = soundEffectDict[se_name];
		}
		else
		{
			AudioClip audioClip = BaseAsset.LoadAsset_named("battle/battle_sound", se_name) as AudioClip;
			soundEffectDict[se_name] = audioClip;
			obj = audioClip;
		}
		AudioClip se_clip = (AudioClip)obj;
		if (battle_effect == null)
		{
			m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].AddEffect(location, null, 0f, 1f, 2, attach_to_unit, se_clip);
		}
		else
		{
			m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].AddEffect(location, battle_effect.name, battle_effect.speed_scale, size_scale, battle_effect.area_type, attach_to_unit, se_clip);
		}
	}

	protected virtual int hasEffectTimed(int loc, bool att, uint id)
	{
		for (int i = 0; i < m_EffectControlList.Count; i++)
		{
			if (m_EffectControlList[i].index == loc && m_EffectControlList[i].attacker == att && m_EffectControlList[i].effect_id == id)
			{
				return i;
			}
		}
		return -1;
	}

	private void updateEffectTimer()
	{
		for (int i = 0; i < m_EffectControlList.Count; i++)
		{
			if (m_EffectControlList[i].isFinished())
			{
				RemoveEffect(m_EffectControlList[i].index, m_EffectControlList[i].attacker, m_EffectControlList[i].effect_id);
				m_EffectControlList.RemoveAt(i);
				i--;
			}
		}
	}

	}

	public virtual void AddFlyEffect(uint eff_id, int attacker_loc, bool attacker_side, List<int> target_loc, bool target_side, float speed, float destroy_delay)
	{
		if (!(Global.getTemplate("battle_effect", eff_id) is _battle_effect battle_effect) || battle_effect.name.Length <= 0)
		{
			return;
		}
		GameObject gameObject = BaseAsset.Instantiate_mainAsset("battle/" + battle_effect.name) as GameObject;
		if (gameObject == null)
		{
			gameObject = new GameObject();
		}
		BaseAsset.SetLayerRecursively(gameObject, 9);
		Vector3 localPosition = gameObject.transform.localPosition;
		Quaternion localRotation = gameObject.transform.localRotation;
		gameObject.transform.parent = m_ArmyBehaviour[(!attacker_side) ? 1u : 0u].getLegionTransform(attacker_loc);
		gameObject.transform.localPosition = localPosition;
		gameObject.transform.localRotation = localRotation;
		gameObject.transform.parent = this.gameObject.transform;
		gameObject.AddComponent<BasicEffectBehaviour>();
		List<Vector3> list = new List<Vector3>();
		if (battle_effect.area_type == 2)
		{
			list.Add(m_ArmyBehaviour[(!target_side) ? 1u : 0u].getLegionOriginPos(-1));
		}
		else
		{
			for (int i = 0; i < target_loc.Count; i++)
			{
				list.Add(m_ArmyBehaviour[(!target_side) ? 1u : 0u].getLegionOriginPos(target_loc[i]));
			}
		}
		m_flyEffect = new FlyEffect(eff_id, gameObject, list, speed, destroy_delay);
	}

	public virtual float skillAction(bool b_attacker, int actor_loc, BattleReport.GameTurn.Action action)
	{
		if (!action.is_normal && Global.getTemplate("skill", (uint)action.skill_id) is _skill skill)
		{
			DoShowBanner(b_attacker, skill.name, 0.3f, 1f, 1f);
		}
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)action.skill_id);
		if (skillAnimation != null && skillAnimation.ground_setting.Length > 0)
		{
			add_setup_ground(skillAnimation.ground_setting);
		}
		if (b_attacker)
		{
			left_turn_move--;
		}
		else
		{
			right_turn_move--;
		}
		return 0f;
	}

	public virtual float skillActionEnd(bool b_attacker, int actor_loc, BattleReport.GameTurn.Action action)
	{
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)action.skill_id);

	public virtual float block(bool b_attacker, int actor_loc, BattleReport.GameTurn.Action action)
	{
		int hit_point = action.hit_point;
		bool critical = action.critical;
		int absorb_point = action.absorb_point;
		for (int i = 0; i < action.buff_effects.Count; i++)
		{
			if (Global.getTemplate("buff_table", action.buff_effects[i]) is _buff_table { hit_effect_id: >0 } buff_table)
			{
				AddEffect(actor_loc, b_attacker, buff_table.hit_effect_id, 1f, attach_to_unit: true);
			}
		}
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)action.skill_id);
		if (skillAnimation != null)
		{
			string text = ((Global.getTemplate("skill_sound", skillAnimation.id) is _skill_sound skill_sound) ? skill_sound.hit_effect : string.Empty);
			if (text.Length == 0)
			{
				if (current_pass_troop_anim != null)
				{
					switch (skillAnimation.skill_handle_animation)
					{
					case 0:
						text = current_pass_troop_anim.hit_sound01;
						break;
					case 1:
						text = current_pass_troop_anim.hit_sound02;
						break;
					case 2:
						text = current_pass_troop_anim.hit_sound03;
						break;
					case 3:
						text = current_pass_troop_anim.hit_sound04;
						break;
					}
				}
			}
			else if (text == "-")
			{
				text = null;
			}
			AddMainAction(new EffectAction2(actor_loc, b_attacker, skillAnimation.skill_hit_effect, skillAnimation.skill_hit_effect_time, attach_to_unit: true, text));
		}
		doShowInfo(BattleShowInfo.Block, actor_loc, b_attacker);
		return doBeHit(actor_loc, b_attacker, hit_point, absorb_point, critical, b_executed: false, skillAnimation);
	}

		if (skillAnimation != null)

			string text = ((Global.getTemplate("skill_sound", skillAnimation.id) is _skill_sound skill_sound) ? skill_sound.hit_effect : string.Empty);
			if (text.Length == 0)

				if (current_pass_troop_anim != null)
				{
					switch (skillAnimation.skill_handle_animation)
					{
					case 0:
						text = current_pass_troop_anim.hit_sound01;

						break;
					case 2:

	public virtual float buffOperation(bool b_attacker, int actor_loc, BattleReport.GameTurn.Action action)
	{
		ushort num = (ushort)action.buff;
		_buff_table buff_table = Global.getTemplate("buff_table", num) as _buff_table;
		if (buff_table == null)
		{
			Debug.Log("no buff id in buff_table:" + num);
		}
		if (action.buff_op == BattleReport.GameTurn.Action.BuffOperationType.add_buff)
		{
			SetBuff(actor_loc, b_attacker, num, flag: true);
			if (buff_table != null)
			{
				if (buff_table.trigger_effect_id > 0)
				{
					AddEffect(actor_loc, b_attacker, buff_table.trigger_effect_id, 1f, attach_to_unit: true);
				}
				if (buff_table.effect_id > 0)
				{
					AddEffect(actor_loc, b_attacker, buff_table.effect_id, -1f, attach_to_unit: true);
				}
			}
			for (int i = 0; i < status_buff_id_list.Length; i++)
			{
				if (status_buff_id_list[i] == num)
				{
					doShowInfo(status_info_list[i], actor_loc, b_attacker);
					break;
				}
			}
		}
		else if (action.buff_op == BattleReport.GameTurn.Action.BuffOperationType.remove_buff)
		{
			SetBuff(actor_loc, b_attacker, num, flag: false);
			if (buff_table != null && buff_table.effect_id > 0)
			{
				AddEffect(actor_loc, b_attacker, 0u, -1f, attach_to_unit: true, buff_table.rem_sound);
				RemoveEffect(actor_loc, b_attacker, buff_table.effect_id);
			}
		}
		return -1f;
	}

	public virtual float actionMove(bool b_attacker, int actor_loc, BattleReport.GameTurn.Action action)
	{
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)action.skill_id);
		if (skillAnimation != null)
		{
			if (skillAnimation.skill_move_type == 3)

			}
			if (action.target_locations.Count > 0)
			{
				return moveAction(actor_loc, b_attacker, action.target_locations[0], action.target_side == BattleReport.GameTurn.Action.LegionSide.LEGION_LEFT, skillAnimation.skill_move_type);

	public float moveAction(int location, bool isAttacker, int target_location, bool target_isAttacker, byte move_type)
	{
		float result = -1f;
		Vector3 resultPos = new Vector3(0f, 0f, 0f);
		if (m_ArmyBehaviour[(!target_isAttacker) ? 1u : 0u].getTargetPosition(target_location, move_type, ref resultPos))
		{
			if (!m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].NeedMove(location, resultPos))
			{
				return 0f;
			}
			float wait_len = 0f;
			float move_len = 0f;
			float move_end = 0f;
			m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].GetMoveTime(location, ref wait_len, ref move_len, ref move_end);
			AddLegionAction(isAttacker, location, new TroopAnimAction2(location, isAttacker, TroopActionID.Jump));
			if (wait_len > 0f)
			{
				AddLegionAction(isAttacker, location, new WaitAction2(wait_len));
			}
			AddLegionAction(isAttacker, location, new MoveAction2(location, isAttacker, resultPos, move_len));
			if (move_len > 0f)
			{
				AddLegionAction(isAttacker, location, new WaitAction2(move_len));
			}
		}
		return result;
	}

	public float movebackAction(int location, bool isAttacker)
	{
		if (!m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].NeedMove(location, m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].getLegionOriginPos(location)))
		{
			return 0f;
		}
		float wait_len = 0f;
		float move_len = 0f;
		float move_end = 0f;
		m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].GetMoveTime(location, ref wait_len, ref move_len, ref move_end);
		AddLegionAction(isAttacker, location, new TroopAnimAction2(location, isAttacker, TroopActionID.Jump));
		if (wait_len > 0f)
		{
			AddLegionAction(isAttacker, location, new WaitAction2(wait_len));
		}
		AddLegionAction(isAttacker, location, new MoveAction2(location, isAttacker, m_ArmyBehaviour[(!isAttacker) ? 1u : 0u].getLegionOriginPos(location), move_len));
		if (move_len > 0f)
		{
			AddLegionAction(isAttacker, location, new WaitAction2(move_len));
		}
		return defaultTurnWaitSpan;
	}

	protected virtual float doActionHandle(int skill_id, int actor_loc, bool b_attacker, List<int> target_loc, bool target_attacker)
	{
		_skill_animation skillAnimation = getSkillAnimation(b_attacker, actor_loc, (uint)skill_id);
		if (skillAnimation != null)
		{
			_skill_sound skill_sound = Global.getTemplate("skill_sound", skillAnimation.id) as _skill_sound;
			if (skillAnimation.light_setting > 0)
			{
				ChangeSceneSetting(skillAnimation.light_setting);
			}
			if (b_attacker && skillAnimation.camera_setting1 > 0)
			{
				ChangeSceneSetting(skillAnimation.camera_setting1);
			}
			else if (!b_attacker && skillAnimation.camera_setting0 > 0)
			{
				ChangeSceneSetting(skillAnimation.camera_setting0);
			}
			AddEffect(actor_loc, b_attacker, skillAnimation.skill_handle_effect, skillAnimation.skill_handle_time, attach_to_unit: true, skill_sound?.handle);
			Hashtable hashtable = new Hashtable();
			hashtable["action"] = skillAnimation.skill_handle_animation;
			AddMainAction(new EffectAction2(actor_loc, b_attacker, skillAnimation.skill_attack_effect, skillAnimation.skill_attack_effect_time, attach_to_unit: false, skill_sound?.attack));
			if (skillAnimation.skill_fly_effect_id != 0)
			{
				AddMainAction(new WaitAction2(skillAnimation.skill_fly_effect_start_time));
				if (skillAnimation.skill_fly_effect_during_time == 0f)
				{
					for (int i = 0; i < target_loc.Count; i++)
					{
						AddMainAction(new EffectAction2(target_loc[i], target_attacker, skillAnimation.skill_fly_effect_id, skillAnimation.skill_fly_effect_time, attach_to_unit: true, skill_sound?.fly_effect));
					}
					AddMainAction(new WaitAction2(skillAnimation.skill_hit_delay_time));
				}
				if (skillAnimation.skill_fly_effect_during_time > 0f)
				{
					AddMainAction(new FlyEffectAction2(actor_loc, b_attacker, skillAnimation.skill_fly_effect_id, target_loc, target_attacker, skillAnimation.skill_fly_effect_during_time, skillAnimation.skill_fly_effect_time));
				}
			}
			else
			{
				AddMainAction(new WaitAction2(skillAnimation.skill_hit_delay_time));
			}
			return doAction(actor_loc, b_attacker, "ActionHandle", hashtable);
		}
		return 0f;
	}

	protected float doBeHit(int actor_loc, bool b_attacker, int point, int absorb_point, bool b_critical, bool b_executed, _skill_animation skill_anim)
	{
		Hashtable hashtable = new Hashtable();
		hashtable["point"] = point;
		hashtable["critical"] = b_critical;
		hashtable["executed"] = b_executed;
		hashtable["can_blow"] = skill_anim?.blow_away_when_dead ?? false;
		hashtable["absorb"] = absorb_point;
		if (skill_anim != null && skill_anim.damage_shake_camera)
		{
			BattleReport.ArmyInfo armyInfo = ((!b_attacker) ? m_report.right_army : m_report.left_army);
			for (int i = 0; i < armyInfo.legions.Count; i++)
			{
				if (armyInfo.legions[i].location == actor_loc)
				{
					if ((float)point >= (float)armyInfo.legions[i].troop_base_num * 0.2f)
					{
						ShakeCamera(0.2f, 2);
					}
					break;
				}
			}
		}
		if (point > 0)
		{
			AddEffect(actor_loc, b_attacker, (!b_critical) ? 27u : 28u, 1f, attach_to_unit: true);
		}
		return doAction(actor_loc, b_attacker, "beHit", hashtable);
	}

	protected void ClearActionQueue()
	{
		_MainActionQueue.Clear();
		for (int i = 0; i < 2; i++)
		{
			foreach (KeyValuePair<int, ActionQueue2> item in _LegionActionQueue[i])
			{
				item.Value.Clear();
			}
		}
	}

	public void AddMainAction(BattleAction2 action)
	{
		_MainActionQueue.AddAction(action);
	}

	public void AddLegionAction(bool b_attack, int index, BattleAction2 action)
	{
		int num = ((!b_attack) ? 1 : 0);
		if (!_LegionActionQueue[num].ContainsKey(index))
		{
			_LegionActionQueue[num][index] = new ActionQueue2(this);
		}
		_LegionActionQueue[num][index].AddAction(action);
	}

	public bool isLegionBlocked(bool b_attack, int index)
	{
		int num = ((!b_attack) ? 1 : 0);
		if (!_LegionActionQueue[num].ContainsKey(index))
		{
			return false;
		}
		return _LegionActionQueue[num][index].isActionQueued() || _LegionActionQueue[num][index].isWaiting();
	}

	public void ExecLegionAction()
	{
		for (int i = 0; i < 2; i++)
		{
			foreach (KeyValuePair<int, ActionQueue2> item in _LegionActionQueue[i])
			{
				while (!item.Value.isWaiting() && item.Value.isActionQueued())
				{
					item.Value.ExecQueuedAction();
				}
			}
		}
	}

	public bool allLegionFinished()
	{
		for (int i = 0; i < 2; i++)
		{
			foreach (KeyValuePair<int, ActionQueue2> item in _LegionActionQueue[i])
			{
				if (!item.Value.isFinished())
				{
					return false;
				}
			}
		}
		return true;
	}

	{
