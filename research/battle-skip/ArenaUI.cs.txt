	private void makeBonuses()
	{
		if (old_rank == m_arena.rank)
		{
			return;
		}
		old_rank = m_arena.rank;
		Hashtable templates = Global.getTemplates("arena_bonus");
		_arena_bonus arenaBonus = Database.Arena.getArenaBonus(m_arena.rank);
		if (arenaBonus != null)
		{
			cur_bonus = arenaBonus.id;
		}
		byte b = cur_bonus;
		if (b == 0)
		{
			b = 8;
		}
		if (b > 3)
		{
			if (b == 8)
			{
				cur_bonuses[0] = templates[0u] as _arena_bonus;
			}
			else
			{
				cur_bonuses[0] = templates[(uint)b] as _arena_bonus;
			}
			cur_bonuses[1] = templates[(uint)(b - 1)] as _arena_bonus;
			cur_bonuses[2] = templates[(uint)(b - 2)] as _arena_bonus;
		}
		else
		{
			cur_bonuses[0] = templates[3u] as _arena_bonus;
			cur_bonuses[1] = templates[2u] as _arena_bonus;
			cur_bonuses[2] = templates[1u] as _arena_bonus;
		}
	}

	private IEnumerator battle_view(ulong id)
	{
		WWWForm form = new WWWForm();
		if (Application.platform == RuntimePlatform.WindowsEditor || Application.platform == RuntimePlatform.Android || Application.platform == RuntimePlatform.IPhonePlayer)
		{
			form.AddField("XDEBUG_SESSION_START", "netbeans-xdebug");
		}
		form.AddField("idreport", id.ToString());
		WWW www = new WWW(ServerSetting.url_base + "battle_view_cmd.php", form);
		yield return www;
		try
		{
